79// Execute-based writes use the axis management-command lifecycle. Immediate
80// mode applies in the triggering scan; queued mode remains Busy until the axis
81// motion queue drains and AxisModel executes the write.
82class FbWriteParameter : public AxisManagementExecuteFb
83{
84 public:
85 axis::AxisParameter parameter_number = axis::AxisParameter::sw_limit_pos;
86 double value = 0.0;
87 axis::ExecutionMode execution_mode = axis::ExecutionMode::immediately;
88
89 void call()
90 {
91 if(rising_edge()) {
92 if(axis_ref == nullptr) {
93 accept_management(rt::Result<std::uint32_t>::failure(
94 rt::ErrorCode::invalid_argument));
95 } else if(execution_mode != axis::ExecutionMode::immediately &&
96 execution_mode != axis::ExecutionMode::queued) {
97 accept_management(rt::Result<std::uint32_t>::failure(
98 rt::ErrorCode::unsupported));
99 } else {
100 accept_management(axis_ref->submit_write_parameter(
101 parameter_number, value,
102 execution_mode == axis::ExecutionMode::queued));
103 }
104 }
105 observe_management();
106 }
107};

136// Enable-based snapshot reads. The selector indirection keeps one implementation
137// for the five MC_Read* value blocks.
138class SnapshotValueReadFb : public EnableReadFb
139{
140 public:
141 axis::AxisModel *axis_ref = nullptr;
142 double value = 0.0;
143
144 protected:
145 enum class Field
146 {
147 actual_position,
148 actual_velocity,
149 actual_torque,
150 command_position,
151 command_velocity,
152 };
153
154 void read(Field field)
155 {
156 if (!begin_enable())
157 {
158 if (!enable)
159 value = 0.0;
160 return;
161 }
162 if (axis_ref == nullptr)
163 {
164 value = 0.0;
165 fail_enable(rt::ErrorCode::invalid_argument);
166 return;
167 }
168 const axis::AxisSnapshot &snapshot = axis_ref->snapshot();
169 switch (field)
170 {
171 case Field::actual_position:
172 value = snapshot.actual_position;
173 break;
174 case Field::actual_velocity:
175 value = snapshot.actual_velocity;
176 break;
177 case Field::actual_torque:
178 value = snapshot.actual_torque;
179 break;
180 case Field::command_position:
181 value = snapshot.command_position;
182 break;
183 case Field::command_velocity:
184 value = snapshot.command_velocity;
185 break;
186 }
187 complete_enable();
188 }
189};

221// MC_ReadStatus: one boolean per PLCopen axis state.
222class FbReadStatus : public EnableReadFb
223{
224 public:
225 axis::AxisModel *axis_ref = nullptr;
226 bool disabled = false;
227 bool standstill = false;
228 bool homing = false;
229 bool discrete_motion = false;
230 bool continuous_motion = false;
231 bool synchronized_motion = false;
232 bool stopping = false;
233 bool error_stop = false;
234
235 void call()
236 {
237 clear_states();
238 if (!begin_enable())
239 {
240 return;
241 }
242 if (axis_ref == nullptr)
243 {
244 fail_enable(rt::ErrorCode::invalid_argument);
245 return;
246 }
247 const axis::AxisStatus status = axis_ref->status();
248 disabled = status == axis::AxisStatus::disabled;
249 standstill = status == axis::AxisStatus::standstill;
250 homing = status == axis::AxisStatus::homing;
251 discrete_motion = status == axis::AxisStatus::discrete_motion;
252 continuous_motion = status == axis::AxisStatus::continuous_motion;
253 synchronized_motion = status == axis::AxisStatus::synchronized_motion;
254 stopping = status == axis::AxisStatus::stopping;
255 error_stop = status == axis::AxisStatus::errorstop;
256 complete_enable();
257 }
258
259 private:
260 void clear_states()
261 {
262 disabled = false;
263 standstill = false;
264 homing = false;
265 discrete_motion = false;
266 continuous_motion = false;
267 synchronized_motion = false;
268 stopping = false;
269 error_stop = false;
270 }
271};
