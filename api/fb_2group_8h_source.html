111// synchronization so gear/cam slaves keep the v0.x observable group state.

113{
114 public:
115 axis::AxisGroup *group_ref = nullptr;
116 bool enable = false;

122 bool moving = false;
123 bool stopping = false;
124 bool error_stop = false;
125 bool interrupted = false;
126 bool group_homing = false;

129 {
130 disabled = false;
131 standby = false;
132 moving = false;
133 stopping = false;
134 error_stop = false;
135 interrupted = false;
136 group_homing = false;
137 if (!enable)
138 {
139 valid = false;
140 error = false;
141 error_id = rt::ErrorCode::ok;
142 return;
143 }
144 if (group_ref == nullptr)
145 {
146 valid = false;
147 error = true;
148 error_id = rt::ErrorCode::invalid_argument;
149 return;

157 if (axis != nullptr && axis->status() == axis::AxisStatus::synchronized_motion)

162 member_homing = true;
163 }
164 }

166 group_homing = status == axis::GroupStatus::standby && member_homing;
167 standby = status == axis::GroupStatus::standby && !member_synchronized && !member_homing;
168 moving = status == axis::GroupStatus::moving ||
169 (status == axis::GroupStatus::standby && member_synchronized);
170 stopping = status == axis::GroupStatus::stopping;
171 error_stop = status == axis::GroupStatus::errorstop;
172 interrupted = status == axis::GroupStatus::interrupted;
173 valid = true;
174 error = false;
175 error_id = rt::ErrorCode::ok;

750 {
751 clear(outputs);
752 if(!result) {

754 outputs.error_id = result.error();
755 tracked_command_id_ = 0;

760 outputs.command_accepted = true;
761 outputs.busy = true;

770 outputs.busy = false;
771 outputs.active = false;
772 return;
773 }

775 if(error != rt::ErrorCode::ok) {
776 outputs.error = true;
777 outputs.error_id = error;
778 outputs.busy = false;
779 outputs.active = false;
780 return;
781 }
782 if(group_ref->management_command_done(tracked_command_id_)) {
783 outputs.done = true;
784 outputs.busy = false;
785 outputs.active = false;
786 return;
787 }
788 outputs.busy = true;
789 outputs.active = group_ref->management_command_active(tracked_command_id_);
790 }
