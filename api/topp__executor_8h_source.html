3// Y3 TOPP-RA executor integration (algorithm contract §3).
4//
5// Bridge between TOPP planning-domain solvers (return optimal_time) and
6// the RT-domain executor (needs Profile1D with integer-tick segments).
7//
8// Pipeline:
9// 1. TOPP solver → ToppTimedResult (optimal_time, velocity profile)
10// 2. build_topp_timed_profile() → Profile1D via solve_fixed_time
11// 3. verify_joint_limits() → check per-axis limits along the path
12// 4. sample_profiled_path() → RT sampling (existing infrastructure)
13//
14// The scalar Profile1D distributes time uniformly (curvature-unaware).
15// TOPP's value is the globally optimal duration — joint-limit safety
16// is verified post-hoc at grid points, with fallback to scalar OTG.

32struct ToppVelocityProfile
33{
34 static constexpr int MaxGrid = 1024;
35 double sdot_sq[MaxGrid] = {};
36 int grid_points = 0;
37 double ds = 0.0;
38 double path_length = 0.0;
39 double optimal_time = 0.0;
40
41 double sdot_at(int k) const
42 {
43 if(k < 0 || k >= grid_points) {
44 return 0.0;
45 }
46 return std::sqrt(std::max(sdot_sq[k], 0.0));
47 }
48
49 double interpolate_sdot(double s) const
50 {
51 if(ds <= 0.0 || grid_points < 2) {
52 return 0.0;
53 }
54 const double fk = s / ds;
55 const int k = static_cast<int>(fk);
56 if(k < 0) {
57 return sdot_at(0);
58 }
59 if(k >= grid_points - 1) {
60 return sdot_at(grid_points - 1);
61 }
62 const double t = fk - static_cast<double>(k);
63 return (1.0 - t) * sdot_at(k) + t * sdot_at(k + 1);
64 }
65};

70inline rt::Result<ToppVelocityProfile> solve_topp_profile_l1(
71 const geom::PathSegment &path,
72 const ToppAxisLimits limits[3],
73 int grid_size)
74{
75 if(grid_size < 2) {
76 return rt::Result<ToppVelocityProfile>::failure(
77 rt::ErrorCode::invalid_argument);
78 }
79 const double L = path.length();
80 if(L <= 0.0) {
81 ToppVelocityProfile vp{};
82 vp.grid_points = 1;
83 return rt::Result<ToppVelocityProfile>::success(vp);
84 }
85
86 constexpr int MaxGrid = ToppVelocityProfile::MaxGrid;
87 if(grid_size + 1 > MaxGrid) {
88 return rt::Result<ToppVelocityProfile>::failure(
89 rt::ErrorCode::invalid_argument);
90 }
91
92 const int N = grid_size;
93 const double ds = L / static_cast<double>(N);
94
95 double x_mvc[MaxGrid];
96 geom::Vec3 qs[MaxGrid];
97 geom::Vec3 qss[MaxGrid];
98
99 for(int k = 0; k <= N; ++k) {
100 const double s = ds * static_cast<double>(k);
101 qs[k] = path.path_derivative(s);
102 qss[k] = path.path_second_derivative(s);
103
104 double mvc = topp_detail::velocity_limit_squared(qs[k], limits);
105 const double cs[3] = {qs[k].x, qs[k].y, qs[k].z};
106 const double css[3] = {qss[k].x, qss[k].y, qss[k].z};
107 for(int i = 0; i < 3; ++i) {
108 mvc = std::min(mvc,
109 topp_detail::acceleration_velocity_limit(
110 cs[i], css[i], limits[i].max_acceleration));
111 }
112 mvc = std::min(mvc,
113 topp_detail::cross_axis_velocity_limit(cs, css, limits));
114 x_mvc[k] = mvc;
115 }
116
117 double x_back[MaxGrid];
118 x_back[N] = 0.0;
119 for(int k = N - 1; k >= 0; --k) {
120 const double cs[3] = {qs[k].x, qs[k].y, qs[k].z};
121 const double css[3] = {qss[k].x, qss[k].y, qss[k].z};
122 double x_max = 1e30;
123 for(int i = 0; i < 3; ++i) {
124 x_max = std::min(x_max,
125 topp_detail::axis_backward_reach(
126 cs[i], css[i], limits[i].max_acceleration,
127 x_back[k + 1], ds));
128 }
129 x_max = std::min(x_max, x_mvc[k]);
130 x_back[k] = std::max(x_max, 0.0);
131 }
132
133 double x_fwd[MaxGrid];
134 x_fwd[0] = 0.0;
135 for(int k = 0; k < N; ++k) {
136 const double cs[3] = {qs[k].x, qs[k].y, qs[k].z};
137 const double css[3] = {qss[k].x, qss[k].y, qss[k].z};
138 double x_next = 1e30;
139 for(int i = 0; i < 3; ++i) {
140 x_next = std::min(x_next,
141 topp_detail::axis_forward_reach(
142 cs[i], css[i], limits[i].max_acceleration,
143 x_fwd[k], ds));
144 }
145 x_next = std::min(x_next, x_back[k + 1]);
146 x_next = std::min(x_next, x_mvc[k + 1]);
147 x_fwd[k + 1] = std::max(x_next, 0.0);
148 }
149
150 double total_time = 0.0;
151 for(int k = 0; k < N; ++k) {
152 const double sd_k = std::sqrt(std::max(x_fwd[k], 0.0));
153 const double sd_k1 = std::sqrt(std::max(x_fwd[k + 1], 0.0));
154 const double sum = sd_k + sd_k1;
155 if(sum < 1e-30) {
156 return rt::Result<ToppVelocityProfile>::failure(
157 rt::ErrorCode::infeasible);
158 }
159 total_time += 2.0 * ds / sum;
160 }
161
162 ToppVelocityProfile vp{};
163 for(int k = 0; k <= N; ++k) {
164 vp.sdot_sq[k] = x_fwd[k];
165 }
166 vp.grid_points = N + 1;
167 vp.ds = ds;
168 vp.path_length = L;
169 vp.optimal_time = total_time;
170
171 return rt::Result<ToppVelocityProfile>::success(vp);
172}

176// Compute effective scalar limits from per-axis limits and path geometry.
177// Uses the midpoint tangent (same method as the shadow oracle baseline).
178inline otg::Limits1D effective_scalar_limits(
179 const geom::PathSegment &path,
180 const ToppAxisLimits limits[3])
181{
182 const geom::Vec3 q_s = path.path_derivative(path.length() * 0.5);
183 const double qs[3] = {q_s.x, q_s.y, q_s.z};
184
185 double v_eff = 1e30;
186 double a_eff = 1e30;
187 for(int i = 0; i < 3; ++i) {
188 const double a = std::fabs(qs[i]);
189 if(a > 1e-15) {
190 if(limits[i].max_velocity > 0.0) {
191 v_eff = std::min(v_eff, limits[i].max_velocity / a);
192 }
193 a_eff = std::min(a_eff, limits[i].max_acceleration / a);
194 }
195 }
196 if(v_eff > 1e29) {
197 v_eff = 1.0;
198 }
199 if(a_eff > 1e29) {
200 a_eff = 1.0;
201 }
202
203 return {v_eff, a_eff, a_eff, a_eff * a_eff / std::max(v_eff * 0.01, 1e-15)};
204}

206inline otg::Limits1D effective_scalar_limits(
207 const geom::PathSegment &path,
208 const ToppJerkAxisLimits limits[3])
209{
210 const geom::Vec3 q_s = path.path_derivative(path.length() * 0.5);
211 const double qs[3] = {q_s.x, q_s.y, q_s.z};
212
213 double v_eff = 1e30;
214 double a_eff = 1e30;
215 double j_eff = 1e30;
216 for(int i = 0; i < 3; ++i) {
217 const double a = std::fabs(qs[i]);
218 if(a > 1e-15) {
219 if(limits[i].max_velocity > 0.0) {
220 v_eff = std::min(v_eff, limits[i].max_velocity / a);
221 }
222 a_eff = std::min(a_eff, limits[i].max_acceleration / a);
223 j_eff = std::min(j_eff, limits[i].max_jerk / a);
224 }
225 }
226 if(v_eff > 1e29) {
227 v_eff = 1.0;
228 }
229 if(a_eff > 1e29) {
230 a_eff = 1.0;
231 }
232 if(j_eff > 1e29) {
233 j_eff = a_eff * 10.0;
234 }
235
236 return {v_eff, a_eff, a_eff, j_eff};
237}

239// Verify that the Profile1D trajectory respects per-axis joint limits
240// when mapped through the given path geometry.
241// Returns true if all sampled points are within tolerance.
242inline bool verify_joint_limits(
243 const otg::Profile1D &profile,
244 const geom::PathSegment &path,
245 const ToppAxisLimits limits[3],
246 int num_samples = 64,
247 double tolerance = 1.01)
248{
249 const std::int64_t total = profile.duration_cycles();
250 if(total <= 0) {
251 return true;
252 }
253
254 const int step = std::max(
255 static_cast<int>(total / static_cast<std::int64_t>(num_samples)), 1);
256
257 for(std::int64_t t = 0; t <= total; t += step) {
258 const otg::State1D st =
259 otg::sample(profile, rt::CycleTick::from_cycles(t));
260 const double s = st.position;
261 const double sdot = st.velocity;
262
263 if(s < 0.0 || s > path.length() * 1.001) {
264 continue;
265 }
266
267 const geom::Vec3 q_s = path.path_derivative(
268 std::min(s, path.length()));
269 const double qs[3] = {q_s.x, q_s.y, q_s.z};
270
271 for(int i = 0; i < 3; ++i) {
272 const double vi = std::fabs(qs[i] * sdot);
273 if(limits[i].max_velocity > 0.0 &&
274 vi > limits[i].max_velocity * tolerance) {
275 return false;
276 }
277 }
278 }
279 return true;
280}

290// Full TOPP executor pipeline: solve TOPP → quantize → build Profile1D.
291//
292// For acceleration-limited TOPP (Layer 1). Uses solve_fixed_time with
293// ceil(T_topp) cycles as the prescribed duration. The resulting Profile1D
294// traverses path_length in exactly that many cycles.
295//
296// If the TOPP time is shorter than the scalar OTG minimum (rare — happens
297// when effective scalar limits are more conservative than the per-axis
298// TOPP limits at non-midpoint sections), the scalar OTG minimum wins.
299inline rt::Result<ToppProfileResult> plan_topp_profiled(
300 const geom::PathSegment &path,
301 const ToppAxisLimits limits[3],
302 int grid_size = 128)
303{
304 const auto topp = topp_executor_detail::solve_topp_profile_l1(
305 path, limits, grid_size);
306 if(!topp) {
307 return rt::Result<ToppProfileResult>::failure(topp.error());
308 }
309
310 const ToppVelocityProfile &vp = topp.value();
311 if(vp.path_length <= 0.0) {
312 ToppProfileResult r{};
313 r.curvature_verified = true;
314 return rt::Result<ToppProfileResult>::success(r);
315 }
316
317 const otg::Limits1D scalar = effective_scalar_limits(path, limits);
318
319 const auto scalar_min = otg::plan_time_optimal(
320 {0.0, 0.0, 0.0}, {vp.path_length, 0.0, 0.0}, scalar);
321 const std::int64_t t_scalar_min =
322 scalar_min ? scalar_min.value().duration_cycles() : 1;
323
324 std::int64_t total_cycles =
325 static_cast<std::int64_t>(std::ceil(vp.optimal_time));
326 if(total_cycles < t_scalar_min) {
327 total_cycles = t_scalar_min;
328 }
329 if(total_cycles < 1) {
330 total_cycles = 1;
331 }
332
333 const auto profile = otg::solve_fixed_time(
334 {0.0, 0.0, 0.0},
335 {vp.path_length, 0.0, 0.0},
336 scalar,
337 total_cycles);
338 if(!profile) {
339 return rt::Result<ToppProfileResult>::failure(profile.error());
340 }
341
342 ToppProfileResult r{};
343 r.profile = profile.value();
344 r.topp_time = vp.optimal_time;
345 r.quantized_cycles = total_cycles;
346 r.curvature_verified = verify_joint_limits(
347 r.profile, path, limits);
348
349 return rt::Result<ToppProfileResult>::success(r);
350}

352// Full TOPP executor pipeline with jerk-aware solver (Layer 2).
353inline rt::Result<ToppProfileResult> plan_topp_profiled_jerk(
354 const geom::PathSegment &path,
355 const ToppJerkAxisLimits limits[3],
356 int grid_size = 128)
357{
358 const auto topp = solve_topp_ra_jerk(path, limits, grid_size);
359 if(!topp) {
360 return rt::Result<ToppProfileResult>::failure(topp.error());
361 }
362
363 const double T = topp.value().optimal_time;
364 const double L = path.length();
365 if(L <= 0.0) {
366 ToppProfileResult r{};
367 r.curvature_verified = true;
368 return rt::Result<ToppProfileResult>::success(r);
369 }
370
371 const otg::Limits1D scalar = effective_scalar_limits(path, limits);
372
373 const auto scalar_min = otg::plan_time_optimal(
374 {0.0, 0.0, 0.0}, {L, 0.0, 0.0}, scalar);
375 const std::int64_t t_scalar_min =
376 scalar_min ? scalar_min.value().duration_cycles() : 1;
377
378 std::int64_t total_cycles =
379 static_cast<std::int64_t>(std::ceil(T));
380 if(total_cycles < t_scalar_min) {
381 total_cycles = t_scalar_min;
382 }
383 if(total_cycles < 1) {
384 total_cycles = 1;
385 }
386
387 const auto profile = otg::solve_fixed_time(
388 {0.0, 0.0, 0.0},
389 {L, 0.0, 0.0},
390 scalar,
391 total_cycles);
392 if(!profile) {
393 return rt::Result<ToppProfileResult>::failure(profile.error());
394 }
395
396 ToppAxisLimits verify_limits[3];
397 for(int i = 0; i < 3; ++i) {
398 verify_limits[i] = {limits[i].max_velocity, limits[i].max_acceleration};
399 }
400
401 ToppProfileResult r{};
402 r.profile = profile.value();
403 r.topp_time = T;
404 r.quantized_cycles = total_cycles;
405 r.curvature_verified = verify_joint_limits(
406 r.profile, path, verify_limits);
407
408 return rt::Result<ToppProfileResult>::success(r);
409}
