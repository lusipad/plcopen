11// through it, codegen lowers each semantic kind to opcodes, the test suite

19enum class ConvKind : std::uint8_t
20{
21 unsupported = 0, // cell explicitly absent (function does not exist)
22 identity, // canonical form already matches (pure type change)
23 wrap, // integer/bit-string re-canonicalization (matrix 4.2/4.5)
24 int_to_float, // signed canonical -> double (+ narrow for REAL)
25 uint_to_float, // unsigned/bit canonical -> double (+ narrow for REAL)
26 float_round, // round-half-even then wrap; NaN/Inf faults (matrix 4.3)
27 float_narrow, // LREAL -> REAL (IEEE nearest; overflow -> +-inf)
28 float_widen, // REAL -> LREAL (exact)
29 to_bool_int, // canonical != 0
30 to_bool_float, // double value != 0.0 (minus zero is FALSE)
31};

41// Semantics of the <from>_TO_<to> cell (matrix 4.2-4.7).
42constexpr ConvKind conv_kind(Type from, Type to)
43{
44 if(from == to) {
45 return ConvKind::unsupported; // no self-conversion function
46 }
47 if(from == Type::time || to == Type::time) {
48 // Only TIME_TO_LINT / LINT_TO_TIME exist (matrix 4.6).
49 if(from == Type::time && to == Type::lint) {
50 return ConvKind::identity;
51 }
52 if(from == Type::lint && to == Type::time) {
53 return ConvKind::identity;
54 }
55 return ConvKind::unsupported;
56 }
57 const bool from_int = is_integer(from) || is_bitstring(from);
58 const bool to_int = is_integer(to) || is_bitstring(to);
59 if(to == Type::bool_) {
60 return is_real_family(from) ? ConvKind::to_bool_float
61 : ConvKind::to_bool_int;
62 }
63 if(from == Type::bool_) {
64 if(to_int) {
65 return ConvKind::wrap; // canonical 0/1 always fits
66 }
67 return ConvKind::int_to_float;
68 }
69 if(from_int && to_int) {
70 // Widening within the canonical form is a pure type change.
71 return widens_to(from, to) ? ConvKind::identity : ConvKind::wrap;
72 }
73 if(from_int && is_real_family(to)) {
74 return is_unsigned_int(from) || is_bitstring(from)
75 ? ConvKind::uint_to_float
76 : ConvKind::int_to_float;
77 }
78 if(is_real_family(from) && to_int) {
79 return ConvKind::float_round;
80 }
81 if(from == Type::lreal && to == Type::real) {
82 return ConvKind::float_narrow;
83 }
84 if(from == Type::real && to == Type::lreal) {
85 return ConvKind::float_widen;
86 }
87 return ConvKind::unsupported;
88}

115// Resolves a lowercase identifier to a conversion descriptor. Covers the
116// pairwise matrix, the two TIME cells, and TRUNC_INT/DINT/LINT (matrix
117// 4.8). Returns false when the name is not a conversion function.
118inline bool resolve_conversion(std::string_view lower, ConvDesc &desc)
119{
120 const auto type_by_lower = [](std::string_view text, Type &out) {
121 for(int i = 0; i < kTypeCount; ++i) {
122 const Type type = static_cast<Type>(i);
123 const char *name = to_string(type);
124 std::string lower_name;
125 for(const char *c = name; *c; ++c) {
126 lower_name.push_back(
127 static_cast<char>(*c >= 'A' && *c <= 'Z' ? *c - 'A' + 'a'
128 : *c));
129 }
130 if(text == lower_name) {
131 out = type;
132 return true;
133 }
134 }
135 return false;
136 };
137
138 if(lower == "trunc_int" || lower == "trunc_dint" || lower == "trunc_lint") {
139 desc.from = Type::lreal; // accepts REAL via widening
140 desc.to = lower == "trunc_int"
141 ? Type::int_
142 : (lower == "trunc_dint" ? Type::dint : Type::lint);
143 desc.kind = ConvKind::float_round;
144 desc.trunc = true;
145 return true;
146 }
147 const std::size_t sep = lower.find("_to_");
148 if(sep == std::string_view::npos) {
149 return false;
150 }
151 Type from = Type::bool_;
152 Type to = Type::bool_;
153 if(!type_by_lower(lower.substr(0, sep), from) ||
154 !type_by_lower(lower.substr(sep + 4), to)) {
155 return false;
156 }
157 const ConvKind kind = conv_kind(from, to);
158 if(kind == ConvKind::unsupported) {
159 return false;
160 }
161 desc.from = from;
162 desc.to = to;
163 desc.kind = kind;
164 desc.trunc = false;
165 return true;
166}
