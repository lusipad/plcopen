486 const auto add = [&cost](std::uint64_t value) {
487 if(cost > std::numeric_limits<std::uint64_t>::max() - value)
488 return false;
489 cost += value;
490 return true;
491 };
492 for(const SfcNetworkInfo &network : sfc_networks) {
493 const std::uint64_t steps = network.steps.size();
494 const std::uint64_t blocks = network.action_blocks.size();
495 const std::uint64_t actions = network.actions.size();
496 if(!add(2U * steps + network.transitions.size() +
497 32U * blocks + actions * 3U))
498 return std::numeric_limits<std::uint64_t>::max();
499 for(std::size_t index = 0; index < network.transitions.size();
500 ++index) {
501 const SfcTransitionInfo &transition = network.transitions[index];
502 std::uint64_t check = 1U + transition.sources.size();
503 for(std::size_t prior = 0; prior < index; ++prior)
504 check += 1U + transition.sources.size() *
505 network.transitions[prior].sources.size();
506 if(!add(check + 1U + transition.sources.size() +
507 transition.targets.size()))
508 return std::numeric_limits<std::uint64_t>::max();
509 }
510 if(!add(blocks + actions * (2U * blocks + 3U) + actions +
511 2U * steps + network.transitions.size() + blocks +
512 actions + steps + 16U * blocks + actions))
513 return std::numeric_limits<std::uint64_t>::max();

954 const char *configuration_name, TaskingReport &report) const
955{

958 const ConfigurationInfo *selected = nullptr;

960 if(tasking_detail::same_name(configuration.lower, configuration_name)) {

966 report.resources = selected->resources.size() >
967 std::numeric_limits<std::uint16_t>::max()
968 ? std::numeric_limits<std::uint16_t>::max()
969 : static_cast<std::uint16_t>(
970 selected->resources.size());

972 for(const Program &program : programs)

981 report.tasks, resource.tasks.size());
982 report.tasks = task_count > std::numeric_limits<std::uint32_t>::max()

986 report.program_mappings, resource.mappings.size());

989 ? std::numeric_limits<std::uint32_t>::max()
990 : static_cast<std::uint32_t>(mapping_count);
991 report.required_runtime_bytes = tasking_detail::saturated_add(
992 report.required_runtime_bytes,
993 tasking_detail::align_runtime(sizeof(ResourceRuntimeStorage)));
994 for(const TaskInfo &task : resource.tasks) {
995 if(task.instruction_budget <= 0)

1037 resource_memory_bytes, program->process_image.memory_bytes);

1055 tasking_detail::align_runtime(parked_per_task));
1056 }
