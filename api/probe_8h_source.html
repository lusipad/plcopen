13// MC_TouchProbe: arms one trigger input and reports the position captured on
14// its rising edge. window_only gates the capture to [first, last]. The trigger
15// levels are fed by adapters through AxisModel::set_digital_input.
16class FbTouchProbe
17{
18public:
19 axis::AxisModel *axis_ref = nullptr;
20 std::size_t trigger_input = 0;
21 bool window_only = false;
22 double first_position = 0.0;
23 double last_position = 0.0;
24 bool execute = false;
25 MotionOutputs outputs{};
26 double recorded_position = 0.0;
27
28 void call()
29 {
30 const bool rising = execute && !last_execute_;
31 const bool falling = !execute && last_execute_;
32 last_execute_ = execute;
33 if(rising) {
34 clear(outputs);
35 recorded_position = 0.0;
36 tracked_command_id_ = 0;
37 terminal_low_cycle_ = false;
38 } else if(!execute && terminal_low_cycle_) {
39 clear(outputs);
40 recorded_position = 0.0;
41 tracked_command_id_ = 0;
42 terminal_low_cycle_ = false;
43 return;
44 } else if(falling &&
45 (outputs.done || outputs.command_aborted || outputs.error)) {
46 clear(outputs);
47 recorded_position = 0.0;
48 tracked_command_id_ = 0;
49 }
50 if(rising) {
51 arm();
52 return;
53 }
54 observe();
55 }
56
57private:
58 void arm()
59 {
60 clear(outputs);
61 recorded_position = 0.0;
62 tracked_command_id_ = 0;
63 if(axis_ref == nullptr) {
64 outputs.error = true;
65 outputs.error_id = rt::ErrorCode::invalid_argument;
66 return;
67 }
68 const rt::Result<std::uint32_t> armed =
69 axis_ref->arm_touch_probe(trigger_input, window_only, first_position, last_position);
70 if(!armed) {
71 outputs.error = true;
72 outputs.error_id = armed.error();
73 return;
74 }
75 tracked_command_id_ = armed.value();
76 outputs.command_id = tracked_command_id_;
77 outputs.command_accepted = true;
78 outputs.busy = true;
79 outputs.active = true;
80 }
81
82 void observe()
83 {
84 if(tracked_command_id_ == 0 || axis_ref == nullptr || outputs.done) {
85 return;
86 }
87 if(axis_ref->probe_command_id(trigger_input) != tracked_command_id_) {
88 outputs.command_aborted = true;
89 outputs.busy = false;
90 outputs.active = false;
91 tracked_command_id_ = 0;
92 return;
93 }
94 if(axis_ref->probe_captured(trigger_input)) {
95 recorded_position = axis_ref->probe_recorded_position(trigger_input);
96 outputs.done = true;
97 outputs.busy = false;
98 outputs.active = false;
99 terminal_low_cycle_ = !execute;
100 return;
101 }
102 outputs.busy = true;
103 outputs.active = true;
104 }
105
106 std::uint32_t tracked_command_id_ = 0;
107 bool last_execute_ = false;
108 bool terminal_low_cycle_ = false;
109};

111// MC_AbortTrigger: disarms the probe on the matching input; disarming an idle
112// input is not an error.
113class FbAbortTrigger
114{
115public:
116 axis::AxisModel *axis_ref = nullptr;
117 std::size_t trigger_input = 0;
118 bool execute = false;
119 MotionOutputs outputs{};
120
121 void call()
122 {
123 const bool rising = execute && !last_execute_;
124 last_execute_ = execute;
125 if(!execute) {
126 clear(outputs);
127 return;
128 }
129 if(!rising) {
130 return;
131 }
132 clear(outputs);
133 const rt::ErrorCode aborted = axis_ref == nullptr
134 ? rt::ErrorCode::invalid_argument
135 : axis_ref->abort_trigger(trigger_input);
136 if(aborted != rt::ErrorCode::ok) {
137 outputs.error = true;
138 outputs.error_id = aborted;
139 return;
140 }
141 outputs.done = true;
142 }
143
144private:
145 bool last_execute_ = false;
146};
