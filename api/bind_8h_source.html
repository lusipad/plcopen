100inline void fb_cycle(FbType type, unsigned char *storage,
101 std::int64_t task_period_ns = 0)
102{
103 // Timer periods are installed once in the load domain. The generated
104 // standalone invoke helper accepts an explicit period, so VM scans keep
105 // the timer's installed value through its direct cycle entry.
106 switch(type) {
107 case FbType::ton:
108 reinterpret_cast<fb::TON *>(storage)->cycle();
109 return;
110 case FbType::tof:
111 reinterpret_cast<fb::TOF *>(storage)->cycle();
112 return;
113 case FbType::tp:
114 reinterpret_cast<fb::TP *>(storage)->cycle();
115 return;
116 default: break;
117 }
118 (void)generated::st_binding_invoke(type, storage, task_period_ns);
119}
