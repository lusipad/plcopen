165 rt::ErrorCode queue_force_index(std::size_t index, TypeId type,
166 const unsigned char *value,
167 std::size_t size, std::uint32_t owner,
168 std::uint64_t release,
169 std::uint64_t &queue_version)
170 {
171 if(program_ == nullptr || index >= force_count_ || value == nullptr)
172 return rt::ErrorCode::invalid_argument;
173 const LocatedVarInfo &var =
174 program_->process_image.variables[index];
175 if(var.type_id != type || size != var.byte_width) {
176 return rt::ErrorCode::invalid_argument;
177 }
178 while(force_queue_lock_.test_and_set(std::memory_order_acquire)) {}
179 ForceState &state = forces_[index];

181 std::size_t count = 0;

185 ++count;
186 }

189 force_queue_lock_.clear(std::memory_order_release);
190 return rt::ErrorCode::capacity_exceeded;
191 }
192 }
193 queue_version =
194 force_queue_version_.fetch_add(1, std::memory_order_acq_rel) + 1U;
195 publish_force(state, true, value, size, owner, release);
196 state.requested_active.store(true, std::memory_order_release);
197 force_queue_lock_.clear(std::memory_order_release);
198 return rt::ErrorCode::ok;

254 SnapshotError encode_snapshot(SnapshotKind kind, unsigned char *bytes,
255 std::size_t capacity,
256 std::size_t &written) const
257 {
258 written = 0;
259 if(program_ == nullptr || bytes == nullptr) {
260 return SnapshotError::invalid_argument;
261 }
262 std::size_t snapshot_bytes = 0;
263 std::uint64_t snapshot_version = 0;
264 if(snapshot_channel(memory_channel_,
265 program_->process_image.memory_bytes,
266 snapshot_scratch_.data(), snapshot_scratch_.size(),
267 snapshot_bytes, snapshot_version) !=
268 rt::ErrorCode::ok) {
269 return SnapshotError::invalid_argument;
270 }
271 (void)snapshot_bytes;
272 (void)snapshot_version;
273 const std::size_t count = snapshot_target_count(kind);
274 std::size_t required = 4U + 4U + 1U + 8U + 2U;
275 for(const LocatedVarInfo &var : program_->process_image.variables) {
276 if(snapshot_member(var, kind)) required += 8U + 4U + 1U + var.byte_width;
277 }
278 if(capacity < required) return SnapshotError::capacity_exceeded;
279 unsigned char *out = bytes;
280 *out++ = 'S'; *out++ = 'T'; *out++ = 'L'; *out++ = '3';
281 put(out, 1U, 4);
282 *out++ = static_cast<unsigned char>(kind);
283 put(out, program_->process_image.fingerprint, 8);
284 put(out, count, 2);
285 for(const LocatedVarInfo &var : program_->process_image.variables) {
286 if(!snapshot_member(var, kind)) continue;
287 put(out, var.stable_id, 8);
288 put(out, var.type_id, 4);
289 *out++ = var.byte_width;
290 read_image_value(snapshot_scratch_.data(), var, out);
291 out += var.byte_width;
292 }
293 written = static_cast<std::size_t>(out - bytes);
294 return SnapshotError::ok;
295 }

297 SnapshotError restore_snapshot(SnapshotKind kind,
298 const unsigned char *bytes,
299 std::size_t size, SnapshotReport &report)
300 {
301 report = SnapshotReport{};
302 if(program_ == nullptr || bytes == nullptr) {
303 return SnapshotError::invalid_argument;
304 }
305 report.defaulted = static_cast<std::uint16_t>(snapshot_target_count(kind));
306 if(size < 19U || std::memcmp(bytes, "STL3", 4) != 0) {
307 return SnapshotError::malformed;
308 }
309 const unsigned char *cursor = bytes + 4;
310 const unsigned char *end = bytes + size;
311 std::uint64_t version = 0;
312 if(!get(cursor, end, version, 4)) return SnapshotError::malformed;
313 if(version != 1U) return SnapshotError::version_mismatch;
314 if(cursor == end || *cursor != static_cast<unsigned char>(kind)) {
315 return SnapshotError::malformed;
316 }
317 ++cursor;
318 std::uint64_t fingerprint = 0;
319 std::uint64_t count = 0;
320 if(!get(cursor, end, fingerprint, 8) ||
321 !get(cursor, end, count, 2)) return SnapshotError::malformed;
322 report.fingerprint_match =
323 fingerprint == program_->process_image.fingerprint;
324 if(kind == SnapshotKind::retain && !report.fingerprint_match) {
325 return SnapshotError::fingerprint_mismatch;
326 }
327 const SnapshotReport rejected_report = report;
328 const int batch_index = claim_restore_slot();
329 RestoreSlot &batch = restore_channel_.slots[
330 static_cast<std::size_t>(batch_index)];
331 for(RestoreState &state : batch.entries) {
332 state.pending = false;
333 state.seen = false;
334 }
335 const auto reject_batch = [&](SnapshotError error) {
336 batch.state.store(slot_empty, std::memory_order_release);
337 report = rejected_report;
338 return error;
339 };
340 for(std::uint64_t item = 0; item < count; ++item) {
341 std::uint64_t id = 0;
342 std::uint64_t type = 0;
343 if(!get(cursor, end, id, 8) || !get(cursor, end, type, 4) ||
344 cursor == end) return reject_batch(SnapshotError::malformed);
345 const std::uint8_t length = *cursor++;
346 if(static_cast<std::size_t>(end - cursor) < length) {
347 return reject_batch(SnapshotError::malformed);
348 }
349 int found = -1;
350 for(std::size_t index = 0;
351 index < program_->process_image.variables.size(); ++index) {
352 const LocatedVarInfo &var = program_->process_image.variables[index];
353 if(snapshot_member(var, kind) && var.stable_id == id) {
354 found = static_cast<int>(index);
355 break;
356 }
357 }
358 if(found >= 0) {
359 RestoreState &state = batch.entries[
360 static_cast<std::size_t>(found)];
361 if(state.seen) return reject_batch(SnapshotError::malformed);
362 state.seen = true;
363 const LocatedVarInfo &var =
364 program_->process_image.variables[static_cast<std::size_t>(found)];
365 if(var.type_id == type && var.byte_width == length) {
366 state.pending = true;
367 std::memcpy(state.value, cursor, length);
368 ++report.restored;
369 --report.defaulted;
370 } else {
371 ++report.rejected;
372 }
373 }
374 cursor += length;
375 }
376 if(cursor != end) return reject_batch(SnapshotError::malformed);
377 publish_restore_batch(batch_index);
378 return SnapshotError::ok;
379 }

463 bool restore_transaction(const unsigned char *output,
464 const unsigned char *output_dirty,
465 std::size_t output_size,
466 const unsigned char *memory,
467 const unsigned char *memory_dirty,
468 std::size_t memory_size) noexcept
469 {
470 if(!transaction_active_ || output_size != output_shadow_.size() ||
471 memory_size != memory_shadow_.size() ||
472 (output_size != 0 && (output == nullptr || output_dirty == nullptr)) ||
473 (memory_size != 0 && (memory == nullptr || memory_dirty == nullptr)))
474 return false;
475 const auto restore = [](unsigned char *destination,
476 std::vector<unsigned char> &dirty,
477 const unsigned char *parked,
478 const unsigned char *parked_dirty,
479 std::size_t size) {
480 for(std::size_t index = 0; index < size; ++index) {
481 const unsigned char mask = parked_dirty[index];
482 destination[index] = static_cast<unsigned char>(
483 (destination[index] & static_cast<unsigned char>(~mask)) |
484 (parked[index] & mask));
485 dirty[index] = mask;
486 }
487 };
488 restore(output_, transaction_output_dirty_, output, output_dirty,
489 output_size);
490 restore(memory_, transaction_memory_dirty_, memory, memory_dirty,
491 memory_size);
492 return true;
493 }

665 int find_located(const char *name) const
666 {
667 if(program_ == nullptr || name == nullptr) return -1;
668 for(std::size_t index = 0;
669 index < program_->process_image.variables.size(); ++index) {
670 const std::string &lower = program_->process_image.variables[index].lower;
671 std::size_t pos = 0;
672 while(name[pos] != '\0' && pos < lower.size()) {
673 char c = name[pos];
674 if(c >= 'A' && c <= 'Z') c = static_cast<char>(c - 'A' + 'a');
675 if(c != lower[pos]) break;
676 ++pos;
677 }
678 if(name[pos] == '\0' && pos == lower.size())
679 return static_cast<int>(index);
680 }
681 return -1;

741 static std::uint8_t physical_mask(const LocatedVarInfo &var,
742 std::size_t byte)
743 {
744 if(var.bit_address)
745 return byte == var.byte_offset
746 ? static_cast<std::uint8_t>(1U << var.bit)
747 : 0;
748 return byte >= var.byte_offset &&
749 byte < static_cast<std::size_t>(var.byte_offset) +
750 var.byte_width
751 ? 0xFFU
752 : 0;

755 void mark_physical(const LocatedVarInfo &var, bool task_write)
756 {
757 if(var.area == ProcessArea::input) return;
758 std::vector<unsigned char> &current =
759 var.area == ProcessArea::output ? current_output_dirty_
760 : current_memory_dirty_;
761 std::vector<unsigned char> &transaction =
762 var.area == ProcessArea::output ? transaction_output_dirty_
763 : transaction_memory_dirty_;
764 for(std::size_t byte = var.byte_offset; byte < current.size(); ++byte) {
765 const std::uint8_t mask = physical_mask(var, byte);
766 if(mask == 0) {
767 if(!var.bit_address &&
768 byte >= static_cast<std::size_t>(var.byte_offset) +
769 var.byte_width)
770 break;
771 continue;
772 }
773 current[byte] = static_cast<std::uint8_t>(current[byte] | mask);
774 if(task_write)
775 transaction[byte] =
776 static_cast<std::uint8_t>(transaction[byte] | mask);
777 }
778 }

780 bool current_dirty(const LocatedVarInfo &var) const
781 {
782 const std::vector<unsigned char> &current =
783 var.area == ProcessArea::output ? current_output_dirty_
784 : current_memory_dirty_;
785 for(std::size_t byte = var.byte_offset; byte < current.size(); ++byte) {
786 const std::uint8_t mask = physical_mask(var, byte);
787 if(mask != 0 && (current[byte] & mask) != 0) return true;
788 if(!var.bit_address &&
789 byte >= static_cast<std::size_t>(var.byte_offset) +
790 var.byte_width)
791 break;
792 }
793 return false;
794 }

877 std::size_t capacity, std::size_t &written, std::uint64_t &version)
878 {
879 written = 0;

882 for(;;) {
883 const int index = channel.published.load(std::memory_order_acquire);
884 if(index < 0) return rt::ErrorCode::invalid_argument;

893 version = slot.version;

918 static void publish_force(ForceState &state, bool active,
919 const unsigned char *value, std::size_t size,
920 std::uint32_t owner, std::uint64_t release)
921 {
922 std::uint64_t packed = 0;
923 for(std::size_t index = 0; index < size; ++index) {
924 packed |= static_cast<std::uint64_t>(value[index]) << (index * 8U);
925 }
926 state.pending_sequence.fetch_add(1, std::memory_order_acq_rel);
927 state.pending_value.store(packed, std::memory_order_relaxed);
928 state.pending_owner.store(owner, std::memory_order_relaxed);
929 state.pending_release.store(release, std::memory_order_relaxed);
930 state.pending_active.store(active, std::memory_order_relaxed);
931 state.pending_sequence.fetch_add(1, std::memory_order_release);
932 }
933
934 static void consume_force(ForceState &state, std::uint32_t boundary_owner,
935 std::uint64_t boundary_release)
936 {
937 std::uint64_t sequence = 0;
938 std::uint64_t value = 0;
939 std::uint64_t release = 0;
940 std::uint32_t owner = wildcard_force_owner;
941 bool active = false;
942 for(;;) {
943 sequence = state.pending_sequence.load(std::memory_order_acquire);
944 if((sequence & 1U) != 0) continue;
945 value = state.pending_value.load(std::memory_order_relaxed);
946 owner = state.pending_owner.load(std::memory_order_relaxed);
947 release = state.pending_release.load(std::memory_order_relaxed);
948 active = state.pending_active.load(std::memory_order_relaxed);
949 if(sequence ==
950 state.pending_sequence.load(std::memory_order_acquire)) {
951 break;
952 }
953 }
954 if(sequence == state.applied_sequence) return;
955 if((owner != wildcard_force_owner && owner != boundary_owner) ||
956 (release != wildcard_force_release &&
957 release != boundary_release))
958 return;
959 state.active = active;
960 if(active) {
961 for(unsigned index = 0; index < sizeof(state.value); ++index) {
962 state.value[index] = static_cast<unsigned char>(
963 value >> (index * 8U));
964 }
965 }
966 state.applied_sequence = sequence;
967 }
