52constexpr WcetOpcodeCost wcet_opcode_cost(Op op) noexcept
53{
54 switch(op) {
55 case Op::string_copy:
56 case Op::string_copy_const:
57 case Op::string_compare:
58 case Op::string_index:
59 case Op::string_length:
60 return {WcetBudgetRule::string_capacity,
61 WcetTimingClass::linear_bytes};
62 case Op::fb_store_object:
63 case Op::fb_load_object:
64 return {WcetBudgetRule::object_size,
65 WcetTimingClass::linear_bytes};
66 case Op::standard_scalar:
67 case Op::standard_string:
68 return {WcetBudgetRule::standard_function,
69 WcetTimingClass::standard_function};
70 case Op::commit_outputs:
71 return {WcetBudgetRule::output_count,
72 WcetTimingClass::linear_bytes};
73 case Op::copy_bytes:
74 return {WcetBudgetRule::fixed_one,
75 WcetTimingClass::linear_bytes};
76 case Op::fb_call:
77 return {WcetBudgetRule::fixed_one, WcetTimingClass::native_fb};
78 default:
79 return static_cast<std::size_t>(op) < kWcetReviewedOpcodeCount
80 ? WcetOpcodeCost{WcetBudgetRule::fixed_one,
81 WcetTimingClass::fixed}
82 : WcetOpcodeCost{};
83 }
84}

130inline void increment(std::uint32_t &value) noexcept
131{
132 if(value != std::numeric_limits<std::uint32_t>::max()) ++value;

135inline void add_size(std::uint32_t &value, std::size_t added) noexcept
136{
137 const std::size_t room =
138 std::numeric_limits<std::uint32_t>::max() - value;
139 value += static_cast<std::uint32_t>(added > room ? room : added);
140}

142inline void add_opcode(WcetReport &report, Op op) noexcept
143{
144 const WcetOpcodeCost cost = wcet_opcode_cost(op);
145 const std::size_t timing =
146 static_cast<std::size_t>(cost.timing_class);
147 if(cost.budget_rule == WcetBudgetRule::unreviewed ||
148 timing >= report.artifact_opcode_counts.size()) {
149 increment(report.unreviewed_opcodes);
150 return;
151 }
152 increment(report.artifact_opcode_counts[timing]);
153}
154
155inline void add_region(WcetReport &report, const SfcRegionInfo &region) noexcept
156{
157 for(const std::uint32_t offset : region.instruction_offsets) {
158 if(offset >= region.code.size()) {
159 increment(report.unreviewed_opcodes);
160 } else {
161 add_opcode(report, static_cast<Op>(region.code[offset]));
162 }
163 }
164}
165
166inline void add_program_body(WcetReport &report,
167 const Program &program) noexcept
168{
169 report.opcode_table_reviewed = report.opcode_table_reviewed &&
170 program.format_version == kWcetReviewedBytecodeFormatVersion;
171 increment(report.program_count);
172 add_size(report.native_fb_instances, program.fbs.size());
173 for(const std::uint32_t offset : program.instruction_offsets) {
174 if(offset >= program.code.size()) {
175 increment(report.unreviewed_opcodes);
176 } else {
177 add_opcode(report, static_cast<Op>(program.code[offset]));
178 }
179 }
180 for(const SfcNetworkInfo &network : program.sfc_networks) {
181 for(const SfcTransitionInfo &transition : network.transitions)
182 add_region(report, transition.condition);
183 for(const SfcActionInfo &action : network.actions)
184 add_region(report, action.region);
185 }
186}
187
188inline void add_artifact_programs(WcetReport &report,
189 const Program &program) noexcept
190{
191 if(program.programs.empty()) {
192 add_program_body(report, program);
193 return;
194 }
195 for(const Program &child : program.programs)
196 add_artifact_programs(report, child);
197}

201// Load/diagnostic-domain report. It walks existing vectors but performs no
202// allocation and never changes the executable artifact.
203inline WcetReport make_wcet_report(const Program &program) noexcept
204{
205 WcetReport report;
206 report.bytecode_format_version = program.format_version;
207 report.opcode_table_reviewed =
208 program.format_version == kWcetReviewedBytecodeFormatVersion;
209 report.bounded = program.worst_case_bounded;
210 report.worst_case_work_units = report.bounded
211 ? program.worst_case_instructions : 0U;
212 report.max_call_depth = program.max_call_depth;
213 report.max_instance_depth = program.max_instance_depth;
214 wcet_detail::add_artifact_programs(report, program);
215
216 const auto count = [&report](WcetTimingClass timing) noexcept {
217 return report.artifact_opcode_counts[
218 static_cast<std::size_t>(timing)];
219 };
220 report.requires_native_fb_profile =
221 report.native_fb_instances != 0 ||
222 count(WcetTimingClass::native_fb) != 0;
223 report.requires_platform_calibration =
224 count(WcetTimingClass::linear_bytes) != 0 ||
225 count(WcetTimingClass::standard_function) != 0 ||
226 report.requires_native_fb_profile;
227 return report;
228}
