249 bool parse_integer_value(Type base, IntegerValue &out)
250 {
251 const bool negative = eat(TokenKind::minus);
252 if(!at(TokenKind::int_literal)) {
253 diag(DiagCode::parse_expected_expression, current_,
254 "integer constant");
255 return false;
256 }
257 const std::uint64_t magnitude = current_.unsigned_value;
258 bump();
259 if(is_unsigned_int(base)) {
260 if(negative) {
261 diag(DiagCode::sema_range_violation, current_,
262 "unsigned bound cannot be negative");
263 return false;
264 }
265 out = IntegerValue::unsigned_value(magnitude);
266 return true;
267 }
268 if(negative) {
269 if(magnitude > 0x8000000000000000ULL) {
270 diag(DiagCode::sema_range_violation, current_);
271 return false;
272 }
273 out = IntegerValue::signed_value(
274 magnitude == 0x8000000000000000ULL
275 ? std::numeric_limits<std::int64_t>::min()
276 : -static_cast<std::int64_t>(magnitude));
277 } else {
278 if(magnitude > 0x7FFFFFFFFFFFFFFFULL) {
279 diag(DiagCode::sema_range_violation, current_);
280 return false;
281 }
282 out = IntegerValue::signed_value(
283 static_cast<std::int64_t>(magnitude));
284 }
285 return true;

288 void parse_user_type()
289 {
290 bump(); // TYPE
291 if(!at(TokenKind::identifier)) {
292 diag(DiagCode::parse_expected_token, current_, "type name");
293 recover_statement();
294 return;
295 }
296 UserTypeDecl decl;
297 decl.line = current_.line;
298 decl.column = current_.column;
299 decl.name.assign(current_.text.data(), current_.text.size());
300 decl.lower = lower_copy(current_.text);
301 bump();
302 if(!expect(TokenKind::colon, "':'")) {
303 recover_statement();
304 return;
305 }
306 if(eat(TokenKind::kw_array)) {
307 decl.kind = UserTypeKind::array;
308 if(!expect(TokenKind::lbracket, "'['")) {
309 recover_statement();
310 return;
311 }
312 if(eat(TokenKind::star)) {
313 diag(DiagCode::unsupported_l1b2_dynamic_array, current_);
314 recover_statement();
315 return;
316 }
317 do {
318 IntegerValue lower;
319 IntegerValue upper;
320 if(!parse_integer_value(Type::lint, lower) ||
321 !expect(TokenKind::dotdot, "'..'") ||
322 !parse_integer_value(Type::lint, upper)) {
323 recover_statement();
324 return;

326 decl.array_bounds.push_back(
327 {lower.as_signed(), upper.as_signed()});
328 } while(eat(TokenKind::comma));
329 if(!expect(TokenKind::rbracket, "']'") ||
330 !expect(TokenKind::kw_of, "OF") ||
331 !parse_type_ref(decl.element_type,
332 decl.element_type_name)) {
333 recover_statement();
334 return;
335 }
336 } else if(eat(TokenKind::kw_struct)) {
337 decl.kind = UserTypeKind::struct_;
338 while(!at(TokenKind::kw_end_struct) &&
339 !at(TokenKind::end_of_input)) {
340 if(!at(TokenKind::identifier)) {
341 diag(DiagCode::parse_expected_token, current_,

344 return;

347 field.line = current_.line;
348 field.column = current_.column;
349 field.name.assign(current_.text.data(), current_.text.size());
350 field.lower = lower_copy(current_.text);
351 bump();

353 !parse_type_ref(field.type, field.type_name) ||
354 !expect(TokenKind::semicolon, "';'")) {
355 recover_statement();
356 return;
357 }
358 decl.struct_fields.push_back(
359 static_cast<StructFieldDecl &&>(field));
360 }
361 expect(TokenKind::kw_end_struct, "END_STRUCT");
362 } else if(eat(TokenKind::lparen)) {
363 decl.kind = UserTypeKind::enum_;
364 while(!at(TokenKind::rparen) &&
365 !at(TokenKind::end_of_input)) {
366 if(!at(TokenKind::identifier)) {
367 diag(DiagCode::parse_expected_token, current_,
368 "enum member");
369 recover_statement();
370 return;
371 }
372 EnumMemberDecl member;
373 member.line = current_.line;
374 member.column = current_.column;
375 member.name.assign(current_.text.data(), current_.text.size());
376 member.lower = lower_copy(current_.text);
377 bump();
378 if(eat(TokenKind::assign)) {
379 IntegerValue value;
380 if(!parse_integer_value(Type::dint, value)) {
381 recover_statement();

384 member.explicit_value = true;
385 member.value = value.as_signed();
386 }
387 decl.enum_members.push_back(
388 static_cast<EnumMemberDecl &&>(member));

391 }
392 }

394 } else {
395 Type base = Type::dint;
396 if(!integer_type_token(current_.kind, base)) {
397 diag(DiagCode::parse_expected_type, current_,

445 void parse_var_block()
446 {
447 bump(); // VAR
448 // VAR CONSTANT block (approved st-l1a-semantics 2.5): every member
449 // is a compile-time constant.
450 const bool constant_block = eat(TokenKind::kw_constant);
451 const bool retain_block = !constant_block && eat(TokenKind::kw_retain);
452 const bool persistent_block =
453 !constant_block && !retain_block && eat(TokenKind::kw_persistent);
454 while(!at(TokenKind::kw_end_var) && !at(TokenKind::end_of_input)) {
455 if(at(TokenKind::unsupported_keyword)) {
456 diag(token_diag(current_), current_);
457 bump();
458 recover_statement();
459 continue;
460 }
461 if(!at(TokenKind::identifier)) {
462 diag(DiagCode::parse_expected_token, current_,
463 "variable declaration");
464 // Consume the offending token before recovery: statement
465 // starters would otherwise satisfy the recovery sync set
466 // without progress and loop forever (fuzz-found).
467 bump();
468 recover_statement();
469 continue;
470 }
471 VarDecl decl;
472 decl.is_constant = constant_block;
473 decl.is_retain = retain_block;
474 decl.is_persistent = persistent_block;
475 decl.line = current_.line;
476 decl.column = current_.column;
477 decl.name.assign(current_.text.data(), current_.text.size());
478 decl.lower = lower_copy(current_.text);
479 bump();
480 if(eat(TokenKind::kw_at)) {
481 if(!at(TokenKind::located_address) ||
482 !parse_location(decl, current_.text)) {
483 diag(DiagCode::parse_expected_token, current_,
484 "located address");
485 recover_statement();
486 continue;
487 }
488 decl.is_located = true;
489 bump();
490 }
491 if(!expect(TokenKind::colon, "':'")) {
492 recover_statement();
493 continue;
494 }
495 if(!parse_type(decl)) {
496 recover_statement();
497 continue;
498 }
499 if(eat(TokenKind::assign)) {
500 decl.init = (at(TokenKind::lbracket) ||
501 (!decl.type_name.empty() &&
502 at(TokenKind::lparen)))
503 ? parse_aggregate_initializer()
504 : parse_expression();
505 if(decl.init == kNoExpr) {
506 recover_statement();
507 continue;
508 }
509 }
510 expect(TokenKind::semicolon, "';'");
511 result_.ast.vars.push_back(static_cast<VarDecl &&>(decl));
512 }
513 expect(TokenKind::kw_end_var, "END_VAR");
514 }

516 static bool parse_location(VarDecl &decl, std::string_view text)
517 {
518 if(text.size() < 4 || text[0] != '%') return false;
519 const auto lower = [](char c) {
520 return c >= 'A' && c <= 'Z' ? static_cast<char>(c - 'A' + 'a')
521 : c;
522 };
523 decl.location_area = static_cast<char>(lower(text[1]) - 'a' + 'A');
524 decl.location_width = static_cast<char>(lower(text[2]) - 'a' + 'A');
525 std::uint64_t byte = 0;
526 std::size_t index = 3;
527 while(index < text.size() && text[index] >= '0' && text[index] <= '9') {
528 const unsigned digit = static_cast<unsigned>(text[index] - '0');
529 if(byte > (std::numeric_limits<std::uint32_t>::max() - digit) /
530 10U) {
531 return false;
532 }
533 byte = byte * 10U + digit;
534 ++index;
535 }
536 decl.location_byte = static_cast<std::uint32_t>(byte);
537 if(decl.location_width == 'X') {
538 if(index >= text.size() || text[index] != '.') return false;
539 ++index;
540 unsigned bit = 0;
541 if(index >= text.size()) return false;
542 while(index < text.size() && text[index] >= '0' &&
543 text[index] <= '9') {
544 bit = bit * 10U + static_cast<unsigned>(text[index] - '0');
545 if(bit > 255U) return false;
546 ++index;
547 }
548 decl.location_bit = static_cast<std::uint8_t>(bit);
549 }
550 return index == text.size();
551 }

553 ExprIndex parse_aggregate_initializer()
554 {
555 const Token open = current_;
556 const TokenKind close = at(TokenKind::lbracket)
557 ? TokenKind::rbracket
558 : TokenKind::rparen;
559 bump();
560 Expr expr;
561 expr.kind = ExprKind::aggregate_init;
562 expr.line = open.line;
563 expr.column = open.column;
564 if(!at(close)) {
565 do {
566 InitItem item;
567 if(close == TokenKind::rparen &&
568 at(TokenKind::identifier) &&
569 peek_next().kind == TokenKind::assign) {
570 item.name.assign(current_.text.data(),
571 current_.text.size());
572 bump();
573 bump();
574 }
575 item.value = (at(TokenKind::lbracket) ||
576 at(TokenKind::lparen))
577 ? parse_aggregate_initializer()
578 : parse_expression();
579 if(item.value == kNoExpr) {
580 return kNoExpr;
581 }
582 expr.items.push_back(static_cast<InitItem &&>(item));
583 } while(eat(TokenKind::comma));
584 }
585 if(!expect(close, close == TokenKind::rbracket ? "']'" : "')'")) {
586 return kNoExpr;
587 }
588 return result_.ast.add_expr(static_cast<Expr &&>(expr));
589 }

729 // Statement list bounded by an explicit stop set. A block-end token that
730 // is NOT in the stop set is a stray closer: it is reported and skipped so
731 // the list (and every later statement) keeps parsing -- recovery must
732 // never silently terminate an outer list (metric 5.8).
733 void parse_statement_list(std::vector<StmtIndex> &out,
734 std::initializer_list<TokenKind> stops)
735 {
736 const auto stopped = [&]() {
737 for(const TokenKind stop : stops) {
738 if(at(stop)) {
739 return true;
740 }
741 }
742 return at(TokenKind::end_of_input);
743 };
744 while(!stopped()) {
745 if(is_block_end(current_.kind)) {
746 diag(DiagCode::parse_unexpected_token, current_,
747 "stray block end");
748 bump();
749 continue;
750 }
751 const StmtIndex index = parse_statement();
752 if(index != -2) { // -2 => recovered, nothing to append
753 out.push_back(index);
754 }
755 }
756 }

758 // Returns stmt index, or -2 when the statement failed and recovery ran.
759 static void copy_debug_provenance(Stmt &stmt, const Token &token)
760 {
