138 // Load domain. The program object must outlive the instance; the buffer
139 // is caller-owned static storage, 8-byte aligned, at least
140 // program.required_bytes() long.
141 rt::ErrorCode load(const Program &program, unsigned char *buffer,
142 std::size_t buffer_bytes, std::int64_t task_period_ns)
143 {
144 unload();
145 if(program.format_version != kBytecodeFormatVersion) {
146 return rt::ErrorCode::bytecode_version_mismatch;
147 }
148 if(buffer == nullptr || task_period_ns <= 0) {
149 return rt::ErrorCode::invalid_argument;
150 }
151 if((reinterpret_cast<std::uintptr_t>(buffer) & 7U) != 0) {
152 return rt::ErrorCode::invalid_argument;
153 }
154 if(buffer_bytes < program.required_bytes()) {
155 return rt::ErrorCode::capacity_exceeded;
156 }
157 vars_ = buffer;
158 visible_vars_ = vars_;
159 execution_vars_ = vars_;
160 fb_area_ = buffer + program.vars_bytes;
161 stack_ = reinterpret_cast<std::uint64_t *>(fb_area_ + program.fb_bytes);
162 std::memset(buffer, 0, program.required_bytes());
163 unsigned char *binding_storage =
164 buffer + program.binding_storage_offset();
165 load_binding_storage(program, binding_storage);
166 sfc_runner_ = ::new(static_cast<void *>(
167 buffer + program.sfc_runner_storage_offset())) SfcRunnerStorage{};
168 sfc_runtime_ = buffer + program.sfc_storage_offset();
169 if(program.needs_debug_runtime_storage()) {
170 ::new(static_cast<void *>(
171 buffer + program.debug_runtime_storage_offset()))
172 DebugRuntimeStorage{};
173 }
174 if(program.initial_data.size() > program.vars_bytes) {
175 return rt::ErrorCode::invalid_argument;
176 }
177 if(!program.initial_data.empty()) {
178 std::memcpy(vars_, program.initial_data.data(),
179 program.initial_data.size());
180 }
181 for(const VarInfo &var : program.vars) {
182 const TypeDesc *desc = program.types.get(var.type_id);
183 if(desc == nullptr ||
184 (desc->kind != TypeKind::string &&
185 desc->kind != TypeKind::wstring)) {
186 continue;
187 }
188 if(var.offset > program.vars_bytes ||
189 desc->size > program.vars_bytes - var.offset ||
190 !valid_string_object(vars_ + var.offset, *desc)) {
191 return rt::ErrorCode::invalid_argument;
192 }
193 }
194 for(const FbInfo &fb : program.fbs) {
195 fb_init(fb.type, fb_area_ + fb.offset, task_period_ns);
196 }
197 task_period_ns_ = task_period_ns;
198 program_ = &program;
199 initialize_sfc_runtime();
200 fault_ = ScanError::ok;
201 execution_state_ = InstanceState::idle;
202 pc_ = 0;
203 instruction_pc_ = 0;
204 sp_ = 0;
205 remaining_budget_ = 0;
206 scan_started_ = false;
207 return rt::ErrorCode::ok;
208 }

223 rt::ErrorCode load(const Program &program, const char *program_name,
224 unsigned char *buffer, std::size_t buffer_bytes,
225 std::int64_t task_period_ns)
226 {
227 if(program_name == nullptr) {
228 return rt::ErrorCode::invalid_argument;
229 }
230 const Program *selected = &program;
231 if(!program.programs.empty()) {
232 selected = nullptr;
233 for(const Program &candidate : program.programs) {
234 std::size_t i = 0;
235 while(i < candidate.program_name.size() &&
236 program_name[i] != '\0') {
237 char c = program_name[i];
238 if(c >= 'A' && c <= 'Z') {
239 c = static_cast<char>(c - 'A' + 'a');
240 }
241 if(c != candidate.program_name[i]) {
242 break;
243 }
244 ++i;
245 }
246 if(i == candidate.program_name.size() &&
247 program_name[i] == '\0') {
248 selected = &candidate;
249 break;
250 }
251 }
252 if(selected == nullptr) {
253 return rt::ErrorCode::invalid_argument;
254 }
255 } else if(!program.program_name.empty()) {
256 std::size_t i = 0;
257 while(i < program.program_name.size() && program_name[i] != '\0') {
258 char c = program_name[i];
259 if(c >= 'A' && c <= 'Z') {
260 c = static_cast<char>(c - 'A' + 'a');
261 }
262 if(c != program.program_name[i]) {
263 return rt::ErrorCode::invalid_argument;
264 }
265 ++i;
266 }
267 if(i != program.program_name.size() || program_name[i] != '\0') {
268 return rt::ErrorCode::invalid_argument;
269 }
270 }
271 return load(*selected, buffer, buffer_bytes, task_period_ns);
272 }

284 BindingError bind_reference(const char *name, BindingTarget target)
285 {
286 if(program_ == nullptr || name == nullptr) {
287 return BindingError::unknown;
288 }
289 if(scan_started_) {
290 return BindingError::locked;
291 }
292 std::uint16_t axis_slot = 0;
293 std::uint16_t group_slot = 0;
294 for(const VarInfo &var : program_->vars) {
295 const BindingTargetKind kind = reference_kind(var.type,
296 var.type_id);
297 if(kind == BindingTargetKind::invalid) {
298 continue;
299 }
300 const std::uint16_t slot = kind == BindingTargetKind::axis
301 ? axis_slot++
302 : group_slot++;
303 if(!ascii_name_equal(name, var.lower)) {
304 continue;
305 }
306 if(target.pointer() == nullptr) {
307 return BindingError::null_target;
308 }
309 if(target.kind() != kind) {
310 return BindingError::wrong_kind;
311 }
312 if((kind == BindingTargetKind::axis &&
313 slot >= axis_targets_->size()) ||
314 (kind == BindingTargetKind::group &&
315 slot >= group_targets_->size())) {
316 return BindingError::capacity_exceeded;
317 }
318 std::uint64_t handle = 0;
319 std::memcpy(&handle, vars_ + var.offset, sizeof(handle));
320 if(handle != 0) {
321 return BindingError::duplicate;
322 }
323 if(kind == BindingTargetKind::axis) {
324 (*axis_targets_)[slot] =
325 static_cast<axis::AxisModel *>(target.pointer());
326 } else {
327 auto *group =
328 static_cast<axis::AxisGroup *>(target.pointer());
329 const rt::ErrorCode period_result =
330 group->set_task_cycle_period_ns(task_period_ns_);
331 if(period_result != rt::ErrorCode::ok &&
332 group->task_cycle_period_ns() != task_period_ns_) {
333 return BindingError::invalid_value;
334 }
335 (*group_targets_)[slot] = group;
336 }
337 handle = static_cast<std::uint64_t>(slot) + 1U;
338 std::memcpy(vars_ + var.offset, &handle, sizeof(handle));
339 return BindingError::ok;
340 }
341 return BindingError::unknown;
342 }

669 const char *step_name,
670 bool &active) const noexcept
671 {
672 active = false;
673 if(program_ == nullptr || network_name == nullptr ||
674 step_name == nullptr || sfc_runtime_ == nullptr)
675 return rt::ErrorCode::invalid_argument;
676 for(const SfcNetworkInfo &network : program_->sfc_networks) {
677 if(!ascii_name_equal(network_name, network.lower)) continue;
678 for(std::size_t index = 0; index < network.steps.size(); ++index) {
679 if(ascii_name_equal(step_name, network.steps[index].lower)) {
680 const std::uint32_t offset =
681 running() && sfc_runner_->phase != SfcRunnerPhase::stage_vars
682 ? network.shadow_active_offset
683 : network.committed_active_offset;
684 active = sfc_runtime_[offset + index] != 0;
685 return rt::ErrorCode::ok;
686 }
687 }
688 return rt::ErrorCode::invalid_argument;
689 }
690 return rt::ErrorCode::invalid_argument;
691 }

810 // Cycle path. budget = number of instructions this scan may execute;
811 // exact boundary: a program needing N instructions completes with
812 // budget N and faults with budget N-1 (matrix 5.5).
813 ScanError scan(std::int64_t budget)
814 {
815 const ScanError started = begin(budget);
816 if(started != ScanError::ok) return started;
817 std::int64_t executed = 0;
818 return resume(std::numeric_limits<std::int64_t>::max(), executed);
819 }
820
821 // Cooperative execution slice. The slice counts bytecode instructions;
822 // weighted L4 operation costs continue to charge only the task budget.
823 ScanError resume(std::int64_t slice, std::int64_t &executed)
824 {
825 executed = 0;
826 if(program_ == nullptr) return ScanError::not_loaded;
827 if(fault_ != ScanError::ok) return fault_;
828 if(execution_state_ != InstanceState::running) {
829 return execution_state_ == InstanceState::complete
830 ? ScanError::ok
831 : ScanError::invalid_argument;
832 }
833 if(slice <= 0) return ScanError::ok;
834 const std::uint8_t *code = nullptr;
835 std::size_t size = 0;
836 const std::uint64_t *constants = nullptr;
837 std::size_t constant_count = 0;
838 std::int32_t stack_limit = 0;
839 const auto select_code = [&]() {
840 if(sfc_runner_->active_region != nullptr) {
841 code = sfc_runner_->active_region->code.data();
842 size = sfc_runner_->active_region->code.size();
843 constants = sfc_runner_->active_region->constants.data();
844 constant_count = sfc_runner_->active_region->constants.size();
845 stack_limit = sfc_runner_->active_region->stack_slots;
846 } else {
847 code = program_->code.data();
848 size = program_->code.size();
849 constants = program_->constants.data();
850 constant_count = program_->constants.size();
851 stack_limit = program_->stack_slots;
852 }
853 };
854 select_code();
855 std::size_t &pc = pc_;
856 std::int32_t &sp = sp_;
857 std::int64_t &remaining = remaining_budget_;
858 const auto charge_operation = [&remaining](std::uint32_t cost) {
859 const std::int64_t extra = cost > 1U
860 ? static_cast<std::int64_t>(cost - 1U)
861 : 0;
862 if(remaining < extra) {
863 return false;
864 }
865 remaining -= extra;
866 return true;
867 };
868
869 for(;;) {
870 if(executed >= slice) return ScanError::ok;
871 if(!program_->sfc_networks.empty() &&
872 sfc_runner_->phase != SfcRunnerPhase::none &&
873 sfc_runner_->phase != SfcRunnerPhase::base && sfc_runner_->active_region == nullptr) {
874 if(remaining <= 0)
875 return latch(ScanError::budget_exceeded);
876 --remaining;
877 ++executed;
878 advance_sfc_runner();
879 if(sfc_runner_->active_region != nullptr) {
880 pc = 0;
881 sp = 0;
882 select_code();
883 continue;
884 }
885 if(sfc_runner_->phase == SfcRunnerPhase::none) {
886 execution_state_ = InstanceState::complete;
887 return ScanError::ok;
888 }
889 continue;
890 }
891 if(remaining <= 0) {
892 return latch(ScanError::budget_exceeded);
893 }
894 --remaining;
895 ++executed;
896 if(pc >= size) {
897 return latch(ScanError::invalid_bytecode);
898 }
899 instruction_pc_ = pc;
900 const Op op = static_cast<Op>(code[pc++]);
901 switch(op) {
902 case Op::debug_probe: {
903 std::uint32_t probe = 0;
904 if(!rd32(code, size, pc, probe))
905 return latch(ScanError::invalid_bytecode);
906 const SourceMap *map = sfc_runner_->active_region == nullptr
907 ? &program_->source_map
908 : &sfc_runner_->active_region->source_map;
909 if(probe >= map->entries.size())
910 return latch(ScanError::invalid_bytecode);
911 DebugRuntimeStorage *runtime = debug_runtime_storage();
912 if(runtime == nullptr)
913 return latch(ScanError::invalid_bytecode);
914 runtime->last_entry = &map->entries[probe];
915 runtime->last_instruction_id = runtime->last_entry->instruction_id;
916 runtime->last_instruction_id.mapping = runtime->mapping;
917 runtime->last_instruction_id.offset =
918 static_cast<std::uint32_t>(instruction_pc_);
919 if(debug_hooks_ != nullptr &&
920 debug_hooks_->on_probe(
921 runtime->mapping, map->entries[probe],
922 static_cast<std::uint32_t>(instruction_pc_))) {
923 pc = instruction_pc_;
924 ++remaining;
925 --executed;
926 return ScanError::paused;
927 }
928 ++remaining;
929 --executed;
930 break;
931 }
932 case Op::halt:
933 if(!program_->sfc_networks.empty()) {
934 if(sfc_runner_->active_region != nullptr)
935 finish_sfc_region();
936 else if(sfc_runner_->phase == SfcRunnerPhase::base)
937 begin_sfc_regions();
938 continue;
939 } else if(process_image_ != nullptr) {
940 process_image_->commit_scan(
941 *program_, vars_, !defer_process_image_publish_);
942 }
943 execution_state_ = InstanceState::complete;
944 return ScanError::ok;
945
946 case Op::push_const: {
947 std::uint16_t index = 0;
948 if(!rd16(code, size, pc, index) || index >= constant_count ||
949 sp >= stack_limit) {
950 return latch(ScanError::invalid_bytecode);
951 }
952 stack_[sp++] = constants[index];
953 break;
954 }
955 case Op::load_var: {
956 std::uint32_t offset = 0;
957 std::uint32_t type = 0;
958 std::uint64_t value = 0;
959 if(!rd32(code, size, pc, offset) ||
960 !rd32(code, size, pc, type) || sp >= stack_limit ||
961 !load_value(offset, type, value)) {
962 return latch(ScanError::invalid_bytecode);
963 }
964 stack_[sp++] = value;
965 break;
966 }
967 case Op::store_var: {
968 std::uint32_t offset = 0;
969 std::uint32_t type = 0;
970 if(!rd32(code, size, pc, offset) ||
971 !rd32(code, size, pc, type) || sp < 1 ||
972 !store_value(offset, type, stack_[sp - 1])) {
973 return latch(ScanError::invalid_bytecode);
974 }
975 --sp;
976 break;
977 }
978 case Op::load_access:
979 case Op::store_access: {
980 std::uint32_t offset = 0;
981 std::uint32_t type = 0;
982 if(!rd32(code, size, pc, offset) ||
983 !rd32(code, size, pc, type) || pc >= size) {
984 return latch(ScanError::invalid_bytecode);
985 }
986 const std::uint8_t count = code[pc++];
987 const int needed = static_cast<int>(count) +
988 (op == Op::store_access ? 1 : 0);
989 if(sp < needed ||
990 (op == Op::load_access &&
991 sp - static_cast<int>(count) >= stack_limit)) {
992 return latch(ScanError::invalid_bytecode);
993 }
994 const int index_base = sp - static_cast<int>(count);
995 for(std::uint8_t i = 0; i < count; ++i) {
996 std::uint64_t lower_raw = 0;
997 std::uint64_t upper_raw = 0;
998 std::uint64_t stride = 0;
999 if(!rd64(code, size, pc, lower_raw) ||
1000 !rd64(code, size, pc, upper_raw) ||
1001 !rd64(code, size, pc, stride)) {
1002 return latch(ScanError::invalid_bytecode);
1003 }
1004 const std::int64_t lower =
1005 static_cast<std::int64_t>(lower_raw);
1006 const std::int64_t upper =
1007 static_cast<std::int64_t>(upper_raw);
1008 const std::int64_t value = as_i64(
1009 stack_[index_base + static_cast<int>(i)]);
1010 if(value < lower || value > upper) {
1011 return latch(ScanError::range_violation);
1012 }
1013 const std::uint64_t delta =
1014 static_cast<std::uint64_t>(value - lower) * stride;
1015 if(delta > std::numeric_limits<std::uint32_t>::max() -
1016 offset) {
1017 return latch(ScanError::invalid_bytecode);
1018 }
1019 offset += static_cast<std::uint32_t>(delta);
1020 }
1021 if(op == Op::load_access) {
1022 std::uint64_t value = 0;
1023 sp -= count;
1024 if(!load_value(offset, type, value)) {
1025 return latch(ScanError::invalid_bytecode);
1026 }
1027 stack_[sp++] = value;
1028 } else {
1029 const std::uint64_t value =
1030 stack_[index_base - 1];
1031 if(!store_value(offset, type, value)) {
1032 return latch(ScanError::invalid_bytecode);
1033 }
1034 sp -= needed;
1035 }
1036 break;
1037 }
1038 case Op::copy_bytes: {
1039 std::uint32_t destination = 0;
1040 std::uint32_t source = 0;
1041 std::uint32_t count = 0;
1042 if(!rd32(code, size, pc, destination) ||
1043 !rd32(code, size, pc, source) ||
1044 !rd32(code, size, pc, count) ||
1045 destination > program_->vars_bytes ||
1046 count > program_->vars_bytes - destination ||
1047 source > program_->vars_bytes ||
1048 count > program_->vars_bytes - source) {
1049 return latch(ScanError::invalid_bytecode);
1050 }
1051 std::memmove(execution_vars_ + destination,
1052 execution_vars_ + source, count);
1053 break;
1054 }
1055 case Op::string_copy:
1056 case Op::string_copy_const: {
1057 std::uint32_t destination = 0;
1058 std::uint32_t destination_type = 0;
1059 std::uint32_t source = 0;
1060 std::uint32_t source_type = 0;
1061 if(!rd32(code, size, pc, destination) ||
1062 !rd32(code, size, pc, destination_type) ||
1063 !rd32(code, size, pc, source) ||
1064 !rd32(code, size, pc, source_type)) {
1065 return latch(ScanError::invalid_bytecode);
1066 }
1067 const TypeDesc *dst = program_->types.get(destination_type);
1068 const TypeDesc *src = program_->types.get(source_type);
1069 if(dst == nullptr || src == nullptr || dst->kind != src->kind ||
1070 (dst->kind != TypeKind::string &&
1071 dst->kind != TypeKind::wstring) ||
1072 destination > program_->vars_bytes ||
1073 dst->size > program_->vars_bytes - destination) {
1074 return latch(ScanError::invalid_bytecode);
1075 }
1076 const std::uint32_t operation_cost = std::max(
1077 dst->string.capacity, src->string.capacity);
1078 if(!charge_operation(operation_cost)) {
1079 return latch(ScanError::budget_exceeded);
1080 }
1081 const unsigned char *source_bytes = nullptr;
1082 if(op == Op::string_copy) {
1083 if(source > program_->vars_bytes ||
1084 src->size > program_->vars_bytes - source) {
1085 return latch(ScanError::invalid_bytecode);
1086 }
1087 source_bytes = execution_vars_ + source;
1088 } else {
1089 if(source > program_->string_constants.size() ||
1090 4U > program_->string_constants.size() - source) {
1091 return latch(ScanError::invalid_bytecode);
1092 }
1093 source_bytes = program_->string_constants.data() + source;
1094 }
1095 const std::uint32_t length = read_le32(source_bytes);
1096 if(length > src->string.capacity) {
1097 return latch(ScanError::invalid_bytecode);
1098 }
1099 const std::uint64_t width =
1100 dst->kind == TypeKind::wstring ? 4U : 1U;
1101 const std::uint64_t count = 4U + width * length;
1102 if(op == Op::string_copy_const &&
1103 count > program_->string_constants.size() - source) {
1104 return latch(ScanError::invalid_bytecode);
1105 }
1106 if(!valid_string_object(source_bytes, *src)) {
1107 return latch(ScanError::invalid_bytecode);
1108 }
1109 if(length > dst->string.capacity) {
1110 return latch(ScanError::string_capacity_exceeded);
1111 }
1112 if(op == Op::string_copy && source == destination) {
1113 break;
1114 }
1115 std::memset(execution_vars_ + destination, 0,
1116 static_cast<std::size_t>(dst->size));
1117 std::memmove(execution_vars_ + destination, source_bytes,
1118 static_cast<std::size_t>(count));
1119 break;
1120 }
1121 case Op::string_compare: {
1122 if(pc >= size) {
1123 return latch(ScanError::invalid_bytecode);
1124 }
1125 const std::uint8_t comparison = code[pc++];
1126 const unsigned char *objects[2] = {nullptr, nullptr};
1127 const TypeDesc *descs[2] = {nullptr, nullptr};
1128 for(int i = 0; i < 2; ++i) {
1129 if(pc >= size) {
1130 return latch(ScanError::invalid_bytecode);
1131 }
1132 const std::uint8_t constant = code[pc++];
1133 std::uint32_t offset = 0;
1134 std::uint32_t id = 0;
1135 if(!rd32(code, size, pc, offset) ||
1136 !rd32(code, size, pc, id)) {
1137 return latch(ScanError::invalid_bytecode);
1138 }
1139 descs[i] = program_->types.get(id);
1140 if(descs[i] == nullptr ||
1141 (descs[i]->kind != TypeKind::string &&
1142 descs[i]->kind != TypeKind::wstring)) {
1143 return latch(ScanError::invalid_bytecode);
1144 }
1145 if(constant != 0) {
1146 if(offset > program_->string_constants.size() ||
1147 4U > program_->string_constants.size() - offset) {
1148 return latch(ScanError::invalid_bytecode);
1149 }
1150 objects[i] = program_->string_constants.data() + offset;
1151 const std::uint32_t length = read_le32(objects[i]);
1152 const std::uint64_t object_size = 4U +
1153 static_cast<std::uint64_t>(length) *
1154 (descs[i]->kind == TypeKind::wstring ? 4U : 1U);
1155 if(length > descs[i]->string.capacity ||
1156 object_size >
1157 program_->string_constants.size() - offset) {
1158 return latch(ScanError::invalid_bytecode);
1159 }
1160 } else {
1161 if(offset > program_->vars_bytes ||
1162 descs[i]->size > program_->vars_bytes - offset) {
1163 return latch(ScanError::invalid_bytecode);
1164 }
1165 objects[i] = execution_vars_ + offset;
1166 }
1167 }
1168 if(descs[0]->kind != descs[1]->kind || sp >= stack_limit) {
1169 return latch(ScanError::invalid_bytecode);
1170 }
1171 const std::uint32_t operation_cost = std::max(
1172 descs[0]->string.capacity, descs[1]->string.capacity);
1173 if(!charge_operation(operation_cost)) {
1174 return latch(ScanError::budget_exceeded);
1175 }
1176 const int order = compare_strings(objects[0], *descs[0],
1177 objects[1], *descs[1]);
1178 if(order == 2) {
1179 return latch(ScanError::invalid_bytecode);
1180 }
1181 bool result = false;
1182 switch(comparison) {
1183 case 8: result = order == 0; break;
1184 case 9: result = order != 0; break;
1185 case 10: result = order < 0; break;
1186 case 11: result = order > 0; break;
1187 case 12: result = order <= 0; break;
1188 case 13: result = order >= 0; break;
1189 default: return latch(ScanError::invalid_bytecode);
1190 }
1191 stack_[sp++] = result ? 1U : 0U;
1192 break;
1193 }
1194 case Op::string_index: {
1195 std::uint32_t offset = 0;
1196 std::uint32_t id = 0;
1197 if(!rd32(code, size, pc, offset) ||
1198 !rd32(code, size, pc, id) || sp < 1) {
1199 return latch(ScanError::invalid_bytecode);
1200 }
1201 const TypeDesc *desc = program_->types.get(id);
1202 if(desc == nullptr ||
1203 (desc->kind != TypeKind::string &&
1204 desc->kind != TypeKind::wstring) ||
1205 offset > program_->vars_bytes ||
1206 desc->size > program_->vars_bytes - offset) {
1207 return latch(ScanError::invalid_bytecode);
1208 }
1209 if(!charge_operation(desc->string.capacity)) {
1210 return latch(ScanError::budget_exceeded);
1211 }
1212 if(!valid_string_object(execution_vars_ + offset, *desc)) {
1213 return latch(ScanError::invalid_bytecode);
1214 }
1215 const std::int64_t index = as_i64(stack_[sp - 1]);
1216 const std::uint32_t length = read_le32(execution_vars_ + offset);
1217 if(index < 1 || static_cast<std::uint64_t>(index) > length) {
1218 return latch(ScanError::range_violation);
1219 }
1220 if(desc->kind == TypeKind::wstring) {
1221 stack_[sp - 1] = read_le32(
1222 execution_vars_ + offset + 4U +
1223 (static_cast<std::size_t>(index) - 1U) * 4U);
1224 } else {
1225 stack_[sp - 1] = execution_vars_[offset + 4U +
1226 static_cast<std::uint32_t>(index) - 1U];
1227 }
1228 break;
1229 }
1230 case Op::check_unicode:
1231 if(sp < 1) {
1232 return latch(ScanError::invalid_bytecode);
1233 }
1234 if(stack_[sp - 1] > 0x10FFFFU ||
1235 (stack_[sp - 1] >= 0xD800U &&
1236 stack_[sp - 1] <= 0xDFFFU)) {
1237 return latch(ScanError::conversion_invalid);
1238 }
1239 break;
1240 case Op::date_arith: {
1241 if(pc >= size || sp < 2) {
1242 return latch(ScanError::invalid_bytecode);
1243 }
1244 const std::uint8_t kind = code[pc++];
1245 const std::int64_t right = as_i64(stack_[--sp]);
1246 const std::int64_t left = as_i64(stack_[sp - 1]);
1247 std::int64_t value = 0;
1248 if(kind <= 1U) {
1249 const std::int64_t delta = kind == 0U ? right : -right;
1250 value = left + delta;
1251 if(value < std::numeric_limits<std::int32_t>::min() ||
1252 value > std::numeric_limits<std::int32_t>::max()) {
1253 return latch(ScanError::date_time_range_violation);
1254 }
1255 } else if(kind <= 3U) {
1256 constexpr std::int64_t day = 86400000000000LL;
1257 const std::int64_t delta = right % day;
1258 value = kind == 2U ? left + delta : left - delta;
1259 value %= day;
1260 if(value < 0) {
1261 value += day;
1262 }
1263 } else if(kind <= 5U) {
1264 if((kind == 4U &&
1265 ((right > 0 && left >
1266 std::numeric_limits<std::int64_t>::max() - right) ||
1267 (right < 0 && left <
1268 std::numeric_limits<std::int64_t>::min() - right))) ||
1269 (kind == 5U &&
1270 ((right < 0 && left >
1271 std::numeric_limits<std::int64_t>::max() + right) ||
1272 (right > 0 && left <
1273 std::numeric_limits<std::int64_t>::min() + right)))) {
1274 return latch(ScanError::date_time_range_violation);
1275 }
1276 value = kind == 4U ? left + right : left - right;
1277 } else {
1278 return latch(ScanError::invalid_bytecode);
1279 }
1280 stack_[sp - 1] = static_cast<std::uint64_t>(value);
1281 break;
1282 }
1283
1284 case Op::add_int:
1285 case Op::sub_int:
1286 case Op::mul_int:
1287 case Op::add_dint:
1288 case Op::sub_dint:
1289 case Op::mul_dint:
1290 case Op::add_time:
1291 case Op::sub_time: {
1292 if(sp < 2) {
1293 return latch(ScanError::invalid_bytecode);
1294 }
1295 const std::int64_t b = as_i64(stack_[--sp]);
1296 const std::int64_t a = as_i64(stack_[--sp]);
1297 std::int64_t value = 0;
1298 switch(op) {
1299 case Op::add_int:
1300 value = detail::wrap16(detail::wrap_add64(a, b));
1301 break;
1302 case Op::sub_int:
1303 value = detail::wrap16(detail::wrap_sub64(a, b));
1304 break;
1305 case Op::mul_int:
1306 value = detail::wrap16(detail::wrap_mul64(a, b));
1307 break;
1308 case Op::add_dint:
1309 value = detail::wrap32(detail::wrap_add64(a, b));
1310 break;
1311 case Op::sub_dint:
1312 value = detail::wrap32(detail::wrap_sub64(a, b));
1313 break;
1314 case Op::mul_dint:
1315 value = detail::wrap32(detail::wrap_mul64(a, b));
1316 break;
1317 case Op::add_time:
1318 value = detail::wrap_add64(a, b);
1319 break;
1320 default:
1321 value = detail::wrap_sub64(a, b);
1322 break;
1323 }
1324 stack_[sp++] = as_u64(value);
1325 break;
1326 }

1328 case Op::div_int:
1329 case Op::mod_int:

3979 static bool next_utf8(const unsigned char *bytes, std::uint32_t length,
3980 std::uint32_t &at, std::uint32_t &scalar)
3981 {
3982 if(at >= length) {
3983 return false;
3984 }
3985 const std::uint8_t lead = bytes[at++];
3986 unsigned count = 0;
3987 if(lead < 0x80U) {
3988 scalar = lead;
3989 return true;
3990 }
3991 if(lead >= 0xC2U && lead <= 0xDFU) {
3992 scalar = lead & 0x1FU;
3993 count = 1;
3994 } else if(lead >= 0xE0U && lead <= 0xEFU) {
3995 scalar = lead & 0x0FU;
3996 count = 2;
3997 } else if(lead >= 0xF0U && lead <= 0xF4U) {
3998 scalar = lead & 0x07U;
3999 count = 3;
4000 } else {
4001 return false;
4002 }
4003 if(count > length - at) {
4004 return false;
4005 }
4006 for(unsigned i = 0; i < count; ++i) {
4007 const std::uint8_t part = bytes[at++];
4008 if((part & 0xC0U) != 0x80U) {
4009 return false;
4010 }
4011 if(i == 0U &&
4012 ((lead == 0xE0U && part < 0xA0U) ||
4013 (lead == 0xEDU && part > 0x9FU) ||
4014 (lead == 0xF0U && part < 0x90U) ||
4015 (lead == 0xF4U && part > 0x8FU))) {
4016 return false;
4017 }
4018 scalar = (scalar << 6U) | (part & 0x3FU);
4019 }
4020 return scalar <= 0x10FFFFU &&
4021 !(scalar >= 0xD800U && scalar <= 0xDFFFU);
4022 }

4058 const unsigned char *right,

4069 ? left_length
