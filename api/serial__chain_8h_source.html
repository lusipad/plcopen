217 {
218 double squared = 0.0;

226 static void pose_error(const Pose6 &current, const Pose6 &target, double error[6],
227 double &position_residual, double &orientation_residual)

239 static void relative_rotation_log(const double from[3][3], const double to[3][3],
240 double base_log[3])
241 {
242 double relative[3][3];
243 for (int row = 0; row < 3; ++row)
244 {
245 for (int column = 0; column < 3; ++column)
246 {
247 relative[row][column] = from[0][row] * to[0][column] +
248 from[1][row] * to[1][column] + from[2][row] * to[2][column];
249 }
250 }
251 double local_log[3] = {0.5 * (relative[2][1] - relative[1][2]),
252 0.5 * (relative[0][2] - relative[2][0]),
253 0.5 * (relative[1][0] - relative[0][1])};
254 const double sine = vector_norm(local_log, 3);
255 double cosine = 0.5 * (relative[0][0] + relative[1][1] + relative[2][2] - 1.0);
256 cosine = cosine > 1.0 ? 1.0 : (cosine < -1.0 ? -1.0 : cosine);
257 if (sine > 1e-12)
258 {
259 const double scale = std::atan2(sine, cosine) / sine;
260 for (double &value : local_log)
261 {
262 value *= scale;
263 }
264 }
265 else if (cosine < 0.0)
266 {
267 double axis[3];
268 double angle = 0.0;
269 geom::relative_axis_angle(from, to, axis, angle);
270 for (int row = 0; row < 3; ++row)
271 {
272 local_log[row] = axis[row] * angle;
273 }
274 }
275 for (int row = 0; row < 3; ++row)
276 {
277 base_log[row] = from[row][0] * local_log[0] + from[row][1] * local_log[1] +
278 from[row][2] * local_log[2];
279 }
280 }

282 static void relative_rotation_increment(const double from[3][3], const double to[3][3],
283 double base_increment[3])
284 {
285 // Numerical Jacobian columns use a fixed 1e-7-rad perturbation. The
286 // skew part is the one-sided infinitesimal rotation increment; the
287 // full SO(3) log remains the convergence/error oracle.
288 double relative[3][3];
289 for (int row = 0; row < 3; ++row)
290 {
291 for (int column = 0; column < 3; ++column)
292 {
293 relative[row][column] = from[0][row] * to[0][column] +
294 from[1][row] * to[1][column] +
295 from[2][row] * to[2][column];
296 }
297 }
298 const double local_increment[3] = {
299 0.5 * (relative[2][1] - relative[1][2]),
300 0.5 * (relative[0][2] - relative[2][0]),
301 0.5 * (relative[1][0] - relative[0][1])};
302 for (int row = 0; row < 3; ++row)
303 {
304 base_increment[row] = from[row][0] * local_increment[0] +
305 from[row][1] * local_increment[1] +
306 from[row][2] * local_increment[2];
307 }
308 }

322 geom::RigidTransform link_transforms[MaxJoints],
323 geom::RigidTransform prefixes[MaxJoints + 1]) const
324 {

334 void numerical_jacobian(
335 const double *joints, const Pose6 &current,
336 const geom::RigidTransform link_transforms[MaxJoints],
337 const geom::RigidTransform prefixes[MaxJoints + 1],
338 double jacobian[6][MaxJoints]) const
339 {
340 // A one-joint perturbation leaves every other local link transform
341 // unchanged. Reuse the current-pose products so each numerical-
342 // Jacobian column recomposes only the affected suffix.
343 for (std::size_t column = 0; column < spec_.joint_count; ++column)
344 {
345 geom::RigidTransform perturbed_transform =
346 geom::compose(prefixes[column],
347 link_transform(column, joints[column] + DifferenceStep));
348 for (std::size_t downstream = column + 1; downstream < spec_.joint_count;
349 ++downstream)
350 {
351 perturbed_transform =
352 geom::compose(perturbed_transform, link_transforms[downstream]);
353 }
354 Pose6 perturbed{};
355 copy_pose(perturbed_transform, perturbed);
356 for (int row = 0; row < 3; ++row)
357 {
358 jacobian[row][column] =
359 (perturbed.position[row] - current.position[row]) / DifferenceStep;
360 }
361 double rotation_delta[3];
362 relative_rotation_increment(current.rotation, perturbed.rotation, rotation_delta);
363 for (int row = 0; row < 3; ++row)
364 {
365 jacobian[row + 3][column] = rotation_delta[row] / DifferenceStep;
366 }
367 }
368 for (std::size_t column = spec_.joint_count; column < MaxJoints; ++column)
369 {
370 for (int row = 0; row < 6; ++row)
371 {
372 jacobian[row][column] = 0.0;
373 }
374 }
375 }

377 static bool solve_linear(double matrix[6][6], const double right_hand_side[6],
378 double solution[6])
379 {
380 double augmented[6][7] = {};
381 for (int row = 0; row < 6; ++row)
382 {
383 for (int column = 0; column < 6; ++column)
384 {
385 augmented[row][column] = matrix[row][column];
386 }
387 augmented[row][6] = right_hand_side[row];
388 }
389
390 for (int column = 0; column < 6; ++column)
391 {
392 int pivot_row = column;
393 double pivot_magnitude = std::fabs(augmented[column][column]);
394 for (int row = column + 1; row < 6; ++row)
395 {
396 const double candidate = std::fabs(augmented[row][column]);
397 if (candidate > pivot_magnitude)
398 {
399 pivot_magnitude = candidate;
400 pivot_row = row;
401 }
402 }
403 if (!std::isfinite(pivot_magnitude) || pivot_magnitude < 1e-24)
404 {
405 return false;
406 }
407 if (pivot_row != column)
408 {
409 for (int item = column; item < 7; ++item)
410 {
411 const double saved = augmented[column][item];
412 augmented[column][item] = augmented[pivot_row][item];
413 augmented[pivot_row][item] = saved;
414 }
415 }
416 for (int row = column + 1; row < 6; ++row)
417 {
418 const double factor = augmented[row][column] / augmented[column][column];
419 for (int item = column; item < 7; ++item)
420 {
421 augmented[row][item] -= factor * augmented[column][item];
422 }
423 }
424 }
425
426 for (int row = 5; row >= 0; --row)
427 {
428 double value = augmented[row][6];
429 for (int column = row + 1; column < 6; ++column)
430 {
431 value -= augmented[row][column] * solution[column];
432 }
433 const double pivot = augmented[row][row];
434 if (!std::isfinite(pivot) || std::fabs(pivot) < 1e-24)
435 {
436 return false;
437 }
438 solution[row] = value / pivot;
439 if (!std::isfinite(solution[row]))
440 {
441 return false;
442 }
443 }
444 return true;
445 }

447 void damped_matrix(const double jacobian[6][MaxJoints], double damping,
448 double matrix[6][6]) const
449 {
450 for (int row = 0; row < 6; ++row)
451 {
452 for (int column = 0; column < 6; ++column)
453 {
454 double value = 0.0;
455 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
456 {
457 value += jacobian[row][joint] * jacobian[column][joint];
458 }
459 if (row == column)
460 {
461 value += damping * damping;
462 }
463 matrix[row][column] = value;
464 }
465 }

468 bool damped_task_step(const double jacobian[6][MaxJoints], const double error[6],
469 double damping, double step[MaxJoints]) const
470 {
471 double matrix[6][6];
472 damped_matrix(jacobian, damping, matrix);
473 double task_solution[6] = {};
474 if (!solve_linear(matrix, error, task_solution))
475 {
476 return false;
477 }
478 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
479 {
480 step[joint] = 0.0;
481 for (int row = 0; row < 6; ++row)
482 {
483 step[joint] += jacobian[row][joint] * task_solution[row];
484 }
485 }
486 return true;
487 }

489 bool nullspace_step(const double jacobian[6][MaxJoints], const double *joints,
490 const SerialChainSolveOptions &options, double damping,
491 double step[MaxJoints]) const
492 {
493 double gradient[MaxJoints] = {};
494 double projected[6] = {};
495 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
496 {
497 gradient[joint] =
498 options.preference_weight * (options.preferred_joints[joint] - joints[joint]);
499 for (int row = 0; row < 6; ++row)
500 {
501 projected[row] += jacobian[row][joint] * gradient[joint];
502 }
503 }
504 double matrix[6][6];
505 damped_matrix(jacobian, damping, matrix);
506 double task_component[6] = {};
507 if (!solve_linear(matrix, projected, task_component))
508 {
509 return false;
510 }
511 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
512 {
513 step[joint] = gradient[joint];
514 for (int row = 0; row < 6; ++row)
515 {
516 step[joint] -= jacobian[row][joint] * task_component[row];
517 }
518 }
519 return true;
520 }

532 }
533 else if (candidate[joint] > link.max_position)
534 {
535 candidate[joint] = link.max_position;
536 limit_clipped = true;
537 }
538 const double step_lower = seed[joint] - max_joint_step;
539 const double step_upper = seed[joint] + max_joint_step;
540 if (candidate[joint] < step_lower)
541 {
542 candidate[joint] = step_lower;
543 step_clipped = true;
544 }
545 else if (candidate[joint] > step_upper)
546 {
547 candidate[joint] = step_upper;
548 step_clipped = true;
549 }
550 if (candidate[joint] < link.min_position)
551 {
552 candidate[joint] = link.min_position;
553 limit_clipped = true;
554 }
555 else if (candidate[joint] > link.max_position)
556 {
557 candidate[joint] = link.max_position;
558 limit_clipped = true;

574 // Called only from the bounded outer loop. An accepted secondary step
575 // rechecks both primary gates and consumes one refinement from the same
576 // MaxIterations budget.
577 bool apply_in_loop_preference_step(const Pose6 &target, const double *seed,
578 double max_joint_step,
579 const SerialChainSolveOptions &options,
580 double joints[MaxJoints], const Pose6 &current,
581 const geom::RigidTransform link_transforms[MaxJoints],
582 const geom::RigidTransform prefixes[MaxJoints + 1],
583 double &position_residual,
584 double &orientation_residual) const
585 {
586 if (spec_.joint_count <= 6 || options.preference_weight == 0.0)
587 {
588 return false;
589 }
590 double jacobian[6][MaxJoints] = {};
591 numerical_jacobian(joints, current, link_transforms, prefixes, jacobian);
592 double secondary_step[MaxJoints] = {};
593 if (!nullspace_step(jacobian, joints, options, MinimumDamping, secondary_step))
594 {
595 return false;
596 }
597 const double current_cost = preference_cost(joints, options);
598 double scale = 1.0;
599 for (std::size_t line = 0; line < LineSearchSteps; ++line)
600 {
601 double candidate[MaxJoints] = {};
602 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
603 {
604 candidate[joint] = joints[joint] + scale * secondary_step[joint];
605 }
606 bool limit_clipped = false;
607 bool step_clipped = false;
608 project_candidate(seed, max_joint_step, candidate, limit_clipped, step_clipped);
609 Pose6 candidate_pose{};
610 forward_from(candidate, candidate_pose);
611 double error[6];
612 double candidate_position = 0.0;
613 double candidate_orientation = 0.0;
614 pose_error(candidate_pose, target, error, candidate_position, candidate_orientation);
615 if (candidate_position <= options.position_tolerance &&
616 candidate_orientation <= options.orientation_tolerance &&
617 preference_cost(candidate, options) < current_cost)
618 {
619 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
620 {
621 joints[joint] = candidate[joint];
622 }
623 position_residual = candidate_position;
624 orientation_residual = candidate_orientation;
625 return true;
626 }
627 scale *= 0.5;
628 }
629 return false;
630 }

632 SerialChainSolveResult solve_internal(const Pose6 &target, const double *seed,
633 double max_joint_step,
634 const SerialChainSolveOptions &options) const
635 {
636 SerialChainSolveResult result{};
637 if (!valid_solve_inputs(target, seed, max_joint_step, options))
638 {
639 result.code = rt::ErrorCode::invalid_argument;
640 return result;
641 }
642
643 double joints[MaxJoints] = {};
644 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
645 {
646 // PoseKinematics carries no array extent; callers provide one seed
647 // per validated joint_count element.
648 // NOLINTNEXTLINE(clang-analyzer-core.uninitialized.Assign)
649 const double seed_value = seed[joint];
650 if (seed_value < spec_.links[joint].min_position ||
651 seed_value > spec_.links[joint].max_position)
652 {
653 result.code = rt::ErrorCode::limit_infeasible;
654 return result;
655 }
656 joints[joint] = seed_value;
657 result.joints[joint] = seed_value;
658 }
659
660 double best_score = 1e300;
661 double damping = MinimumDamping;
662 geom::RigidTransform link_transforms[MaxJoints] = {};
663 geom::RigidTransform prefixes[MaxJoints + 1] = {};
664 for (std::size_t iteration = 0; iteration < MaxIterations; ++iteration)
665 {
666 Pose6 current{};
667 evaluate_chain(joints, current, link_transforms, prefixes);
668 double error[6];
669 double position_residual = 0.0;
670 double orientation_residual = 0.0;
671 pose_error(current, target, error, position_residual, orientation_residual);
672 const double current_score = std::sqrt(position_residual * position_residual +
673 orientation_residual * orientation_residual);
674 if (current_score < best_score)
675 {
676 best_score = current_score;
677 result.position_residual = position_residual;
678 result.orientation_residual = orientation_residual;
679 result.iterations = iteration;
680 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
681 {
682 result.joints[joint] = joints[joint];
683 }
684 }
685
686 if (position_residual <= options.position_tolerance &&
687 orientation_residual <= options.orientation_tolerance)
688 {
689 double preference_position = 0.0;
690 double preference_orientation = 0.0;
691 if (apply_in_loop_preference_step(target, seed, max_joint_step, options, joints,
692 current, link_transforms, prefixes,
693 preference_position, preference_orientation))
694 {
695 result.code = rt::ErrorCode::ok;
696 result.position_residual = preference_position;
697 result.orientation_residual = preference_orientation;
698 result.iterations = iteration + 1;
699 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
700 {
701 result.joints[joint] = joints[joint];
702 }
703 return result;
704 }
705 result.code = rt::ErrorCode::ok;
706 result.position_residual = position_residual;
707 result.orientation_residual = orientation_residual;
708 result.iterations = iteration;
709 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
710 {
711 result.joints[joint] = joints[joint];
712 }
713 return result;
714 }
715
716 double jacobian[6][MaxJoints] = {};
717 numerical_jacobian(joints, current, link_transforms, prefixes, jacobian);
718 bool accepted = false;
719 bool any_limit_clipped = false;
720 bool any_step_clipped = false;
721 double attempt_damping = damping;
722 for (std::size_t damping_attempt = 0; damping_attempt < DampingAttempts;
723 ++damping_attempt)
724 {
725 double step[MaxJoints] = {};
726 if (!damped_task_step(jacobian, error, attempt_damping, step))
727 {
728 attempt_damping *= 10.0;
729 if (attempt_damping > MaximumDamping)
730 {
731 attempt_damping = MaximumDamping;
732 }
733 continue;
734 }
735 double scale = 1.0;
736 for (std::size_t line = 0; line < LineSearchSteps; ++line)
737 {
738 double candidate[MaxJoints] = {};
739 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
740 {
741 candidate[joint] = joints[joint] + scale * step[joint];
742 }
743 bool limit_clipped = false;
744 bool step_clipped = false;
745 project_candidate(seed, max_joint_step, candidate, limit_clipped, step_clipped);
746 any_limit_clipped = any_limit_clipped || limit_clipped;
747 any_step_clipped = any_step_clipped || step_clipped;
748
749 Pose6 candidate_pose{};
750 forward_from(candidate, candidate_pose);
751 double candidate_error[6];
752 double candidate_position = 0.0;
753 double candidate_orientation = 0.0;
754 pose_error(candidate_pose, target, candidate_error, candidate_position,
755 candidate_orientation);
756 const double candidate_score =
757 std::sqrt(candidate_position * candidate_position +
758 candidate_orientation * candidate_orientation);
759 if (candidate_score < current_score)
760 {
761 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
762 {
763 joints[joint] = candidate[joint];
764 }
765 damping = attempt_damping * 0.2;
766 if (damping < MinimumDamping)
767 {
768 damping = MinimumDamping;
769 }
770 accepted = true;
771 break;
772 }
773 scale *= 0.5;
774 }
775 if (accepted)
776 {
777 break;
778 }
779 attempt_damping *= 10.0;
780 if (attempt_damping > MaximumDamping)
781 {
782 attempt_damping = MaximumDamping;
783 }
784 }
785
786 if (!accepted)
787 {
788 result.code = any_limit_clipped
789 ? rt::ErrorCode::limit_infeasible
790 : (any_step_clipped ? rt::ErrorCode::infeasible
791 : rt::ErrorCode::singular_region);
792 return result;
793 }
794 }
795
796 Pose6 final_pose{};
797 forward_from(joints, final_pose);
798 double final_error[6];
799 double final_position = 0.0;
800 double final_orientation = 0.0;
801 pose_error(final_pose, target, final_error, final_position, final_orientation);
802 const double final_score =
803 std::sqrt(final_position * final_position + final_orientation * final_orientation);
804 if (final_score < best_score)
805 {
806 result.position_residual = final_position;
807 result.orientation_residual = final_orientation;
808 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
809 {
810 result.joints[joint] = joints[joint];
811 }
812 }
813 if (final_position <= options.position_tolerance &&
814 final_orientation <= options.orientation_tolerance)
815 {
816 result.code = rt::ErrorCode::ok;
817 result.position_residual = final_position;
818 result.orientation_residual = final_orientation;
819 result.iterations = MaxIterations;
820 for (std::size_t joint = 0; joint < spec_.joint_count; ++joint)
821 {
822 result.joints[joint] = joints[joint];
823 }
824 return result;
825 }
826 result.code = rt::ErrorCode::not_converged;
827 result.iterations = MaxIterations;
828 return result;
829 }

831 geom::RigidTransform link_transform(std::size_t index, double joint) const
832 {
833 const DhLink &link = spec_.links[index];
834 const double theta = joint + link.theta_offset;
835 const double ct = std::cos(theta);
836 const double st = std::sin(theta);
837 const double ca = alpha_cosines_[index];
838 const double sa = alpha_sines_[index];

840 if (spec_.convention == DhConvention::standard)
841 {
842 transform.rotation[0][0] = ct;
843 transform.rotation[0][1] = -st * ca;
844 transform.rotation[0][2] = st * sa;
845 transform.rotation[1][0] = st;
846 transform.rotation[1][1] = ct * ca;
847 transform.rotation[1][2] = -ct * sa;
848 transform.rotation[2][0] = 0.0;
849 transform.rotation[2][1] = sa;

853 }

857 transform.rotation[0][2] = 0.0;

859 transform.rotation[1][1] = ct * ca;
860 transform.rotation[1][2] = -sa;
861 transform.rotation[2][0] = st * sa;
862 transform.rotation[2][1] = ct * sa;
863 transform.rotation[2][2] = ca;
864 transform.translation = geom::Vec3{link.a, -link.d * sa, link.d * ca};
865 return transform;
