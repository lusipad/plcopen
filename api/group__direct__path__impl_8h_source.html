9inline rt::Result<std::uint32_t> AxisGroup::submit_direct(GroupCommand command)
10{
11 const bool blending = command.buffer_mode == BufferMode::blending_low ||
12 command.buffer_mode == BufferMode::blending_high;
13 if(status_ != GroupStatus::standby && status_ != GroupStatus::moving) {
14 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument);
15 }
16 if(command.target.size != axes_.size() || !finite(command.target) ||
17 !std::isfinite(command.velocity) || command.velocity <= 0.0 ||
18 !std::isfinite(command.acceleration) || command.acceleration <= 0.0 ||
19 !std::isfinite(command.deceleration) || command.deceleration <= 0.0 ||
20 !std::isfinite(command.jerk) || command.jerk <= 0.0 ||
21 !std::isfinite(command.transition_velocity) ||
22 command.transition_velocity < 0.0 ||
23 command.transition_velocity > command.velocity ||
24 !std::isfinite(command.transition_parameter)) {
25 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument);
26 }
27 if(blending) {
28 if(command.transition_mode != TransitionMode::max_corner_deviation) {
29 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::unsupported);
30 }
31 if(command.transition_parameter <= 0.0) {
32 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument);
33 }
34 command.kind = GroupCommandKind::motion;
35 command.direct_semantics = true;
36 return submit_linear(command);
37 }
38 if(command.buffer_mode != BufferMode::aborting &&
39 command.buffer_mode != BufferMode::buffered) {
40 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::unsupported);
41 }
42 if(command.transition_mode != TransitionMode::none) {
43 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::unsupported);
44 }
45 if(command.transition_velocity != 0.0 || command.transition_parameter != 0.0) {
46 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument);
47 }
48 const rt::ErrorCode framed = apply_coordinate_frame(command);
49 if(framed != rt::ErrorCode::ok) {
50 return rt::Result<std::uint32_t>::failure(framed);
51 }
52 command = normalize(command);
53 command.kind = GroupCommandKind::direct;
54 command.direct_semantics = true;
55 command.command_id = next_command_id_++;
56 if(!members_ready_for_group_motion()) {
57 return rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument);
58 }
59 const rt::ErrorCode limited = preflight_member_targets(command.target);
60 if(limited != rt::ErrorCode::ok) {
61 return rt::Result<std::uint32_t>::failure(limited);
62 }
63 if(command.buffer_mode == BufferMode::aborting) {
64 abort_wait();
65 abort_halt();
66 abort_direct_members();
67 abort_motion();
68 }
69 if(command.buffer_mode == BufferMode::aborting ||
70 (!active_ && !direct_path_.direct_active_ && !joint_window_.active_ &&
71 !wait_blocks_motion_start())) {
72 const rt::ErrorCode started = start_direct(command);
73 if(started != rt::ErrorCode::ok) {
74 return rt::Result<std::uint32_t>::failure(started);
75 }
76 } else {
77 const rt::ErrorCode queued = queue_.push_back(command);
78 if(queued != rt::ErrorCode::ok) {
79 return rt::Result<std::uint32_t>::failure(queued);
80 }
81 }
82 return rt::Result<std::uint32_t>::success(command.command_id);
83}

145inline rt::ErrorCode AxisGroup::start_direct(const GroupCommand &command)
146{
147 if(!members_ready_for_group_motion()) return rt::ErrorCode::invalid_argument;
148 std::array<AxisCommand, MaxAxes> commands{};
149 for(std::size_t i = 0; i < axes_.size(); ++i) {
150 AxisCommand &member = commands[i];
151 member.kind = CommandKind::move_absolute;
152 member.value = command.target.value[i];
153 member.velocity = command.velocity;
154 member.acceleration = command.acceleration;
155 member.deceleration = command.deceleration;
156 member.jerk = command.jerk;
157 const rt::ErrorCode preflight = axes_[i]->preflight_group_owned(member);
158 if(preflight != rt::ErrorCode::ok) return preflight;
159 }
160 snapshot_selections();
161 snapshot_active_tool_transform();
162 for(std::size_t i = 0; i < axes_.size(); ++i) {
163 const rt::Result<std::uint32_t> submitted =
164 axes_[i]->submit_group_owned(commands[i]);
165 if(!submitted) {
166 abort_direct_members();
167 set_group_error(submitted.error());
168 return submitted.error();
169 }
170 }
171 direct_path_.direct_active_ = true;
172 direct_path_.direct_stopping_ = false;
173 direct_path_.direct_command_id_ = command.command_id;
174 for(std::size_t i = 0; i < axes_.size(); ++i) {
175 active_finish_[i] = command.target.value[i];
176 }
177 status_ = GroupStatus::moving;
178 return rt::ErrorCode::ok;
179}

191inline rt::ErrorCode AxisGroup::stop_direct_members(double deceleration, double jerk)
192{
193 for(std::size_t i = 0; i < axes_.size(); ++i) {
194 if(!axes_[i]->powered() || axes_[i]->status() == AxisStatus::errorstop) {
195 abort_direct_members();
196 abort_motion();
197 set_group_error(rt::ErrorCode::precondition_failed);
198 return rt::ErrorCode::precondition_failed;
199 }
200 }
201
202 for(std::size_t i = 0; i < axes_.size(); ++i) {
203 AxisCommand halt{};
204 halt.kind = CommandKind::halt;
205 halt.velocity = 1.0;
206 halt.acceleration = deceleration;
207 halt.deceleration = deceleration;
208 halt.jerk = jerk;
209 const rt::Result<std::uint32_t> submitted = axes_[i]->submit_group_owned(halt);
210 if(!submitted) {
211 direct_path_.last_aborted_direct_id_ = direct_path_.direct_command_id_;
212 abort_direct_members();
213 abort_motion();
214 status_ = GroupStatus::standby;
215 return submitted.error();
216 }
217 }
218 direct_path_.direct_stopping_ = true;
219 status_ = GroupStatus::stopping;
220 return rt::ErrorCode::ok;
221}
