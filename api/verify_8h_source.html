49// Returns ok when the plugin honors the contract over `iterations` random
50// joint states; the first violation aborts with the failing error code.
51inline rt::ErrorCode verify_kinematics(const Kinematics &plugin,
52 const VerifyRange *joint_ranges,
53 int iterations,
54 double tolerance,
55 VerifyReport &report)
56{
57 const std::size_t joints = plugin.joint_count();
58 if(joints < 1 || joints > 8 || plugin.cartesian_count() < 2 ||
59 plugin.cartesian_count() > 3) {
60 return rt::ErrorCode::invalid_argument;
61 }
62
63 detail::Lcg rng{};
64 double q[8] = {};
65 double solved[8] = {};
66
67 // Round trip: inverse(forward(q), seed=q) must reproduce q.
68 for(int i = 0; i < iterations; ++i) {
69 for(std::size_t j = 0; j < joints; ++j) {
70 q[j] = rng.range(joint_ranges[j].minimum, joint_ranges[j].maximum);
71 }
72 geom::Vec3 cartesian{};
73 rt::ErrorCode result = plugin.forward(q, joints, cartesian);
74 if(result != rt::ErrorCode::ok) {
75 return result;
76 }
77 result = plugin.inverse(cartesian, q, joints, solved);
78 if(result != rt::ErrorCode::ok) {
79 return result;
80 }
81 for(std::size_t j = 0; j < joints; ++j) {
82 const double error = std::fabs(solved[j] - q[j]);
83 if(error > report.worst_round_trip_error) {
84 report.worst_round_trip_error = error;
85 }
86 if(error > tolerance) {
87 return rt::ErrorCode::infeasible;
88 }
89 }
90 ++report.round_trips;
91 }
92
93 // Branch stability: along a continuous joint-space walk, solving each
94 // pose with the previous solution as the seed must not jump branches
95 // (bounded joint steps).
96 for(std::size_t j = 0; j < joints; ++j) {
97 q[j] = 0.5 * (joint_ranges[j].minimum + joint_ranges[j].maximum);
98 }
99 double previous[8] = {};
100 for(std::size_t j = 0; j < joints; ++j) {
101 previous[j] = q[j];
102 }
103 const int walk = iterations < 512 ? iterations : 512;
104 for(int i = 0; i < walk; ++i) {
105 for(std::size_t j = 0; j < joints; ++j) {
106 const double span = joint_ranges[j].maximum - joint_ranges[j].minimum;
107 double next = q[j] + rng.range(-0.01, 0.01) * span;
108 if(next < joint_ranges[j].minimum) {
109 next = joint_ranges[j].minimum;
110 } else if(next > joint_ranges[j].maximum) {
111 next = joint_ranges[j].maximum;
112 }
113 q[j] = next;
114 }
115 geom::Vec3 cartesian{};
116 if(plugin.forward(q, joints, cartesian) != rt::ErrorCode::ok) {
117 return rt::ErrorCode::infeasible;
118 }
119 if(plugin.inverse(cartesian, previous, joints, solved) != rt::ErrorCode::ok) {
120 return rt::ErrorCode::infeasible;
121 }
122 for(std::size_t j = 0; j < joints; ++j) {
123 const double span = joint_ranges[j].maximum - joint_ranges[j].minimum;
124 if(std::fabs(solved[j] - previous[j]) > 0.25 * span) {
125 return rt::ErrorCode::precondition_failed; // branch jump
126 }
127 previous[j] = solved[j];
128 }
129 ++report.branch_steps;
130 }
131
132 return rt::ErrorCode::ok;
133}
