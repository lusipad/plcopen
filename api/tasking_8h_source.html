182inline bool same_name(std::string_view lower, const char *value)
183{
184 if(value == nullptr || lower.empty()) return false;

197{
198 if(token.text.empty()) return source.size();

200 const char *data = token.text.data();

250inline std::string mask_configuration_blocks(std::string_view source)
251{
252 std::string masked(source);
253 Lexer lexer(source);
254 bool in_configuration = false;
255 std::size_t begin = source.size();
256 for(Token token = lexer.next(); token.kind != TokenKind::end_of_input;
257 token = lexer.next()) {
258 if(!in_configuration && token_is(token, "configuration")) {
259 in_configuration = true;
260 begin = token_offset(source, token);
261 } else if(in_configuration && token_is(token, "end_configuration")) {
262 const std::size_t end = token_offset(source, token) +
263 token.text.size();
264 for(std::size_t at = begin; at < end; ++at)
265 if(masked[at] != '\n' && masked[at] != '\r') masked[at] = ' ';
266 in_configuration = false;
267 }
268 }
269 if(in_configuration) {
270 for(std::size_t at = begin; at < masked.size(); ++at)
271 if(masked[at] != '\n' && masked[at] != '\r') masked[at] = ' ';
272 }
273 return masked;
274}

319inline Token source_token(std::string_view source, std::size_t offset)
320{
321 Token token;
322 token.line = 1;
323 token.column = 1;
324 const std::size_t limit = std::min(offset, source.size());
325 for(std::size_t at = 0; at < limit; ++at) {
326 if(source[at] == '\n') {
327 ++token.line;
328 token.column = 1;
329 } else {
330 ++token.column;
331 }
332 }
333 return token;
334}

430inline bool read_signed_integer(TokenCursor &cursor, std::int64_t &value)
431{
432 bool negative = false;
433 if(cursor.current().kind == TokenKind::minus ||
434 cursor.current().kind == TokenKind::plus) {
435 negative = cursor.current().kind == TokenKind::minus;
436 cursor.bump();
437 }
438 if(cursor.current().kind != TokenKind::int_literal ||
439 cursor.current().based ||
440 cursor.current().unsigned_value >
441 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max())) {
442 return false;
443 }
444 value = static_cast<std::int64_t>(cursor.current().unsigned_value);
445 if(negative) value = -value;
446 cursor.bump();
447 return true;
448}

450inline bool parse_task(TokenCursor &cursor, std::string_view source,
451 const ParseLimits &limits, ResourceInfo &resource,
452 DiagnosticSink &sink)
453{
454 const Token start = cursor.current();
455 cursor.bump();
456 TaskInfo task;
457 task.source_offset = static_cast<std::uint32_t>(token_offset(source, start));
458 bool ok = read_name(cursor, task.name, sink);
459 task.lower = lower_copy(task.name);
460 if(!task.lower.empty() && duplicate_name(task.lower, resource.tasks)) {
461 sink.add(DiagCode::sema_duplicate_identifier, start);
462 ok = false;
463 }
464 if(!expect(cursor, "(", sink)) return false;
465
466 enum : unsigned
467 {
468 interval_bit = 1U << 0,
469 phase_bit = 1U << 1,
470 event_bit = 1U << 2,
471 priority_bit = 1U << 3,
472 budget_bit = 1U << 4,
473 };
474 unsigned seen = 0;
475 std::int64_t interval_ns = 0;
476 std::int64_t phase_ns = 0;
477 bool have_interval = false;
478 bool have_phase = false;
479 bool have_event = false;
480 bool have_priority = false;
481 bool have_budget = false;
482
483 while(!cursor.at_end() && !cursor.at(")")) {
484 const Token key_token = cursor.current();
485 if(!identifier(key_token)) {
486 sink.add(DiagCode::parse_expected_token, key_token);
487 ok = false;
488 cursor.bump();
489 continue;
490 }
491 const std::string key = lower_copy(key_token.text);
492 cursor.bump();
493 if(!expect(cursor, ":=", sink)) return false;
494 unsigned bit = 0;
495 if(key == "interval") bit = interval_bit;
496 else if(key == "phase") bit = phase_bit;
497 else if(key == "event") bit = event_bit;
498 else if(key == "priority") bit = priority_bit;
499 else if(key == "budget") bit = budget_bit;
500 else {
501 sink.add(DiagCode::sema_invalid_argument, key_token);
502 ok = false;
503 }
504 if(bit != 0 && (seen & bit) != 0) {
505 sink.add(DiagCode::sema_duplicate_argument, key_token);
506 ok = false;
507 }
508 seen |= bit;
509
510 if(bit == interval_bit || bit == phase_bit) {
511 if(cursor.current().kind != TokenKind::time_literal) {
512 sink.add(DiagCode::lex_bad_time_literal, cursor.current());
513 ok = false;
514 cursor.bump();
515 } else {
516 if(bit == interval_bit) {
517 interval_ns = cursor.current().signed_value;
518 have_interval = true;
519 } else {
520 phase_ns = cursor.current().signed_value;
521 have_phase = true;
522 }
523 cursor.bump();
524 }
525 } else if(bit == event_bit) {
526 std::string event;
527 if(read_name(cursor, event, sink)) {
528 task.event = lower_copy(event);
529 have_event = true;
530 } else {
531 ok = false;
532 }
533 } else if(bit == priority_bit || bit == budget_bit) {
534 std::int64_t parsed = 0;
535 if(!read_signed_integer(cursor, parsed)) {
536 sink.add(DiagCode::sema_invalid_argument, cursor.current());
537 ok = false;
538 if(!cursor.at_end()) cursor.bump();
539 } else if(bit == priority_bit) {
540 have_priority = true;
541 if(parsed < 0 || parsed > std::numeric_limits<std::int32_t>::max())
542 task.priority = -1;
543 else
544 task.priority = static_cast<std::int32_t>(parsed);
545 } else {
546 have_budget = parsed > 0;
547 task.instruction_budget = parsed;
548 }
549 } else {
550 if(!cursor.at_end()) cursor.bump();
551 }
552
553 if(cursor.at(",")) {
554 cursor.bump();
555 if(cursor.at(")")) {
556 sink.add(DiagCode::parse_expected_token, cursor.current());
557 ok = false;
558 }
559 } else if(!cursor.at(")")) {
560 sink.add(DiagCode::parse_unexpected_token, cursor.current());
561 ok = false;
562 while(!cursor.at_end() && !cursor.at(",") && !cursor.at(")"))
563 cursor.bump();
564 if(cursor.at(",")) cursor.bump();
565 }
566 }
567 if(!expect(cursor, ")", sink) || !expect(cursor, ";", sink)) return false;
568
569 if(have_interval == have_event ||
570 (have_interval &&
571 (interval_ns <= 0 || limits.base_tick_ns == 0 ||
572 static_cast<std::uint64_t>(interval_ns) % limits.base_tick_ns != 0))) {
573 sink.add(DiagCode::sema_task_interval_invalid, start);
574 ok = false;
575 }
576 if(have_interval) {
577 task.kind = TaskKind::periodic;
578 if(interval_ns > 0 && limits.base_tick_ns != 0)
579 task.interval_ticks =
580 static_cast<std::uint64_t>(interval_ns) / limits.base_tick_ns;
581 if(!have_phase) phase_ns = 0;
582 if(phase_ns < 0 || interval_ns <= 0 || phase_ns >= interval_ns ||
583 limits.base_tick_ns == 0 ||
584 static_cast<std::uint64_t>(phase_ns) % limits.base_tick_ns != 0) {
585 sink.add(DiagCode::sema_task_phase_invalid, start);
586 ok = false;
587 } else {
588 task.phase_ticks =
589 static_cast<std::uint64_t>(phase_ns) / limits.base_tick_ns;
590 }
591 } else if(have_event) {
592 task.kind = TaskKind::event;
593 if(have_phase) {
594 sink.add(DiagCode::sema_task_phase_invalid, start);
595 ok = false;
596 }
597 }
598 if(!have_priority || task.priority < 0) {
599 sink.add(DiagCode::sema_task_priority_invalid, start);
600 ok = false;
601 }
602 if(!have_budget) {
603 sink.add(DiagCode::sema_range_violation, start);
604 ok = false;
605 }
606 if(resource.tasks.size() >= limits.max_tasks) {
607 sink.add(DiagCode::capacity_tasks, start);
608 return false;
609 }
610 task.declaration_order =
611 static_cast<std::uint16_t>(resource.tasks.size());
612 resource.tasks.push_back(std::move(task));
613 return ok;
614}

616inline bool parse_mapping(TokenCursor &cursor, std::string_view source,
617 const ParseLimits &limits, ResourceInfo &resource,
618 DiagnosticSink &sink)
619{
620 const Token start = cursor.current();
621 cursor.bump();
622 ProgramMappingInfo mapping;
623 mapping.source_offset =
624 static_cast<std::uint32_t>(token_offset(source, start));
625 bool ok = read_name(cursor, mapping.name, sink);
626 mapping.lower = lower_copy(mapping.name);
627 if(!mapping.lower.empty() &&
628 duplicate_name(mapping.lower, resource.mappings)) {
629 sink.add(DiagCode::sema_program_duplicate_mapping, start);
630 ok = false;
631 }
632 if(!expect(cursor, "with", sink)) return false;
633 std::string task;
634 ok = read_name(cursor, task, sink) && ok;
635 mapping.task = lower_copy(task);
636 if(!expect(cursor, ":", sink)) return false;
637 std::string program;
638 ok = read_name(cursor, program, sink) && ok;
639 mapping.program = lower_copy(program);
640 if(!expect(cursor, ";", sink)) return false;
641
642 const std::size_t maximum =
643 static_cast<std::size_t>(limits.max_tasks) *
644 limits.max_program_mappings;
645 if(resource.mappings.size() >= maximum ||
646 resource.mappings.size() >=
647 std::numeric_limits<std::uint16_t>::max()) {
648 sink.add(DiagCode::capacity_program_mappings, start);
649 return false;
650 }
651 const int task_index = find_task(resource, mapping.task);
652 if(task_index < 0) {
653 sink.add(DiagCode::sema_unknown_identifier, start);
654 ok = false;
655 } else if(resource.tasks[static_cast<std::size_t>(task_index)]
656 .mappings.size() >= limits.max_program_mappings) {
657 sink.add(DiagCode::capacity_program_mappings, start);
658 return false;
659 }
660 mapping.declaration_order =
661 static_cast<std::uint16_t>(resource.mappings.size());
662 resource.mappings.push_back(std::move(mapping));
663 if(task_index >= 0) {
664 resource.tasks[static_cast<std::size_t>(task_index)]
665 .mappings.push_back(static_cast<std::uint16_t>(
666 resource.mappings.size() - 1));
667 }
668 return ok;
669}

671inline bool parse_resource(TokenCursor &cursor, std::string_view source,
672 const ParseLimits &limits,
673 ConfigurationInfo &configuration,
674 DiagnosticSink &sink)
675{
676 const Token start = cursor.current();
677 cursor.bump();
678 ResourceInfo resource;
679 resource.source_offset =
680 static_cast<std::uint32_t>(token_offset(source, start));
681 bool ok = read_name(cursor, resource.name, sink);
682 resource.lower = lower_copy(resource.name);
683 if(!resource.lower.empty() &&
684 duplicate_name(resource.lower, configuration.resources)) {
685 sink.add(DiagCode::sema_duplicate_identifier, start);
686 ok = false;
687 }
688 if(!expect(cursor, "on", sink)) return false;
689 ok = read_name(cursor, resource.target, sink) && ok;
690
691 while(!cursor.at_end() && !cursor.at("end_resource")) {
692 if(cursor.at("task")) {
693 ok = parse_task(cursor, source, limits, resource, sink) && ok;
694 } else if(cursor.current().kind == TokenKind::kw_program) {
695 ok = parse_mapping(cursor, source, limits, resource, sink) && ok;
696 } else {
697 sink.add(cursor.current().kind == TokenKind::error
698 ? static_cast<DiagCode>(cursor.current().diag_payload)
699 : DiagCode::parse_unexpected_token,
700 cursor.current());
701 ok = false;
702 cursor.bump();
703 }
704 }
705 if(!expect(cursor, "end_resource", sink)) return false;
706 if(configuration.resources.size() >= limits.max_resources) {
707 sink.add(DiagCode::capacity_resources, start);
708 return false;
709 }
710 configuration.resources.push_back(std::move(resource));
711 return ok;
712}

714inline bool parse_configurations(std::string_view source,
715 const ParseLimits &limits,
716 std::vector<ConfigurationInfo> &result,
717 std::vector<Diagnostic> &diagnostics)
718{
719 TokenCursor cursor(source);
720 DiagnosticSink sink(diagnostics, limits.max_diagnostics);
721 bool ok = true;
722 while(!cursor.at_end()) {
723 if(!cursor.at("configuration")) {
724 cursor.bump();
725 continue;
726 }
727 const Token start = cursor.current();
728 cursor.bump();
729 ConfigurationInfo configuration;
730 configuration.base_tick_ns = limits.base_tick_ns;
731 configuration.source_offset =
732 static_cast<std::uint32_t>(token_offset(source, start));
733 ok = read_name(cursor, configuration.name, sink) && ok;
734 configuration.lower = lower_copy(configuration.name);
735 for(const ConfigurationInfo &prior : result) {
736 if(prior.lower == configuration.lower) {
737 sink.add(DiagCode::sema_duplicate_identifier, start);
738 ok = false;
739 break;
740 }
741 }
742 while(!cursor.at_end() && !cursor.at("end_configuration")) {
743 if(cursor.at("resource")) {
744 ok = parse_resource(cursor, source, limits, configuration,
745 sink) && ok;
746 } else {
747 sink.add(cursor.current().kind == TokenKind::error
748 ? static_cast<DiagCode>(
749 cursor.current().diag_payload)
750 : DiagCode::parse_unexpected_token,
751 cursor.current());

753 cursor.bump();
754 }
755 }

757 if(result.size() >= limits.max_configurations) {
758 sink.add(DiagCode::capacity_exceeded, start);
759 ok = false;
760 continue;
761 }
762 result.push_back(std::move(configuration));
763 }
764 return ok;
765}

767inline std::uint64_t saturated_add(std::uint64_t left,
768 std::uint64_t right) noexcept
769{
770 return right > std::numeric_limits<std::uint64_t>::max() - left
771 ? std::numeric_limits<std::uint64_t>::max()
772 : left + right;
773}
774
775inline std::uint64_t align_runtime(std::uint64_t value) noexcept
776{
777 return value > std::numeric_limits<std::uint64_t>::max() - 7U
778 ? std::numeric_limits<std::uint64_t>::max()
779 : (value + 7U) & ~std::uint64_t{7U};
780}
