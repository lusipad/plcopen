14// L0 code generation (approved st-l0-semantics 2.4/3.1/3.3): deterministic
15// AST-to-bytecode emission -- the byte stream depends only on the AST walk
16// order and a value-keyed constant pool, never on addresses or iteration
17// order of unordered containers. Evaluation-stack depth is computed here at
18// compile time and capped; hidden temporaries (FOR bounds, CASE selectors)
19// use scoped slots above the declared variables. Constant subtrees folded by
20// sema are emitted as single pushes, so runtime and fold semantics agree by
21// construction. Load domain only.

128 {

136 std::size_t instruction_at(std::uint32_t byte_offset) const
137 {
138 const auto found = std::lower_bound(instruction_offsets_.begin(),
139 instruction_offsets_.end(),
140 byte_offset);

154 void analyze_worst_case(Program &program) const
155 {
156 program.worst_case_bounded = true;
157 program.worst_case_instructions = 0;
158 const std::size_t count = instruction_offsets_.size();
159 if(count == 0 || instruction_costs_.size() != count) return;
160
161 for(std::size_t index = 0; index < count; ++index) {
162 const Op op = static_cast<Op>(code_[instruction_offsets_[index]]);
163 if(op != Op::jmp && op != Op::jmp_if_false) continue;
164 const std::uint32_t target = read_u32(
165 static_cast<std::size_t>(instruction_offsets_[index]) + 1U);
166 if(instruction_at(target) == count ||
167 target <= instruction_offsets_[index]) {
168 program.worst_case_bounded = false;
169 return;
170 }
171 }
172
173 std::vector<std::uint64_t> longest(count, 0);
174 for(std::size_t reverse = count; reverse-- > 0;) {
175 const Op op = static_cast<Op>(code_[instruction_offsets_[reverse]]);
176 std::uint64_t tail = 0;
177 if(op == Op::jmp || op == Op::jmp_if_false) {
178 const std::uint32_t target = read_u32(
179 static_cast<std::size_t>(instruction_offsets_[reverse]) + 1U);
180 const std::size_t target_index = instruction_at(target);
181 if(target_index == count) {
182 program.worst_case_bounded = false;
183 program.worst_case_instructions = 0;
184 return;
185 }
186 tail = longest[target_index];
187 if(op == Op::jmp_if_false && reverse + 1U < count) {
188 tail = std::max(tail, longest[reverse + 1U]);
189 }
190 } else if(op != Op::halt && reverse + 1U < count) {
191 tail = longest[reverse + 1U];
192 }
193 longest[reverse] = add_cost(instruction_costs_[reverse], tail);
194 }
195 program.worst_case_instructions = longest[0];
196 }

277 std::uint32_t add_string_constant(const ExprInfo &note)
278 {
279 if(note.object_bytes.size() < 4U) {
280 fail(DiagCode::capacity_code);
281 return 0;
282 }
283 const TypeDesc *desc = sema_.types.get(note.type_id);
284 if(desc == nullptr ||
285 (desc->kind != TypeKind::string &&
286 desc->kind != TypeKind::wstring)) {
287 fail(DiagCode::capacity_code);
288 return 0;
289 }
290 const std::uint32_t length =
291 static_cast<std::uint32_t>(note.object_bytes[0]) |
292 (static_cast<std::uint32_t>(note.object_bytes[1]) << 8U) |
293 (static_cast<std::uint32_t>(note.object_bytes[2]) << 16U) |
294 (static_cast<std::uint32_t>(note.object_bytes[3]) << 24U);
295 const std::size_t object_size = 4U +
296 static_cast<std::size_t>(length) *
297 (desc->kind == TypeKind::wstring ? 4U : 1U);
298 if(object_size > note.object_bytes.size() ||
299 object_size >
300 std::numeric_limits<std::uint32_t>::max() -
301 string_constants_.size()) {
302 fail(DiagCode::capacity_code);
303 return 0;
304 }
305 const std::uint32_t offset =
306 static_cast<std::uint32_t>(string_constants_.size());
307 string_constants_.insert(string_constants_.end(),
308 note.object_bytes.begin(),
309 note.object_bytes.begin() +
310 static_cast<std::ptrdiff_t>(object_size));
311 return offset;
312 }

339 void emit_expr(ExprIndex index)
340 {
341 if(failed_ || index == kNoExpr) {
342 return;
343 }
344 const Expr &expr = ast_.exprs[static_cast<std::size_t>(index)];
345 const ExprInfo &note = info(index);
346 if(note.is_const) {
347 push_const(note.bits);
348 return;
349 }
350 switch(expr.kind) {
351 case ExprKind::variable:
352 if(note.string_index != kNoExpr) {
353 emit_expr(note.string_index);
354 emit_op(Op::string_index);
355 emit_u32(note.offset);
356 emit_u32(note.storage_type_id);
357 const TypeDesc *desc = sema_.types.get(note.storage_type_id);
358 if(desc != nullptr) {
359 set_last_cost(static_cast<std::uint32_t>(
360 desc->string.capacity));
361 }
362 return;
363 }
364 emit_access(note, false);
365 return;
366 case ExprKind::pin_read:
367 if(note.memory_access) {
368 emit_access(note, false);
369 } else {
370 emit_op(Op::fb_load_out);
371 emit_u16(note.fb_index);
372 emit_u8(note.pin_id);
373 push();
374 }
375 return;
376 case ExprKind::unary:
377 emit_expr(expr.lhs);
378 if(expr.unary_op == UnaryOp::logical_not) {
379 if(is_bitstring(note.type)) {
380 emit_op(Op::bit_not);
381 emit_u8(static_cast<std::uint8_t>(note.type));
382 } else {
383 emit_op(Op::not_bool);
384 }
385 } else {
386 switch(note.type) {
387 case Type::int_: emit_op(Op::neg_int); break;
388 case Type::dint: emit_op(Op::neg_dint); break;
389 case Type::real: emit_op(Op::neg_real); break;
390 case Type::lreal: emit_op(Op::neg_lreal); break;
391 default:
392 emit_op(Op::iarith);
393 emit_u8(5); // neg
394 emit_u8(static_cast<std::uint8_t>(note.type));
395 break;
396 }
397 }
398 return;
399 case ExprKind::binary: {
400 if(info(expr.lhs).type == Type::string_ ||
401 info(expr.lhs).type == Type::wstring ||
402 info(expr.rhs).type == Type::string_ ||
403 info(expr.rhs).type == Type::wstring) {
404 emit_op(Op::string_compare);
405 emit_u8(static_cast<std::uint8_t>(expr.binary_op));
406 emit_string_operand(info(expr.lhs));
407 emit_string_operand(info(expr.rhs));
408 const ExprInfo &left = info(expr.lhs);
409 const ExprInfo &right = info(expr.rhs);
410 const TypeDesc *left_desc = sema_.types.get(
411 left.storage_type_id == invalid_type_id
412 ? left.type_id : left.storage_type_id);
413 const TypeDesc *right_desc = sema_.types.get(
414 right.storage_type_id == invalid_type_id
415 ? right.type_id : right.storage_type_id);
416 if(left_desc != nullptr && right_desc != nullptr) {
417 set_last_cost(static_cast<std::uint32_t>(std::max(
418 left_desc->string.capacity,
419 right_desc->string.capacity)));
420 }
421 push();
422 return;
423 }
424 emit_expr(expr.lhs);
425 emit_expr(expr.rhs);
426 if(expr.binary_op == BinaryOp::power) {
427 // integer exponents widen to double first (L1a 5.2)
428 const Type exp_type = info(expr.rhs).type;
429 if(is_integer(exp_type)) {
430 emit_op(is_unsigned_int(exp_type) ? Op::conv_u2d
431 : Op::conv_i2d);
432 }
433 emit_op(Op::power);
434 emit_u8(static_cast<std::uint8_t>(note.type));
435 } else if(note.type == Type::time &&
436 (expr.binary_op == BinaryOp::multiply ||
437 expr.binary_op == BinaryOp::divide)) {
438 const bool floaty = is_real_family(info(expr.rhs).type);
439 const std::uint8_t sub =
440 expr.binary_op == BinaryOp::multiply
441 ? (floaty ? 2 : 0)
442 : (floaty ? 3 : 1);
443 emit_op(Op::time_scale);
444 emit_u8(sub);
445 } else if(note.type == Type::date || note.type == Type::tod ||
446 note.type == Type::dt) {
447 emit_op(Op::date_arith);
448 const std::uint8_t base = note.type == Type::date ? 0
449 : note.type == Type::tod ? 2
450 : 4;
451 emit_u8(static_cast<std::uint8_t>(
452 base + (expr.binary_op == BinaryOp::subtract ? 1 : 0)));
453 } else {
454 emit_binary_op(expr.binary_op, info(expr.lhs).type);
455 }
456 pop(); // two operands popped, one result pushed
457 return;
458 }
459 case ExprKind::call: {
460 if(note.standard_function != StandardFunction::count) {
461 if(note.standard_function >= StandardFunction::len &&
462 note.standard_function <= StandardFunction::find) {
463 int scalar_count = 0;
464 for(const ExprIndex argument : expr.arguments) {
465 const ExprInfo &argument_info = info(argument);
466 if(argument_info.type == Type::string_ ||
467 argument_info.type == Type::wstring) {
468 const Expr &argument_expr =
469 ast_.exprs[static_cast<std::size_t>(argument)];
470 if(argument_expr.kind == ExprKind::call &&
471 argument_info.standard_function != StandardFunction::count)
472 emit_expr(argument);
473 } else {
474 emit_expr(argument);
475 ++scalar_count;
476 }
477 }
478 emit_op(Op::standard_string);
479 emit_u8(static_cast<std::uint8_t>(note.standard_function));
480 emit_u8(note.standard_argc);
481 const bool string_result = note.type == Type::string_ ||
482 note.type == Type::wstring;
483 emit_u32(string_result ? note.offset
484 : std::numeric_limits<std::uint32_t>::max());
485 emit_u32(string_result ? note.storage_type_id : note.type_id);
486 for(const ExprIndex argument : expr.arguments) {
487 const ExprInfo &argument_info = info(argument);
488 const bool string_argument =
489 argument_info.type == Type::string_ ||
490 argument_info.type == Type::wstring;
491 emit_u8(string_argument ? 1U : 0U);
492 if(string_argument) emit_string_operand(argument_info);
493 }
494 set_last_cost(note.standard_cost);
495 if(string_result) {
496 pop(scalar_count);
497 } else {
498 pop(scalar_count);
499 push();
500 }
501 return;
502 }
503 for(const ExprIndex argument : expr.arguments) {
504 emit_expr(argument);
505 }
506 emit_op(Op::standard_scalar);
507 emit_u8(static_cast<std::uint8_t>(note.standard_function));
508 emit_u8(note.standard_argc);
509 emit_u8(static_cast<std::uint8_t>(note.type));
510 set_last_cost(note.standard_cost);
511 if(note.standard_argc > 0) pop(note.standard_argc - 1);
512 return;
513 }
514 // Conversion function (L1a 4.x): argument then lowering ops.
515 const std::string call_name = lower_name(expr.name);
516 if(call_name == "len") {
517 emit_op(Op::string_length);
518 emit_string_operand(info(expr.lhs));
519 const TypeDesc *desc = sema_.types.get(
520 info(expr.lhs).storage_type_id == invalid_type_id
521 ? info(expr.lhs).type_id
522 : info(expr.lhs).storage_type_id);
523 if(desc != nullptr) {
524 set_last_cost(static_cast<std::uint32_t>(
525 desc->string.capacity));
526 }
527 push();
528 return;
529 }
530 emit_expr(expr.lhs);
531 if(call_name == "l2b_alias_guard") {
532 emit_op(Op::alias_guard);
533 return;
534 }
535 if(call_name == "usint_to_char" ||
536 call_name == "char_to_usint" ||
537 call_name == "wchar_to_udint") {
538 return;
539 }
540 if(call_name == "udint_to_wchar") {
541 emit_op(Op::check_unicode);
542 return;
543 }
544 ConvDesc desc;
545 if(!resolve_conversion(lower_name(expr.name), desc)) {
546 const ExprInfo &arg = info(expr.lhs);
547 if(arg.type != note.type) {
548 emit_op(Op::conv_wrap);
549 emit_u8(static_cast<std::uint8_t>(note.type));
550 }
551 if(note.type_id >= first_load_type_id) {
552 emit_op(Op::check_range);
553 emit_u32(note.type_id);
554 }
555 return;
556 }
557 emit_conversion(desc);
558 return;
559 }
560 default:
561 // Constant literals always fold; reaching here is a compiler
562 // defect surfaced as an explicit failure, never silent output.
563 fail(DiagCode::capacity_code);
564 return;
565 }
566 }

618 void emit_conversion(const ConvDesc &desc)
619 {
620 switch(desc.kind) {
621 case ConvKind::identity:
622 case ConvKind::float_widen:
623 return; // canonical form already matches
624 case ConvKind::wrap:
625 emit_op(Op::conv_wrap);
626 emit_u8(static_cast<std::uint8_t>(desc.to));
627 return;
628 case ConvKind::int_to_float:
629 case ConvKind::uint_to_float:
630 emit_op(desc.kind == ConvKind::int_to_float ? Op::conv_i2d
631 : Op::conv_u2d);
632 if(desc.to == Type::real) {
633 emit_op(Op::f_narrow);
634 }
635 return;
636 case ConvKind::float_round:
637 emit_op(desc.trunc ? Op::conv_trunc : Op::conv_round);
638 emit_u8(static_cast<std::uint8_t>(desc.to));
639 return;
640 case ConvKind::float_narrow:
641 emit_op(Op::f_narrow);
642 return;
643 case ConvKind::to_bool_int:
644 emit_op(Op::conv_to_bool);
645 return;
646 case ConvKind::to_bool_float:
647 emit_op(Op::conv_f_to_bool);
648 return;
649 default:
650 fail(DiagCode::capacity_code);
651 return;
652 }

672 void emit_binary_op(BinaryOp op, Type operand)
673 {
674 const bool floaty = operand == Type::real || operand == Type::lreal;
675 switch(op) {
676 case BinaryOp::add:
677 if(!legacy_arith_type(operand)) {
678 emit_iarith(0, operand);
679 return;
680 }
681 switch(operand) {
682 case Type::int_: emit_op(Op::add_int); return;
683 case Type::dint: emit_op(Op::add_dint); return;
684 case Type::real: emit_op(Op::add_real); return;
685 case Type::lreal: emit_op(Op::add_lreal); return;
686 default: emit_op(Op::add_time); return;
687 }
688 case BinaryOp::subtract:
689 if(!legacy_arith_type(operand)) {
690 emit_iarith(1, operand);
691 return;
692 }
693 switch(operand) {
694 case Type::int_: emit_op(Op::sub_int); return;
695 case Type::dint: emit_op(Op::sub_dint); return;
696 case Type::real: emit_op(Op::sub_real); return;
697 case Type::lreal: emit_op(Op::sub_lreal); return;
698 default: emit_op(Op::sub_time); return;
699 }
700 case BinaryOp::multiply:
701 if(!legacy_arith_type(operand)) {
702 emit_iarith(2, operand);
703 return;
704 }
705 switch(operand) {
706 case Type::int_: emit_op(Op::mul_int); return;
707 case Type::dint: emit_op(Op::mul_dint); return;
708 case Type::real: emit_op(Op::mul_real); return;
709 default: emit_op(Op::mul_lreal); return;
710 }
711 case BinaryOp::divide:
712 if(!legacy_arith_type(operand)) {
713 emit_iarith(3, operand);
714 return;
715 }
716 switch(operand) {
717 case Type::int_: emit_op(Op::div_int); return;
718 case Type::dint: emit_op(Op::div_dint); return;
719 case Type::real: emit_op(Op::div_real); return;
720 default: emit_op(Op::div_lreal); return;
721 }
722 case BinaryOp::modulo:
723 if(operand != Type::int_ && operand != Type::dint) {
724 emit_iarith(4, operand);
725 return;
726 }
727 emit_op(operand == Type::int_ ? Op::mod_int : Op::mod_dint);
728 return;
729 case BinaryOp::logical_and:
730 if(is_bitstring(operand)) {
731 emit_op(Op::bit_and);
732 return;
733 }
734 emit_op(Op::and_bool);
735 return;
736 case BinaryOp::logical_or:
737 if(is_bitstring(operand)) {
738 emit_op(Op::bit_or);
739 return;
740 }
741 emit_op(Op::or_bool);
742 return;
743 case BinaryOp::logical_xor:
744 if(is_bitstring(operand)) {
745 emit_op(Op::bit_xor);
746 return;
747 }
748 emit_op(Op::xor_bool);
749 return;
750 case BinaryOp::cmp_eq:
751 emit_op(floaty ? Op::cmp_eq_f : Op::cmp_eq_i);
752 return;
753 case BinaryOp::cmp_ne:
754 emit_op(floaty ? Op::cmp_ne_f : Op::cmp_ne_i);
755 return;
756 case BinaryOp::cmp_lt:
757 if(is_unsigned_int(operand)) {
758 emit_op(Op::cmp_u);
759 emit_u8(0);
760 return;
761 }
762 emit_op(floaty ? Op::cmp_lt_f : Op::cmp_lt_i);
763 return;
764 case BinaryOp::cmp_gt:
765 if(is_unsigned_int(operand)) {
766 emit_op(Op::cmp_u);
767 emit_u8(1);
768 return;
769 }
770 emit_op(floaty ? Op::cmp_gt_f : Op::cmp_gt_i);
771 return;
772 case BinaryOp::cmp_le:
773 if(is_unsigned_int(operand)) {
774 emit_op(Op::cmp_u);
775 emit_u8(2);
776 return;
777 }
778 emit_op(floaty ? Op::cmp_le_f : Op::cmp_le_i);
779 return;
780 default:
781 if(is_unsigned_int(operand)) {
782 emit_op(Op::cmp_u);
783 emit_u8(3);
784 return;
785 }
786 emit_op(floaty ? Op::cmp_ge_f : Op::cmp_ge_i);
787 return;
788 }

939 case StmtKind::output_commit: emit_output_commit(stmt); return;

954 }
955 return;
956 case StmtKind::return_:
957 emit_op(Op::halt);
958 return;
959 case StmtKind::empty:
960 return;
961 }
