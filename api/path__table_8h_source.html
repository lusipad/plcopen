53inline rt::ErrorCode validate_path_waypoint(const PathWaypoint &waypoint,
54 std::size_t axis_count)
55{
56 if(waypoint.target.size != axis_count) return rt::ErrorCode::invalid_argument;
57 for(std::size_t axis_index = 0; axis_index < axis_count; ++axis_index) {
58 if(!std::isfinite(waypoint.target.value[axis_index])) {
59 return rt::ErrorCode::invalid_argument;
60 }
61 }
62 if(!std::isfinite(waypoint.velocity) || waypoint.velocity <= 0.0 ||
63 !std::isfinite(waypoint.acceleration) || waypoint.acceleration <= 0.0 ||
64 !std::isfinite(waypoint.deceleration) || waypoint.deceleration <= 0.0 ||
65 !std::isfinite(waypoint.jerk) || waypoint.jerk <= 0.0 ||
66 !std::isfinite(waypoint.transition_parameter)) {
67 return rt::ErrorCode::invalid_argument;
68 }
69 if(waypoint.interpolation_space != axis::InterpolationSpace::joint &&
70 waypoint.interpolation_space != axis::InterpolationSpace::cartesian) {
71 return rt::ErrorCode::invalid_argument;
72 }
73 if(waypoint.transition_mode == axis::TransitionMode::none) {
74 return waypoint.transition_parameter == 0.0
75 ? rt::ErrorCode::ok
76 : rt::ErrorCode::invalid_argument;
77 }
78 if(waypoint.transition_mode == axis::TransitionMode::max_corner_deviation) {
79 return waypoint.transition_parameter > 0.0
80 ? rt::ErrorCode::ok
81 : rt::ErrorCode::invalid_argument;
82 }
83 return rt::ErrorCode::unsupported;
84}

86// MC_PathSelect: validate a source description and publish a selected result.
87class FbPathSelect
88{
89public:
90 axis::AxisGroup *group_ref = nullptr;
91 PathTable *path_data = nullptr;
92 const PathDescription *path_description = nullptr;
93 bool execute = false;
94 MotionOutputs outputs{};
95
96 void call()
97 {
98 const bool rising = execute && !last_execute_;
99 last_execute_ = execute;
100 if(!execute) {
101 clear(outputs);
102 return;
103 }
104 if(!rising) {
105 return;
106 }
107 clear(outputs);
108 if(group_ref == nullptr || path_data == nullptr || path_description == nullptr) {
109 outputs.error = true;
110 outputs.error_id = rt::ErrorCode::invalid_argument;
111 return;
112 }
113 if(path_description->count < 2 ||
114 path_description->count > PathTable::MaxWaypoints) {
115 outputs.error = true;
116 outputs.error_id = rt::ErrorCode::invalid_argument;
117 return;
118 }
119 const std::size_t n_axes = group_ref->member_count();
120 if(n_axes < 2) {
121 outputs.error = true;
122 outputs.error_id = rt::ErrorCode::invalid_argument;
123 return;
124 }
125 for(std::size_t i = 0; i < path_description->count; ++i) {
126 const PathWaypoint &wp = path_description->waypoints[i];
127 const rt::ErrorCode valid = validate_path_waypoint(wp, n_axes);
128 if(valid != rt::ErrorCode::ok) {
129 outputs.error = true;
130 outputs.error_id = valid;
131 return;
132 }
133 }
134 PathTable selected{};
135 selected.count = path_description->count;
136 selected.axis_count = n_axes;
137 selected.handle = ++next_handle_;
138 if(selected.handle == 0) {
139 selected.handle = ++next_handle_;
140 }
141 for(std::size_t i = 0; i < selected.count; ++i) {
142 selected.waypoints[i] = path_description->waypoints[i];
143 }
144 *path_data = selected;
145 outputs.done = true;
146 }
147
148private:
149 bool last_execute_ = false;
150 std::uint32_t next_handle_ = 0;
151};

153// MC_MovePath: consume a validated path table handle and submit all
154// waypoints through the group's window machine as blending commands.
155class FbMovePath : public GroupExecuteFb
156{
157public:
158 PathTable *path_data = nullptr;
159 axis::CoordSystem coord_system = axis::CoordSystem::acs;
160 axis::BufferMode buffer_mode = axis::BufferMode::aborting;
161 axis::TransitionMode transition_mode = axis::TransitionMode::none;
162 double transition_parameter = 0.0;
163
164 void call()
165 {
166 if(rising_edge()) {
167 if(group_ref == nullptr || path_data == nullptr || path_data->handle == 0 ||
168 path_data->count < 2 ||
169 path_data->count > PathTable::MaxWaypoints ||
170 path_data->axis_count != group_ref->member_count()) {
171 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument));
172 return;
173 }
174 const rt::ErrorCode options = validate_options();
175 if(options != rt::ErrorCode::ok) {
176 accept(rt::Result<std::uint32_t>::failure(options));
177 return;
178 }
179 for(std::size_t i = 0; i < path_data->count; ++i) {
180 const rt::ErrorCode valid =
181 validate_path_waypoint(path_data->waypoints[i], path_data->axis_count);
182 if(valid != rt::ErrorCode::ok) {
183 accept(rt::Result<std::uint32_t>::failure(valid));
184 return;
185 }
186 }
187 const PathWaypoint &first = path_data->waypoints[0];
188 axis::GroupCommand cmd{};
189 cmd.target = first.target;
190 cmd.velocity = first.velocity;
191 cmd.acceleration = first.acceleration;
192 cmd.deceleration = first.deceleration;
193 cmd.jerk = first.jerk;
194 cmd.buffer_mode = buffer_mode;
195 cmd.coord_system = coord_system;
196 cmd.transition_mode = transition_mode;
197 cmd.transition_parameter = transition_parameter;
198 cmd.interpolation_space = first.interpolation_space;
199 const auto result = group_ref->submit_linear(cmd);
200 if(!result) {
201 accept(result);
202 return;
203 }
204 for(std::size_t i = 1; i < path_data->count; ++i) {
205 const PathWaypoint &wp = path_data->waypoints[i];
206 axis::GroupCommand seg{};
207 seg.target = wp.target;
208 seg.velocity = wp.velocity;
209 seg.acceleration = wp.acceleration;
210 seg.deceleration = wp.deceleration;
211 seg.jerk = wp.jerk;
212 seg.transition_mode = wp.transition_mode;
213 seg.transition_parameter = wp.transition_parameter;
214 seg.interpolation_space = wp.interpolation_space;
215 seg.coord_system = coord_system;
216 if(i < path_data->count - 1 &&
217 wp.transition_mode != axis::TransitionMode::none) {
218 seg.buffer_mode = axis::BufferMode::blending_low;
219 } else {
220 seg.buffer_mode = axis::BufferMode::buffered;
221 }
222 const auto seg_result = group_ref->submit_linear(seg);
223 if(!seg_result) {
224 accept(seg_result);
225 return;
226 }
227 }
228 accept(result);
229 }
230 observe_group();
231 }
232
233private:
234 rt::ErrorCode validate_options() const
235 {
236 switch(coord_system) {
237 case axis::CoordSystem::acs:
238 case axis::CoordSystem::mcs:
239 case axis::CoordSystem::pcs:
240 break;
241 case axis::CoordSystem::wcs:
242 case axis::CoordSystem::fcs:
243 case axis::CoordSystem::tcs:
244 return rt::ErrorCode::unsupported;
245 default:
246 return rt::ErrorCode::invalid_argument;
247 }
248 switch(buffer_mode) {
249 case axis::BufferMode::aborting:
250 case axis::BufferMode::buffered:
251 case axis::BufferMode::blending_low:
252 case axis::BufferMode::blending_high:
253 break;
254 default:
255 return rt::ErrorCode::invalid_argument;
256 }
257 if(!std::isfinite(transition_parameter)) {
258 return rt::ErrorCode::invalid_argument;
259 }
260 if(transition_mode == axis::TransitionMode::none) {
261 return transition_parameter == 0.0
262 ? rt::ErrorCode::ok
263 : rt::ErrorCode::invalid_argument;
264 }
265 if(transition_mode != axis::TransitionMode::max_corner_deviation) {
266 return rt::ErrorCode::unsupported;
267 }
268 if(transition_parameter <= 0.0 || buffer_mode == axis::BufferMode::aborting) {
269 return rt::ErrorCode::invalid_argument;
270 }
271 if(buffer_mode == axis::BufferMode::buffered) {
272 return rt::ErrorCode::unsupported;
273 }
274 return rt::ErrorCode::ok;
275 }

359 outputs.command_aborted = true;
360 }

362 outputs.active = false;
363 in_sync = false;
364 tracked_command_id_ = 0;
