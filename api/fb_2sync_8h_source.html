15// MC_GearIn / MC_GearInPos / MC_CamIn require master and slave to be members
16// of the same enabled group (KB boundary carried over from the v0.x tests).
17inline bool same_enabled_group(const axis::AxisModel &master, const axis::AxisModel &slave)
18{
19 void *owner = slave.group_owner();
20 if(owner == nullptr || owner != master.group_owner()) {
21 return false;
22 }
23 const auto *group = static_cast<const axis::AxisGroup *>(owner);
24 return group->status() != axis::GroupStatus::disabled;
25}

27// Shared facade state for master/slave synchronization commands. in_sync maps
28// to the v0.x InGear/InSync outputs; start_sync pulses one cycle at approach
29// start and at sync entry.
30class SyncExecuteFb
31{
32public:
33 axis::AxisModel *master_ref = nullptr;
34 axis::AxisModel *slave_ref = nullptr;
35 bool execute = false;
36 bool continuous_update = false;
37 MotionOutputs outputs{};
38 bool in_sync = false;
39 bool start_sync = false;
40
41protected:
42 bool rising_edge()
43 {
44 const bool rising = execute && !last_execute_;
45 const bool falling = !execute && last_execute_;
46 last_execute_ = execute;
47 if(rising) {
48 clear(outputs);
49 in_sync = false;
50 start_sync = false;
51 tracked_command_id_ = 0;
52 last_phase_ = axis::SyncPhase::idle;
53 continuous_update_enabled_ = continuous_update;
54 terminal_low_cycle_ = false;
55 } else if(!execute && terminal_low_cycle_) {
56 clear(outputs);
57 tracked_command_id_ = 0;
58 terminal_low_cycle_ = false;
59 } else if(falling &&
60 (outputs.done || outputs.command_aborted || outputs.error)) {
61 clear(outputs);
62 tracked_command_id_ = 0;
63 }
64 return rising;
65 }
66
67 void accept(rt::Result<std::uint32_t> accepted)
68 {
69 outputs.command_accepted = false;
70 outputs.command_aborted = false;
71 outputs.error = false;
72 outputs.error_id = rt::ErrorCode::ok;
73 in_sync = false;
74 start_sync = false;
75 last_phase_ = axis::SyncPhase::idle;
76 if(!accepted) {
77 outputs.error = true;
78 outputs.error_id = accepted.error();
79 outputs.busy = false;
80 outputs.active = false;
81 tracked_command_id_ = 0;
82 return;
83 }
84 tracked_command_id_ = accepted.value();
85 outputs.command_id = tracked_command_id_;
86 outputs.command_accepted = true;
87 outputs.busy = true;
88 outputs.active = true;
89 }
90
91 void observe_sync(bool pulse_approach = true)
92 {
93 start_sync = false;
94 if(tracked_command_id_ == 0 || slave_ref == nullptr) {
95 return;
96 }
97 const rt::ErrorCode command_error = slave_ref->command_error(tracked_command_id_);
98 if(command_error != rt::ErrorCode::ok) {
99 outputs.error = true;
100 outputs.error_id = command_error;
101 outputs.command_aborted = false;
102 outputs.busy = false;
103 outputs.active = false;
104 in_sync = false;
105 terminal_low_cycle_ = !execute;
106 return;
107 }
108 if(slave_ref->sync_command_id() != tracked_command_id_) {
109 outputs.command_aborted = true;
110 outputs.busy = false;
111 outputs.active = false;
112 in_sync = false;
113 terminal_low_cycle_ = !execute;
114 return;
115 }
116
117 const axis::SyncPhase phase = slave_ref->sync_phase();
118 outputs.busy = true;
119 outputs.active = phase != axis::SyncPhase::queued;
120 in_sync = phase == axis::SyncPhase::engaged;
121 start_sync = phase != last_phase_ &&
122 (phase == axis::SyncPhase::engaged ||
123 (pulse_approach && phase == axis::SyncPhase::approaching));
124 last_phase_ = phase;
125 }
126
127 std::uint32_t tracked_command_id_ = 0;
128
129 bool continuous_update_allowed() const
130 {
131 return continuous_update_enabled_;
132 }
133
134private:
135 bool last_execute_ = false;
136 bool continuous_update_enabled_ = false;
137 bool terminal_low_cycle_ = false;
138 axis::SyncPhase last_phase_ = axis::SyncPhase::idle;
139};

416 const double effective_slave_offset =
417 slave_offset + selected_slave_origin_ + relative_slave_bias_;

434private:
435 void submit()
436 {
437 if(master_ref == nullptr || slave_ref == nullptr) {
438 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument));
439 return;
440 }
441 if(!same_enabled_group(*master_ref, *slave_ref)) {
442 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::precondition_failed));
443 return;
444 }
445 if(start_mode == axis::CamStartMode::ramp_in) {
446 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::unsupported));
447 return;
448 }
449 axis::CamInCommand command{};
450 command.master = master_ref;
451 command.table = cam_table;
452 command.master_offset = master_offset;
453 command.master_scaling = master_scaling;
454 command.slave_offset = slave_offset;
455 command.slave_scaling = slave_scaling;
456 selected_master_origin_ = 0.0;
457 selected_slave_origin_ = 0.0;
458 relative_slave_bias_ = 0.0;
459 if(cam_table_id != 0) {
460 const rt::Result<axis::CamTableSelection> selected =
461 slave_ref->cam_table_selection(cam_table_id);
462 if(!selected || selected.value().master != master_ref) {
463 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument));
464 return;
465 }
466 const axis::CamTableSelection selection = selected.value();
467 command.table = selection.table;
468 if(!selection.master_absolute) {
469 selected_master_origin_ = selection.master_origin;
470 command.master_offset += selected_master_origin_;
471 }
472 if(!selection.slave_absolute) {
473 selected_slave_origin_ = selection.slave_origin;
474 command.slave_offset += selected_slave_origin_;
475 }
476 }
477 if(start_mode == axis::CamStartMode::relative) {
478 if(!command.table.valid() || !std::isfinite(command.master_scaling) ||
479 command.master_scaling == 0.0 || !std::isfinite(command.slave_scaling)) {
480 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument));
481 return;
482 }
483 const axis::AxisSnapshot &master_snapshot = master_ref->snapshot();
484 const double master_position =
485 master_value_source == axis::MasterValueSource::actual
486 ? master_snapshot.actual_position
487 : master_snapshot.command_position;
488 const double table_input =
489 (master_position - command.master_offset) / command.master_scaling;
490 const rt::Result<double> sampled = command.table.sample(table_input);
491 if(!sampled) {
492 accept(rt::Result<std::uint32_t>::failure(sampled.error()));
493 return;
494 }
495 command.slave_offset = slave_ref->snapshot().command_position -
496 command.slave_scaling * sampled.value();
497 relative_slave_bias_ =
498 command.slave_offset - slave_offset - selected_slave_origin_;
499 }
500 command.source = master_value_source;
501 command.buffer_mode = buffer_mode;
502 command.master_sync_position = master_sync_position;
503 command.master_start_distance = master_start_distance;
504 command.approach_velocity = velocity;
505 const rt::Result<std::uint32_t> accepted = slave_ref->cam_in(command);
506 if(accepted) {
507 profile_table_ = command.table;
508 profile_master_offset_ = command.master_offset;
509 profile_master_scaling_ = command.master_scaling;
510 profile_source_ = command.source;
511 }
512 accept(accepted);
513 }

515 void reset_profile_tracking()
516 {
517 end_of_profile = false;
518 profile_period_valid_ = false;

522 void update_end_of_profile()
523 {
524 end_of_profile = false;
525 if(!in_sync || master_ref == nullptr || !profile_table_.valid() ||
526 profile_master_scaling_ == 0.0) {
527 profile_period_valid_ = false;
528 return;
529 }
530 const axis::AxisSnapshot &snapshot = master_ref->snapshot();
531 const double master_position = profile_source_ == axis::MasterValueSource::actual
532 ? snapshot.actual_position
533 : snapshot.command_position;
534 const double input =
535 (master_position - profile_master_offset_) / profile_master_scaling_;
536 const double first = profile_table_.points[0].master;
537 const double last = profile_table_.points[profile_table_.size - 1].master;
538 if(!profile_table_.periodic) {
539 end_of_profile = input < first || input > last;
540 return;
541 }
542 const double span = last - first;
543 const std::int64_t period = static_cast<std::int64_t>(std::floor((input - first) / span));
544 if(profile_period_valid_) {
545 end_of_profile = period != profile_period_;
546 }
547 profile_period_ = period;
548 profile_period_valid_ = true;
549 }
550
551 exec::CamTableView profile_table_{};
552 double profile_master_offset_ = 0.0;
553 double profile_master_scaling_ = 1.0;
554 axis::MasterValueSource profile_source_ = axis::MasterValueSource::command;
555 double selected_master_origin_ = 0.0;
556 double selected_slave_origin_ = 0.0;
557 double relative_slave_bias_ = 0.0;
558 std::int64_t profile_period_ = 0;
559 bool profile_period_valid_ = false;

617// Phase inputs are latched on the rising edge; changing them while the
618// transition runs has no effect (v0.x latching boundary).
619class PhasingFb
620{
621public:
622 axis::AxisModel *master_ref = nullptr;
623 axis::AxisModel *slave_ref = nullptr;
624 double phase_shift = 0.0;
625 double velocity = 0.0;
626 double acceleration = 0.0;
627 double deceleration = 0.0;
628 double jerk = 0.0;
629 axis::BufferMode buffer_mode = axis::BufferMode::aborting;
630 bool execute = false;
631 MotionOutputs outputs{};
632 double absolute_phase_shift = 0.0;
633 double covered_phase_shift = 0.0;
634
635protected:
636 explicit PhasingFb(bool relative)
637 : relative_(relative)
638 {
639 }
640
641 void step()
642 {
643 const bool rising = execute && !last_execute_;
644 const bool falling = !execute && last_execute_;
645 last_execute_ = execute;
646 if(rising) {
647 clear(outputs);
648 started_ = false;
649 tracked_command_id_ = 0;
650 absolute_phase_shift = 0.0;
651 covered_phase_shift = 0.0;
652 terminal_low_cycle_ = false;
653 } else if(!execute && terminal_low_cycle_) {
654 clear(outputs);
655 started_ = false;
656 tracked_command_id_ = 0;
657 terminal_low_cycle_ = false;
658 return;
659 } else if(falling &&
660 (outputs.done || outputs.command_aborted || outputs.error)) {
661 clear(outputs);
662 started_ = false;
663 tracked_command_id_ = 0;
664 }
665 if(rising) {
666 start();
667 return;
668 }
669 if(started_ && slave_ref != nullptr) {
670 absolute_phase_shift = slave_ref->gear_phase_offset();
671 covered_phase_shift = slave_ref->phasing_covered_shift(tracked_command_id_);
672 const axis::PhasingCommandState state =
673 slave_ref->phasing_command_state(tracked_command_id_);
674 outputs.busy = state == axis::PhasingCommandState::queued ||
675 state == axis::PhasingCommandState::active;
676 outputs.active = state == axis::PhasingCommandState::active;
677 if(state == axis::PhasingCommandState::completed) {
678 outputs.done = true;
679 outputs.busy = false;
680 outputs.active = false;
681 terminal_low_cycle_ = !execute;
682 } else if(state == axis::PhasingCommandState::aborted) {
683 outputs.command_aborted = true;
684 outputs.busy = false;
685 outputs.active = false;
686 terminal_low_cycle_ = !execute;
687 }
688 }
689 }
690
691private:
692 void start()
693 {
694 clear(outputs);
695 started_ = false;
696 if(master_ref == nullptr || slave_ref == nullptr || master_ref == slave_ref ||
697 !slave_ref->gear_engaged_with(master_ref)) {
698 outputs.error = true;
699 outputs.error_id = rt::ErrorCode::invalid_argument;
700 return;
701 }
702 axis::PhasingCommand command{};
703 command.phase_shift = phase_shift;
704 command.velocity = velocity;
705 command.acceleration = acceleration;
706 command.deceleration = deceleration;
707 command.jerk = jerk;
708 command.buffer_mode = buffer_mode;
709 command.relative = relative_;
710 const rt::Result<std::uint32_t> requested = slave_ref->submit_phasing(command);
711 if(!requested) {
712 outputs.error = true;
713 outputs.error_id = requested.error();
714 return;
715 }
716 started_ = true;
717 tracked_command_id_ = requested.value();
718 outputs.command_id = tracked_command_id_;
719 outputs.command_accepted = true;
720 absolute_phase_shift = slave_ref->gear_phase_offset();
721 covered_phase_shift = slave_ref->phasing_covered_shift(tracked_command_id_);
722 const axis::PhasingCommandState state =
723 slave_ref->phasing_command_state(tracked_command_id_);
724 outputs.busy = state == axis::PhasingCommandState::queued ||
725 state == axis::PhasingCommandState::active;
726 outputs.active = state == axis::PhasingCommandState::active;
727 outputs.done = state == axis::PhasingCommandState::completed;
728 }
729
730 bool relative_ = false;
731 bool last_execute_ = false;
732 bool started_ = false;
733 bool terminal_low_cycle_ = false;
734 std::uint32_t tracked_command_id_ = 0;
735};
