230// MC_GroupSetOverride: enable-based cyclic group override.
231class FbGroupSetOverride
232{
233public:
234 axis::AxisGroup *group_ref = nullptr;
235 bool enable = false;
236 double vel_factor = 1.0;
237 double acc_factor = 1.0;
238 double jerk_factor = 1.0;
239 bool enabled = false;
240 bool busy = false;
241 bool error = false;
242 rt::ErrorCode error_id = rt::ErrorCode::ok;
243
244 void call()
245 {
246 if(!enable) {
247 enabled = false;
248 busy = false;
249 error = false;
250 error_id = rt::ErrorCode::ok;
251 return;
252 }
253 busy = true;
254 enabled = false;
255 error = false;
256 error_id = rt::ErrorCode::ok;
257 if(group_ref == nullptr) {
258 busy = false;
259 error = true;
260 error_id = rt::ErrorCode::invalid_argument;
261 return;
262 }
263 const auto clamp_factor = [](double factor) {
264 return factor < 0.0 ? 0.0 : (factor > 1.0 ? 1.0 : factor);
265 };
266 const rt::ErrorCode result = group_ref->set_group_override(
267 clamp_factor(vel_factor), clamp_factor(acc_factor),
268 clamp_factor(jerk_factor));
269 busy = false;
270 if(result != rt::ErrorCode::ok) {
271 error = true;
272 error_id = result;
273 return;
274 }
275 enabled = true;
276 }

279// MC_GroupInterrupt: controlled pause preserving motion state.
280class FbGroupInterrupt : public GroupExecuteFb
281{
282public:
283 double deceleration = 1.0;
284 double jerk = 1.0;
285
286 void call()
287 {
288 if(rising_edge()) {
289 if(group_ref == nullptr) {
290 accept(rt::Result<std::uint32_t>::failure(rt::ErrorCode::invalid_argument));
291 } else {
292 const rt::ErrorCode result = group_ref->interrupt(deceleration, jerk);
293 accept(result == rt::ErrorCode::ok

295 : rt::Result<std::uint32_t>::failure(result));
296 }
297 }
298 observe_interrupt();
299 }
300
301private:
302 void observe_interrupt()
303 {
304 if(!execute || tracked_command_id_ == 0 || group_ref == nullptr) {
305 return;
306 }
307 const axis::GroupStatus gs = group_ref->status();
308 if(gs == axis::GroupStatus::stopping) {
309 outputs.busy = true;
310 outputs.active = true;
311 outputs.done = false;
312 return;
313 }
314 if(gs == axis::GroupStatus::interrupted) {
315 outputs.done = true;
316 outputs.busy = false;
317 outputs.active = false;
318 return;
319 }
320 if(gs == axis::GroupStatus::errorstop) {
321 outputs.error = true;
322 outputs.error_id = rt::ErrorCode::precondition_failed;
323 outputs.busy = false;
324 outputs.active = false;
325 }
