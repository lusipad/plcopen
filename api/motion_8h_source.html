36
37protected:
38 bool rising_edge(bool retain_active_on_falling = false)
39 {
40 const bool rising = execute && !last_execute_;
41 const bool falling = !execute && last_execute_;
42 last_execute_ = execute;
43 if(rising || (!execute && terminal_low_cycle_)) {
44 clear(outputs);
45 tracked_command_id_ = 0;
46 terminal_low_cycle_ = false;
47 } else if(falling && terminal() && !retain_active_on_falling) {
48 clear(outputs);
49 tracked_command_id_ = 0;
50 }

53
54 void accept(rt::Result<std::uint32_t> accepted)
55 {
56 outputs.command_accepted = false;
57 outputs.command_aborted = false;
58 outputs.error = false;
59 outputs.error_id = rt::ErrorCode::ok;
60 if(!accepted) {
61 outputs.error = true;
62 outputs.error_id = accepted.error();
63 outputs.busy = false;
64 outputs.active = false;
65 tracked_command_id_ = 0;
66 return;
67 }
68 tracked_command_id_ = accepted.value();
69 outputs.command_id = tracked_command_id_;
70 outputs.command_accepted = true;
71 outputs.busy = true;
72 outputs.active = true;
73 outputs.done = false;
74 }

76 void observe_axis()
77 {
78 if(tracked_command_id_ == 0 || axis_ref == nullptr) {
79 return;
80 }
81 // Done holds while Execute stays high; a later takeover command must
82 // not retroactively turn a completed command into an aborted one.
83 if(outputs.done) {
84 return;
85 }
86 const axis::AxisSnapshot &snapshot = axis_ref->snapshot();
87 const rt::ErrorCode command_error = axis_ref->command_error(tracked_command_id_);
88 if(command_error != rt::ErrorCode::ok) {
89 outputs.error = true;
90 outputs.error_id = command_error;
91 outputs.command_aborted = false;
92 outputs.done = false;
93 outputs.busy = false;
94 outputs.active = false;
95 terminal_observed();
96 return;
97 }
98 if(snapshot.active_command_id == tracked_command_id_) {
99 outputs.busy = true;
100 outputs.active = true;
101 outputs.done = false;
102 return;
103 }
104 // A completed command reports done even when a queued or blending
105 // successor started within the same cycle.
106 if(snapshot.last_completed_command_id == tracked_command_id_) {
107 outputs.done = true;
108 outputs.busy = false;
109 outputs.active = false;
110 terminal_observed();
111 return;
112 }
113 // A command waiting in the buffered queue is busy, not aborted.
114 if(axis_ref->command_pending(tracked_command_id_)) {
115 outputs.busy = true;
116 outputs.active = false;
117 return;
118 }
119 outputs.command_aborted = true;
120 outputs.busy = false;
121 outputs.active = false;
122 terminal_observed();
123 }
124
125 void observe_axis_management()
126 {
127 if(tracked_command_id_ == 0 || axis_ref == nullptr || outputs.done || outputs.error) {
128 return;
129 }
130 const rt::ErrorCode command_error =
131 axis_ref->management_command_error(tracked_command_id_);
132 if(command_error != rt::ErrorCode::ok) {
133 outputs.error = true;
134 outputs.error_id = command_error;
135 outputs.busy = false;
136 outputs.active = false;
137 terminal_observed();
138 return;
139 }
140 if(axis_ref->management_command_done(tracked_command_id_)) {
141 outputs.done = true;
142 outputs.busy = false;
143 outputs.active = false;
144 terminal_observed();
145 return;
146 }
147 outputs.busy = axis_ref->management_command_pending(tracked_command_id_) ||
148 axis_ref->management_command_active(tracked_command_id_);
149 outputs.active = axis_ref->management_command_active(tracked_command_id_);
150 }
151
152 std::uint32_t tracked_command_id_ = 0;
153
154 void fail_runtime(rt::ErrorCode code)
155 {
156 if(axis_ref != nullptr && tracked_command_id_ != 0) {
157 axis_ref->fail_command(tracked_command_id_, code);
158 }
159 }
160
161 void terminal_observed()
162 {
163 terminal_low_cycle_ = !execute;
164 }
165
166private:
167 bool terminal() const
168 {
169 return outputs.done || outputs.command_aborted || outputs.error;
170 }
171
172 bool last_execute_ = false;
173 bool terminal_low_cycle_ = false;

451 axis_ref != nullptr &&

453 const double signed_direction = resolved_direction();
454 const rt::ErrorCode updated =
455 signed_direction == 0.0

472 {

474 !std::isfinite(direction) || direction == 0.0) {

476 }
477 double mode = direction;
478 if(direction_mode == axis::Direction::positive) {
479 mode = 1.0;

503 command.acceleration = acceleration;
504 command.deceleration = deceleration;
505 command.jerk = jerk;
506 command.buffer_mode = buffer_mode;
507 const rt::Result<std::uint32_t> accepted = axis_ref->submit(command);

510 last_direction_ = direction;
511 }
512 accept(accepted);
513 }

700class FbMoveSuperimposed
701{
702public:
703 axis::AxisModel *axis_ref = nullptr;
704 double distance = 0.0;
705 double velocity = 1.0;
706 double acceleration = 1.0;
707 double deceleration = 1.0;
708 double jerk = 1.0;
709 bool continuous_update = false;
710 bool execute = false;
711 double covered_distance = 0.0;
712 MotionOutputs outputs{};
713
714 void call()
715 {
716 const bool rising = execute && !last_execute_;
717 const bool falling = !execute && last_execute_;
718 last_execute_ = execute;
719 if(rising) {
720 clear(outputs);
721 tracked_command_id_ = 0;
722 terminal_low_cycle_ = false;
723 continuous_update_enabled_ = continuous_update;
724 last_distance_ = distance;
725 covered_distance = 0.0;
726 } else if(!execute && terminal_low_cycle_) {
727 clear(outputs);
728 tracked_command_id_ = 0;
729 terminal_low_cycle_ = false;
730 return;
731 } else if(falling &&
732 (outputs.done || outputs.command_aborted || outputs.error)) {
733 clear(outputs);
734 tracked_command_id_ = 0;
735 }
736 if(rising) {
737 submit();
738 return;
739 }
740 if(execute && continuous_update_enabled_ && tracked_command_id_ != 0 &&
741 axis_ref != nullptr && distance != last_distance_) {
742 last_distance_ = distance;
743 const rt::ErrorCode updated = axis_ref->update_superimposed_target(
744 tracked_command_id_, distance, velocity, acceleration,
745 deceleration, jerk);
746 if(updated != rt::ErrorCode::ok) {
747 axis_ref->abort_superimposed();
748 outputs.error = true;
749 outputs.error_id = updated;
750 outputs.busy = false;
751 outputs.active = false;
752 return;
753 }
754 }
755 observe();
756 if(axis_ref != nullptr && tracked_command_id_ != 0) {
757 covered_distance = axis_ref->superimposed_distance();
758 }
759 }
760
761private:
762 void submit()
763 {
764 clear(outputs);
765 if(axis_ref == nullptr) {
766 outputs.error = true;
767 outputs.error_id = rt::ErrorCode::invalid_argument;
768 tracked_command_id_ = 0;
769 return;
770 }
771 const rt::Result<std::uint32_t> accepted =
772 axis_ref->submit_superimposed(distance, velocity, acceleration, deceleration, jerk);
773 if(!accepted) {
774 outputs.error = true;
775 outputs.error_id = accepted.error();
776 tracked_command_id_ = 0;
777 return;
778 }
779 tracked_command_id_ = accepted.value();
780 outputs.command_id = tracked_command_id_;
781 outputs.command_accepted = true;
782 outputs.busy = true;
783 outputs.active = true;
784 }
785
786 void observe()
787 {
788 if(tracked_command_id_ == 0 || axis_ref == nullptr || outputs.done) {
789 return;
790 }
791 if(axis_ref->superimposed_command_id() == tracked_command_id_) {
792 outputs.busy = true;
793 outputs.active = true;
794 return;
795 }
796 if(axis_ref->superimposed_completed_id() == tracked_command_id_) {
797 outputs.done = true;
798 outputs.busy = false;
799 outputs.active = false;
800 terminal_low_cycle_ = !execute;
801 return;
802 }
803 outputs.command_aborted = true;
804 outputs.busy = false;
805 outputs.active = false;
806 terminal_low_cycle_ = !execute;
807 }
808
809 std::uint32_t tracked_command_id_ = 0;
810 bool last_execute_ = false;
811 bool terminal_low_cycle_ = false;
812 bool continuous_update_enabled_ = false;
