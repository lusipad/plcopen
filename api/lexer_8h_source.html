11// L0 lexer (approved st-l0-semantics 1.5/1.13/2.7): case-insensitive
12// identifiers and keywords, nested (* *) plus // comments, decimal and

14// literals (dot or exponent), signed TIME literals with descending
15// d/h/m/s/ms/us/ns segments (decimal tail on the last segment only).

50inline bool ascii_iequals(std::string_view text, std::string_view lower_ref)
51{
52 if(text.size() != lower_ref.size()) {
53 return false;
54 }
55 for(std::size_t i = 0; i < text.size(); ++i) {
56 if(to_lower(text[i]) != lower_ref[i]) {
57 return false;

189 explicit Lexer(std::string_view source,
190 bool trusted_debug_provenance = false)
191 : source_(source)

196 Token next()
197 {
198 skip_trivia();
199 Token token;
200 token.line = line_;
201 token.column = column_;
202 token.debug_provenance = pending_debug_provenance_;
203 token.debug_pou = pending_debug_pou_;
204 token.debug_call_path = pending_debug_call_path_;
205 token.debug_line = pending_debug_line_;
206 token.debug_column = pending_debug_column_;
207 token.debug_call_depth = pending_debug_depth_;
208 pending_debug_provenance_ = false;
209 if(pending_comment_error_) {
210 pending_comment_error_ = false;
211 token.kind = TokenKind::error;
212 token.diag_payload =
213 static_cast<std::uint16_t>(DiagCode::lex_unterminated_comment);
214 return token;
215 }
216 if(at_end()) {
217 token.kind = TokenKind::end_of_input;
218 return token;
219 }
220 const std::size_t start = pos_;
221 const char c = peek();
222 if(detail::is_ident_start(c)) {
223 return lex_word(token, start);
224 }
225 if(detail::is_digit(c)) {
226 return lex_number(token, start);
227 }
228 if(c == '\'' || c == '"') {
229 return lex_string(token, start, c);
230 }
231 if(c == '%') {
232 return lex_located_address(token, start);
233 }
234 return lex_punct(token, start);
235 }

249 void advance()
250 {
251 if(at_end()) {
252 return;
253 }
254 if(source_[pos_] == '\n') {
255 ++line_;
256 column_ = 1;
257 } else {
258 ++column_;
259 }
260 ++pos_;

262
263 void skip_trivia()
264 {
265 while(!at_end()) {
266 const char c = peek();
267 if(c == ' ' || c == '\t' || c == '\r' || c == '\n') {
268 advance();
269 continue;
270 }
271 if(c == '/' && peek(1) == '/') {
272 while(!at_end() && peek() != '\n') {
273 advance();
274 }
275 continue;
276 }
277 if(c == '(' && peek(1) == '*') {
278 if(trusted_debug_provenance_ && peek(2) == '@' &&
279 peek(3) == 'D' && peek(4) == 'B' &&
280 peek(5) == 'G' && peek(6) == ' ') {
281 parse_debug_directive();
282 continue;
283 }
284 advance();
285 advance();
286 int depth = 1;
287 while(!at_end() && depth > 0) {
288 if(peek() == '(' && peek(1) == '*') {
289 advance();
290 advance();
291 ++depth;
292 } else if(peek() == '*' && peek(1) == ')') {
293 advance();
294 advance();
295 --depth;
296 } else {
297 advance();
298 }
299 }
300 if(depth > 0) {
301 pending_comment_error_ = true;
302 return;
303 }
304 continue;
305 }
306 return;
307 }
308 }

310 void parse_debug_directive()
311 {
312 for(unsigned i = 0; i < 7; ++i) advance();
313 const auto field = [&]() {
314 while(!at_end() && peek() == ' ') advance();
315 const std::size_t begin = pos_;
316 while(!at_end() && peek() != ' ' &&
317 !(peek() == '*' && peek(1) == ')'))
318 advance();
319 return source_.substr(begin, pos_ - begin);
320 };
321 const auto number = [&](std::string_view text) {
322 std::uint32_t value = 0;
323 for(char c : text) {
324 if(c < '0' || c > '9') return std::uint32_t{0};
325 value = value * 10U + static_cast<std::uint32_t>(c - '0');
326 }
327 return value;
328 };
329 pending_debug_pou_ = field();
330 pending_debug_call_path_ = field();
331 pending_debug_line_ = static_cast<std::int32_t>(number(field()));
332 pending_debug_column_ = static_cast<std::int32_t>(number(field()));
333 pending_debug_depth_ = static_cast<std::uint16_t>(number(field()));
334 while(!at_end() && !(peek() == '*' && peek(1) == ')')) advance();
335 if(!at_end()) { advance(); advance(); }
336 pending_debug_provenance_ = !pending_debug_pou_.empty() &&
337 pending_debug_line_ > 0 &&
338 pending_debug_column_ > 0 &&
339 pending_debug_depth_ > 0;
340 }

351 Token lex_word(Token token, std::size_t start)
352 {
353 bool bad_underscore = false;
354 char previous = '\0';
355 while(!at_end() && detail::is_ident_char(peek())) {
356 if(peek() == '_' && previous == '_') {
357 bad_underscore = true;
358 }
359 previous = peek();
360 advance();
361 }
362 if(previous == '_') {
363 bad_underscore = true;
364 }
365 token.text = source_.substr(start, pos_ - start);
366
367 if(peek() == '#') {
368 TokenKind literal = TokenKind::error;
369 if(detail::ascii_iequals(token.text, "d") ||
370 detail::ascii_iequals(token.text, "date")) {
371 literal = TokenKind::date_literal;
372 } else if(detail::ascii_iequals(token.text, "tod") ||
373 detail::ascii_iequals(token.text, "time_of_day")) {
374 literal = TokenKind::tod_literal;
375 } else if(detail::ascii_iequals(token.text, "dt") ||
376 detail::ascii_iequals(token.text, "date_and_time")) {
377 literal = TokenKind::dt_literal;
378 }
379 if(literal != TokenKind::error) {
380 advance();
381 return lex_date_time(token, start, literal);
382 }
383 }
384
385 // TIME literal prefixes: T#... / TIME#...
386 if(peek() == '#' &&
387 (detail::ascii_iequals(token.text, "t") ||
388 detail::ascii_iequals(token.text, "time"))) {
389 advance(); // '#'
390 return lex_time_literal(token, start);
391 }
392
393 // Typed literal prefixes: TYPE#... (approved st-l1a-semantics 3.4).
394 if(peek() == '#') {
395 static constexpr struct
396 {
397 std::string_view lower;
398 Type type;
399 } kTypedPrefixes[] = {
400 {"bool", Type::bool_}, {"sint", Type::sint},
401 {"int", Type::int_}, {"dint", Type::dint},
402 {"lint", Type::lint}, {"usint", Type::usint},
403 {"uint", Type::uint_}, {"udint", Type::udint},
404 {"ulint", Type::ulint}, {"real", Type::real},
405 {"lreal", Type::lreal}, {"byte", Type::byte_},
406 {"word", Type::word}, {"dword", Type::dword},
407 {"lword", Type::lword},
408 };
409 for(const auto &prefix : kTypedPrefixes) {
410 if(detail::ascii_iequals(token.text, prefix.lower)) {
411 advance(); // '#'
412 return lex_typed_literal(token, start, prefix.type);
413 }
414 }
415 }
416
417 if(detail::ascii_iequals(token.text, "true")) {
418 token.kind = TokenKind::bool_literal;
419 token.unsigned_value = 1;
420 return token;
421 }
422 if(detail::ascii_iequals(token.text, "false")) {
423 token.kind = TokenKind::bool_literal;
424 token.unsigned_value = 0;
425 return token;
426 }
427 for(const detail::KeywordEntry &entry : detail::kKeywords) {
428 if(detail::ascii_iequals(token.text, entry.lower)) {
429 token.kind = entry.kind;
430 return token;
431 }
432 }
433 for(const detail::UnsupportedEntry &entry : detail::kUnsupported) {
434 if(detail::ascii_iequals(token.text, entry.lower)) {
435 token.kind = TokenKind::unsupported_keyword;
436 token.diag_payload = static_cast<std::uint16_t>(entry.code);
437 return token;
438 }
439 }
440 if(bad_underscore) {
441 return make_error(token, DiagCode::lex_bad_identifier, start);
442 }
443 token.kind = TokenKind::identifier;
444 return token;
445 }

447 Token lex_located_address(Token token, std::size_t start)
448 {
449 advance(); // '%'
450 const char area = detail::to_lower(peek());
451 if(area != 'i' && area != 'q' && area != 'm') {
452 return make_error(token, DiagCode::lex_invalid_character, start);
453 }
454 advance();
455 const char width = detail::to_lower(peek());
456 if(width != 'x' && width != 'b' && width != 'w' &&
457 width != 'd' && width != 'l') {
458 return make_error(token, DiagCode::lex_invalid_character, start);
459 }
460 advance();
461 if(!detail::is_digit(peek())) {
462 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
463 }
464 while(detail::is_digit(peek())) advance();
465 if(width == 'x' && peek() == '.') {
466 advance();
467 if(!detail::is_digit(peek())) {
468 return make_error(token, DiagCode::lex_bad_numeric_literal,
469 start);
470 }
471 while(detail::is_digit(peek())) advance();
472 }
473 token.kind = TokenKind::located_address;
474 token.text = source_.substr(start, pos_ - start);
475 return token;
476 }

478 // Accumulate digits of the given base into value; reports overflow and
479 // presence of at least one digit. Underscore separators are skipped.
480 bool read_digits(std::uint64_t base, std::uint64_t &value, bool &any,
481 bool &overflow)
482 {
483 any = false;
484 overflow = false;
485 while(!at_end()) {
486 const char c = peek();
487 if(c == '_') {
488 advance();
489 continue;
490 }
491 std::uint64_t digit = 0;
492 if(detail::is_digit(c)) {
493 digit = static_cast<std::uint64_t>(c - '0');
494 } else if(base == 16 && detail::is_alpha(c)) {
495 const char lower = detail::to_lower(c);
496 if(lower < 'a' || lower > 'f') {
497 break;
498 }
499 digit = static_cast<std::uint64_t>(lower) -
500 static_cast<std::uint64_t>('a') + 10;
501 } else {
502 break;
503 }
504 if(digit >= base) {
505 return false;
506 }
507 if(value > (~static_cast<std::uint64_t>(0) - digit) / base) {
508 overflow = true;
509 }
510 value = value * base + digit;
511 any = true;
512 advance();
513 }
514 return true;
515 }

517 Token lex_number(Token token, std::size_t start)
518 {
519 std::uint64_t value = 0;
520 bool any = false;
521 bool overflow = false;
522 if(!read_digits(10, value, any, overflow)) {
523 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
524 }
525
526 // Based literal: <base>#<digits>
527 if(peek() == '#') {
528 if(value != 2 && value != 8 && value != 16) {
529 advance();
530 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
531 }
532 advance();
533 const std::uint64_t base = value;
534 value = 0;
535 if(!read_digits(base, value, any, overflow) || !any) {
536 skip_ident_tail();
537 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
538 }
539 if(overflow) {
540 return make_error(token, DiagCode::lex_literal_overflow, start);
541 }
542 token.kind = TokenKind::int_literal;
543 token.unsigned_value = value;
544 token.based = true;
545 token.text = source_.substr(start, pos_ - start);
546 return token;
547 }
548
549 // Real literal: fraction (not '..') and/or exponent.
550 const bool has_fraction = peek() == '.' && peek(1) != '.';
551 bool is_real = has_fraction;
552 if(has_fraction) {
553 advance();
554 std::uint64_t ignored = 0;
555 bool frac_any = false;
556 bool frac_overflow = false;
557 if(!read_digits(10, ignored, frac_any, frac_overflow) || !frac_any) {
558 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
559 }
560 }
561 if(peek() == 'e' || peek() == 'E') {
562 const char sign = peek(1);
563 const std::size_t digit_at = (sign == '+' || sign == '-') ? 2 : 1;
564 if(detail::is_digit(peek(digit_at))) {
565 is_real = true;
566 advance();
567 if(sign == '+' || sign == '-') {
568 advance();
569 }
570 std::uint64_t ignored = 0;
571 bool exp_any = false;
572 bool exp_overflow = false;
573 read_digits(10, ignored, exp_any, exp_overflow);
574 }
575 }
576 token.text = source_.substr(start, pos_ - start);
577 if(is_real) {
578 token.kind = TokenKind::real_literal;
579 token.real_value = parse_real(token.text);
580 return token;
581 }
582 if(overflow) {
583 return make_error(token, DiagCode::lex_literal_overflow, start);
584 }
585 if(detail::is_ident_start(peek())) {
586 skip_ident_tail();
587 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
588 }
589 token.kind = TokenKind::int_literal;
590 token.unsigned_value = value;
591 return token;
592 }

601 // Deterministic decimal-to-binary conversion relies on the platform's
602 // correctly-rounded strtod (matrix 2.5); the text slice is copied to a
603 // bounded local buffer with underscores stripped.
604 static double parse_real(std::string_view text)
605 {
606 char buffer[64];
607 std::size_t n = 0;
608 for(char c : text) {
609 if(c == '_') {
610 continue;
611 }
612 if(n + 1 >= sizeof(buffer)) {
613 break;
614 }
615 buffer[n++] = c;
616 }
617 buffer[n] = '\0';
618 return text_to_double(buffer);
619 }

621 static double text_to_double(const char *text)
622 {
623 return std::strtod(text, nullptr);
624 }

626 Token lex_time_literal(Token token, std::size_t start)
627 {
628 bool negative = false;
629 if(peek() == '-') {
630 negative = true;
631 advance();
632 } else if(peek() == '+') {
633 advance();
634 }
635
636 // Unit ranks in mandatory descending order, no repeats (declared).
637 // rank: d=6, h=5, m=4, s=3, ms=2, us=1, ns=0
638 static constexpr std::int64_t kUnitNs[] = {
639 1, // ns
640 1000, // us
641 1000000, // ms
642 1000000000, // s
643 60000000000, // m
644 3600000000000, // h
645 86400000000000, // d
646 };
647 std::int64_t total = 0;
648 int last_rank = 7;
649 bool any_segment = false;
650 bool fraction_seen = false;
651 bool overflow = false;
652
653 while(!at_end()) {
654 if(peek() == '_') {
655 advance();
656 continue;
657 }
658 if(!detail::is_digit(peek())) {
659 break;
660 }
661 if(fraction_seen) {
662 // A fractional part is only allowed on the final segment.
663 return time_error(token, start);
664 }
665 std::uint64_t whole = 0;
666 bool any = false;
667 bool digit_overflow = false;
668 read_digits(10, whole, any, digit_overflow);
669 if(digit_overflow) {
670 overflow = true;
671 }
672 std::uint64_t frac_digits = 0;
673 std::uint64_t frac_scale = 1;
674 if(peek() == '.' && peek(1) != '.') {
675 advance();
676 fraction_seen = true;
677 while(!at_end() && (detail::is_digit(peek()) || peek() == '_')) {
678 if(peek() == '_') {
679 advance();
680 continue;
681 }
682 if(frac_scale <= 1000000000000000000ULL / 10ULL) {
683 frac_digits = frac_digits * 10 +
684 static_cast<std::uint64_t>(peek() - '0');
685 frac_scale *= 10;
686 }
687 advance();
688 }
689 }
690 const int rank = read_time_unit();
691 if(rank < 0 || rank >= last_rank) {
692 return time_error(token, start);
693 }
694 last_rank = rank;
695 any_segment = true;
696
697 const std::int64_t unit = kUnitNs[rank];
698 if(whole > static_cast<std::uint64_t>(
699 9223372036854775807LL / (unit > 0 ? unit : 1))) {
700 overflow = true;
701 } else {
702 const std::int64_t segment =
703 static_cast<std::int64_t>(whole) * unit;
704 if(total > 9223372036854775807LL - segment) {
705 overflow = true;
706 } else {
707 total += segment;
708 }
709 }
710 if(fraction_seen && frac_scale > 1) {
711 // frac_digits/frac_scale of one unit, truncated toward zero
712 // at the nanosecond grain (declared).
713 const double fractional =
714 static_cast<double>(frac_digits) /
715 static_cast<double>(frac_scale);
716 const std::int64_t extra = static_cast<std::int64_t>(
717 fractional * static_cast<double>(unit));
718 if(total > 9223372036854775807LL - extra) {
719 overflow = true;
720 } else {
721 total += extra;
722 }
723 }
724 }
725 if(!any_segment) {
726 return time_error(token, start);
727 }
728 if(overflow) {
729 return make_error(token, DiagCode::lex_literal_overflow, start);
730 }
731 token.kind = TokenKind::time_literal;
732 token.signed_value = negative ? -total : total;
733 token.text = source_.substr(start, pos_ - start);
734 return token;
735 }

737 Token lex_typed_literal(Token token, std::size_t start, Type type)
738 {
739 token.literal_type = type;
740 if(type == Type::bool_) {
741 const std::size_t word_start = pos_;
742 if(detail::is_ident_start(peek())) {
743 while(!at_end() && detail::is_ident_char(peek())) {
744 advance();
745 }
746 const std::string_view word =
747 source_.substr(word_start, pos_ - word_start);
748 if(detail::ascii_iequals(word, "true")) {
749 token.kind = TokenKind::typed_literal;
750 token.unsigned_value = 1;
751 } else if(detail::ascii_iequals(word, "false")) {
752 token.kind = TokenKind::typed_literal;
753 token.unsigned_value = 0;
754 } else {
755 return make_error(token, DiagCode::lex_bad_numeric_literal,
756 start);
757 }
758 } else if(peek() == '0' || peek() == '1') {
759 token.kind = TokenKind::typed_literal;
760 token.unsigned_value =
761 static_cast<std::uint64_t>(peek() - '0');
762 advance();
763 } else {
764 return make_error(token, DiagCode::lex_bad_numeric_literal,
765 start);
766 }
767 token.text = source_.substr(start, pos_ - start);
768 return token;
769 }
770
771 bool negative = false;
772 if(peek() == '-') {
773 negative = true;
774 advance();
775 } else if(peek() == '+') {
776 advance();
777 }
778 if(!detail::is_digit(peek())) {
779 skip_ident_tail();
780 return make_error(token, DiagCode::lex_bad_numeric_literal, start);
781 }
782 Token number;
783 number.line = token.line;
784 number.column = token.column;
785 number = lex_number(number, pos_);
786 if(number.kind == TokenKind::error) {
787 number.text = source_.substr(start, pos_ - start);
788 return number;
789 }
790 token.kind = TokenKind::typed_literal;
791 token.real_form = number.kind == TokenKind::real_literal;
792 token.real_value =
793 negative ? -number.real_value : number.real_value;
794 token.unsigned_value = number.unsigned_value;
795 token.based = number.based;
796 token.signed_value = negative ? -1 : 0; // sign marker
797 token.text = source_.substr(start, pos_ - start);
798 return token;
799 }

837 static constexpr unsigned month_days(unsigned year, unsigned month)
838 {
839 constexpr unsigned days[] = {0,31,28,31,30,31,30,31,31,30,31,30,31};
840 return month == 2U && leap_year(year) ? 29U
841 : month <= 12U ? days[month] : 0U;

844 static constexpr std::int64_t civil_days(std::int64_t year,
845 unsigned month, unsigned day)
846 {
847 year -= month <= 2U ? 1 : 0;
848 const std::int64_t era = (year >= 0 ? year : year - 399) / 400;
849 const unsigned yoe = static_cast<unsigned>(year - era * 400);
850 const unsigned shifted = static_cast<unsigned>(
851 static_cast<int>(month) + (month > 2U ? -3 : 9));
852 const unsigned doy = (153U * shifted + 2U) / 5U + day - 1U;
853 const unsigned doe = yoe * 365U + yoe / 4U - yoe / 100U + doy;
854 return era * 146097 + static_cast<std::int64_t>(doe) - 719468;
855 }

875 Token lex_date_time(Token token, std::size_t start, TokenKind kind)
876 {
877 constexpr std::int64_t kSecondNs = 1000000000LL;
878 constexpr std::int64_t kDayNs = 86400LL * kSecondNs;
879 unsigned year = 0;
880 unsigned month = 0;
881 unsigned day = 0;
882 unsigned hour = 0;
883 unsigned minute = 0;
884 unsigned second = 0;
885 std::int64_t nanoseconds = 0;
886 if(kind != TokenKind::tod_literal) {
887 if(!read_fixed(4, year) || peek() != '-') {
888 return date_error(token, start, DiagCode::date_time_range_violation);
889 }
890 advance();
891 if(!read_fixed(2, month) || peek() != '-') {
892 return date_error(token, start, DiagCode::date_time_range_violation);
893 }
894 advance();
895 if(!read_fixed(2, day) || month < 1U || month > 12U ||
896 day < 1U || day > month_days(year, month)) {
897 return date_error(token, start, DiagCode::date_time_range_violation);
898 }
899 if(kind == TokenKind::date_literal) {
900 token.signed_value = civil_days(year, month, day);
901 } else {
902 if(peek() != '-') {
903 return date_error(token, start, DiagCode::date_time_range_violation);
904 }
905 advance();
906 }
907 }
908 if(kind != TokenKind::date_literal) {
909 if(!read_fixed(2, hour) || peek() != ':') {
910 return date_error(token, start, DiagCode::date_time_range_violation);
911 }
912 advance();
913 if(!read_fixed(2, minute) || peek() != ':') {
914 return date_error(token, start, DiagCode::date_time_range_violation);
915 }
916 advance();
917 if(!read_fixed(2, second) || hour > 23U || minute > 59U ||
918 second > 59U) {
919 return date_error(token, start, DiagCode::date_time_range_violation);
920 }
921 unsigned fraction_digits = 0;
922 if(peek() == '.') {
923 advance();
924 while(detail::is_digit(peek()) && fraction_digits < 9U) {
925 nanoseconds = nanoseconds * 10 + (peek() - '0');
926 ++fraction_digits;
927 advance();
928 }
929 if(fraction_digits == 0U || detail::is_digit(peek())) {
930 return date_error(token, start, DiagCode::date_time_range_violation);
931 }
932 while(fraction_digits++ < 9U) {
933 nanoseconds *= 10;
934 }
935 }
936 const std::int64_t tod =
937 (static_cast<std::int64_t>(hour) * 3600LL +
938 static_cast<std::int64_t>(minute) * 60LL + second) *
939 kSecondNs + nanoseconds;
940 if(kind == TokenKind::tod_literal) {
941 token.signed_value = tod;
942 } else {
943 const std::int64_t days = civil_days(year, month, day);
944 if(days >= 0) {
945 const std::uint64_t limit = static_cast<std::uint64_t>(
946 std::numeric_limits<std::int64_t>::max() - tod);
947 if(static_cast<std::uint64_t>(days) >
948 limit / static_cast<std::uint64_t>(kDayNs)) {
949 return date_error(
950 token, start,
951 DiagCode::date_time_range_violation);
952 }
953 token.signed_value =
954 days * kDayNs + tod;
955 } else {
956 const std::uint64_t magnitude =
957 static_cast<std::uint64_t>(-days);
958 const std::uint64_t min_magnitude =
959 static_cast<std::uint64_t>(
960 std::numeric_limits<std::int64_t>::max()) + 1U;
961 const std::uint64_t limit =
962 min_magnitude + static_cast<std::uint64_t>(tod);
963 if(magnitude >
964 limit / static_cast<std::uint64_t>(kDayNs)) {
965 return date_error(
966 token, start,
967 DiagCode::date_time_range_violation);
968 }
969 const std::uint64_t delta =
970 magnitude * static_cast<std::uint64_t>(kDayNs) -
971 static_cast<std::uint64_t>(tod);
972 token.signed_value = delta == min_magnitude
973 ? std::numeric_limits<std::int64_t>::min()
974 : -static_cast<std::int64_t>(delta);
975 }
976 }
977 }
978 if(detail::is_alpha(peek()) || peek() == '+' || peek() == '-') {
979 return date_error(token, start, DiagCode::unsupported_l1b3_timezone);
980 }
981 token.kind = kind;
982 token.text = source_.substr(start, pos_ - start);
983 return token;
984 }

996 // Returns the unit rank (see kUnitNs) or -1. A digit may follow a unit
997 // (next segment); an alpha or underscore may not ("1msx" is malformed).
998 int read_time_unit()
999 {
1000 const char a = detail::to_lower(peek());
1001 const char b = detail::to_lower(peek(1));
1002 int rank = -1;
1003 int length = 0;
1004 if(a == 'm' && b == 's') {
1005 rank = 2;
1006 length = 2;
1007 } else if(a == 'u' && b == 's') {
1008 rank = 1;
1009 length = 2;
1010 } else if(a == 'n' && b == 's') {
1011 rank = 0;
1012 length = 2;
1013 } else if(a == 'd') {
1014 rank = 6;
1015 length = 1;
1016 } else if(a == 'h') {
1017 rank = 5;
1018 length = 1;
1019 } else if(a == 'm') {
1020 rank = 4;
1021 length = 1;
1022 } else if(a == 's') {
1023 rank = 3;
1024 length = 1;
1025 } else {
1026 return -1;
1027 }
1028 const char after = peek(static_cast<std::size_t>(length));
1029 if(detail::is_alpha(after) || after == '_') {
1030 return -1;
1031 }
1032 for(int i = 0; i < length; ++i) {
1033 advance();
1034 }
1035 return rank;
1036 }
