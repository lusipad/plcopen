14public:
15 static_assert(Capacity > 0, "StaticVector capacity must be positive");
16 // Storage is an eagerly-constructed std::array; pop_back()/clear() only
17 // adjust the size and never destroy elements, and push_back assigns into
18 // pre-constructed slots. That contract is only sound for trivial element
19 // types — fence it so a non-trivial type cannot silently "revive"
20 // dropped elements. (is_default_constructible is deliberately NOT
21 // asserted: nested aggregates with NSDMIs instantiated from inside their
22 // enclosing class would fail it spuriously.)
23 static_assert(std::is_trivially_destructible_v<T>,
24 "StaticVector stores elements eagerly and pop_back()/clear() only adjust "
25 "size; T must be trivially destructible");

27 "StaticVector assigns into pre-constructed storage; T must be trivially "
