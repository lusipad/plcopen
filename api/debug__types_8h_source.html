53inline SymbolId stable_symbol_id(std::string_view qualified_name) noexcept
54{
55 std::uint64_t hash = 1469598103934665603ULL;
56 for(unsigned char byte : qualified_name) {
57 if(byte >= 'A' && byte <= 'Z')
58 byte = static_cast<unsigned char>(byte - 'A' + 'a');
59 hash ^= byte;
60 hash *= 1099511628211ULL;
61 }
62 return SymbolId{hash};
63}

144struct SourceMap
145{
146 std::vector<SourceMapEntry> entries;
147
148 std::uint32_t first_instruction_at(const char *source,
149 std::uint32_t line) const noexcept
150 {
151 if(source == nullptr) return UINT32_MAX;
152 for(const SourceMapEntry &entry : entries) {
153 if(entry.line != line) continue;
154 std::string_view left(entry.source_name);
155 std::string_view right(source);
156 if(left.size() != right.size()) continue;
157 bool same = true;
158 for(std::size_t index = 0; index < left.size(); ++index) {
159 char a = left[index];
160 char b = right[index];
161 if(a >= 'A' && a <= 'Z') a = static_cast<char>(a - 'A' + 'a');
162 if(b >= 'A' && b <= 'Z') b = static_cast<char>(b - 'A' + 'a');
163 if(a != b) { same = false; break; }
164 }
165 if(same) return entry.instruction;
166 }
167 return UINT32_MAX;
168 }
