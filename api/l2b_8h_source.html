165inline std::vector<std::string> split_top(std::string_view text, char delimiter)
166{
167 std::vector<std::string> result;
168 int round = 0;
169 int square = 0;
170 bool quoted = false;
171 char quote = 0;
172 std::size_t begin = 0;
173 for(std::size_t i = 0; i < text.size(); ++i) {
174 const char c = text[i];
175 if(quoted) {
176 if(c == quote) {
177 if(i + 1 < text.size() && text[i + 1] == quote) {
178 ++i;
179 } else {
180 quoted = false;
181 }
182 }
183 continue;
184 }
185 if(c == '\'' || c == '"') {
186 quoted = true;
187 quote = c;
188 } else if(c == '(') {
189 ++round;
190 } else if(c == ')') {
191 --round;
192 } else if(c == '[') {
193 ++square;
194 } else if(c == ']') {
195 --square;
196 } else if(c == delimiter && round == 0 && square == 0) {
197 result.push_back(trim(text.substr(begin, i - begin)));
198 begin = i + 1;
199 }
200 }
201 result.push_back(trim(text.substr(begin)));
202 return result;
203}

211 char quote = 0;

213 const char c = text[i];
214 if(quoted) {
215 if(c == quote) {
216 quoted = false;
217 }
218 continue;

221 quoted = true;
222 quote = c;

226 --round;
227 } else if(c == '[') {
228 ++square;
229 } else if(c == ']') {

238inline void parse_declarations(std::string_view block, VarKind kind,
239 std::vector<Var> &out)
240{
241 for(const std::string &entry : split_top(block, ';')) {
242 if(entry.empty()) {
243 continue;
244 }
245 const std::size_t colon = find_top_operator(entry, ":");
246 if(colon == std::string::npos) {
247 continue;
248 }
249 const std::string names = trim(std::string_view(entry).substr(0, colon));
250 std::string tail = trim(std::string_view(entry).substr(colon + 1));
251 const std::size_t assign = find_top_operator(tail, ":=");
252 std::string type = assign == std::string::npos
253 ? trim(tail)
254 : trim(std::string_view(tail).substr(0, assign));
255 const std::string init = assign == std::string::npos
256 ? std::string{}
257 : trim(std::string_view(tail).substr(assign + 2));
258 for(const std::string &raw_name : split_top(names, ',')) {
259 if(raw_name.empty()) {
260 continue;
261 }
262 Var var;
263 const std::string lowered_name = lower_copy(raw_name);
264 const std::size_t located = lowered_name.find(" at ");
265 var.name = trim(std::string_view(raw_name).substr(0, located));
266 if(located != std::string::npos)
267 var.location = trim(std::string_view(raw_name).substr(located));
268 var.lower = lower_copy(var.name);
269 var.type = type;
270 var.type_lower = lower_copy(type);
271 var.init = init;
272 var.kind = kind;
273 out.push_back(static_cast<Var &&>(var));
274 }
275 }

289inline void parse_unit_body(std::string_view content, Pou &pou)
290{
291 std::size_t at = 0;
292 for(;;) {
293 at = skip_space(content, at);
294 std::size_t word_end = at;
295 std::string keyword = lower_copy(read_ident(content, word_end));
296 if(keyword != "var" && keyword != "var_input" &&
297 keyword != "var_output" && keyword != "var_in_out" &&
298 keyword != "var_temp" && keyword != "var_external") {
299 break;
300 }
301 const std::size_t end = find_word(content, "end_var", word_end);
302 if(end == std::string::npos) {
303 pou.body = std::string(content.substr(at));
304 return;
305 }
306 parse_declarations(content.substr(word_end, end - word_end),
307 declaration_kind(keyword), pou.vars);
308 at = end + 7;
309 }
310 pou.body = trim(content.substr(at));
311}

313inline bool parse_project(std::string_view source, Project &project,
314 std::vector<Diagnostic> &diagnostics)
315{
316 std::size_t at = 0;
317 while(at < source.size()) {
318 at = skip_space(source, at);
319 if(at >= source.size()) {
320 break;
321 }
322 if(source.substr(at, 2) == "(*") {
323 const std::size_t close = source.find("*)", at + 2);
324 at = close == std::string::npos ? source.size() : close + 2;
325 continue;
326 }
327 if(word_at(source, at, "type")) {
328 const std::size_t end = find_word(source, "end_type", at + 4);
329 if(end == std::string::npos) return false;
330 const std::size_t after = end + 8;
331 TypeUnit unit;
332 std::size_t name_at = at + 4;
333 unit.name = read_ident(source, name_at);
334 unit.lower = lower_copy(unit.name);
335 unit.source.assign(source.substr(at, after - at));
336 project.type_units.push_back(static_cast<TypeUnit &&>(unit));
337 project.types.append(source.substr(at, after - at));
338 project.types.push_back('\n');
339 at = after;
340 continue;
341 }
342 if(word_at(source, at, "var_global")) {
343 const std::size_t end = find_word(source, "end_var", at + 10);
344 if(end == std::string::npos) return false;
345 parse_declarations(source.substr(at + 10, end - at - 10),
346 VarKind::global, project.globals);
347 at = end + 7;
348 continue;
349 }
350 PouKind kind;
351 std::string_view opener;
352 std::string_view closer;
353 if(word_at(source, at, "function_block")) {
354 kind = PouKind::function_block;
355 opener = "function_block";
356 closer = "end_function_block";
357 } else if(word_at(source, at, "function")) {
358 kind = PouKind::function_;
359 opener = "function";
360 closer = "end_function";
361 } else if(word_at(source, at, "program")) {
362 kind = PouKind::program;
363 opener = "program";
364 closer = "end_program";
365 } else {
366 ++at;
367 continue;
368 }
369 std::size_t header = at + opener.size();
370 Pou pou;
371 pou.kind = kind;
372 pou.name = read_ident(source, header);
373 pou.lower = lower_copy(pou.name);
374 if(kind == PouKind::function_) {
375 header = skip_space(source, header);
376 if(header < source.size() && source[header] == ':') {
377 ++header;
378 std::size_t line = source.find_first_of("\r\n", header);
379 for(const std::string_view declaration :
380 {std::string_view{"var"}, std::string_view{"var_input"},
381 std::string_view{"var_output"},
382 std::string_view{"var_in_out"},
383 std::string_view{"var_temp"}}) {
384 const std::size_t found = find_word(source, declaration,
385 header);
386 if(found != std::string_view::npos &&
387 (line == std::string_view::npos || found < line))
388 line = found;
389 }
390 pou.return_type = trim(source.substr(
391 header, (line == std::string_view::npos ? source.size() : line) - header));
392 header = line == std::string_view::npos ? source.size() : line;
393 }
394 }
395 const std::size_t end = find_word(source, closer, header);
396 if(end == std::string::npos || pou.name.empty()) {
397 Diagnostic diagnostic;
398 diagnostic.code = DiagCode::parse_expected_token;
399 diagnostic.message = to_string(diagnostic.code);
400 diagnostics.push_back(static_cast<Diagnostic &&>(diagnostic));
401 return false;
402 }
403 parse_unit_body(source.substr(header, end - header), pou);
404 const std::size_t relative_body =
405 source.substr(header, end - header).find(pou.body);
406 pou.body_source_offset = relative_body == std::string_view::npos
407 ? header
408 : header + relative_body;
409 for(std::size_t index = 0; index < pou.body_source_offset; ++index)
410 if(source[index] == '\n') ++pou.body_line;
411 project.pous.push_back(static_cast<Pou &&>(pou));
412 at = end + closer.size();
413 }
414 if(project.pous.empty()) {
415 Diagnostic diagnostic;
416 diagnostic.code = DiagCode::parse_expected_token;
417 diagnostic.message = to_string(diagnostic.code);
418 diagnostics.push_back(static_cast<Diagnostic &&>(diagnostic));
419 return false;
420 }
421 return true;
422}

424inline void normalize_project_order(Project &project,
425 const CompileOptions &options)
426{
427 std::sort(project.globals.begin(), project.globals.end(),
428 [](const Var &left, const Var &right) {
429 return left.lower < right.lower;
430 });
431 if(project.type_units.empty()) return;
432
433 std::map<std::string, std::size_t> by_name;
434 bool duplicate = false;
435 for(std::size_t index = 0; index < project.type_units.size(); ++index) {
436 if(!by_name.emplace(project.type_units[index].lower, index).second)
437 duplicate = true;
438 }
439
440 std::vector<std::set<std::string>> dependencies(
441 project.type_units.size());
442 bool parsed_all = true;
443 for(std::size_t index = 0; index < project.type_units.size(); ++index) {
444 const std::string probe = project.type_units[index].source +
445 "\nPROGRAM L2BTypeOrder\nEND_PROGRAM\n";
446 Parser parser(probe, options.max_diagnostics, options.max_nesting);
447 ParseResult parsed = parser.parse();
448 if(!parsed.ok || parsed.ast.user_types.size() != 1U) {
449 parsed_all = false;
450 continue;
451 }
452 const UserTypeDecl &decl = parsed.ast.user_types.front();
453 project.type_units[index].name = decl.name;
454 project.type_units[index].lower = decl.lower;
455 if(decl.kind == UserTypeKind::array &&
456 !decl.element_type_name.empty()) {
457 dependencies[index].insert(lower_copy(decl.element_type_name));
458 } else if(decl.kind == UserTypeKind::struct_) {
459 for(const StructFieldDecl &field : decl.struct_fields) {
460 if(!field.type_name.empty())
461 dependencies[index].insert(lower_copy(field.type_name));
462 }
463 }
464 }
465
466 std::vector<std::size_t> order;
467 if(parsed_all && !duplicate) {
468 std::vector<std::size_t> indegree(project.type_units.size(), 0U);
469 std::vector<std::vector<std::size_t>> dependents(
470 project.type_units.size());
471 for(std::size_t node = 0; node < dependencies.size(); ++node) {
472 for(const std::string &dependency : dependencies[node]) {
473 const auto found = by_name.find(dependency);
474 if(found == by_name.end()) continue;
475 ++indegree[node];
476 dependents[found->second].push_back(node);
477 }
478 }
479 std::set<std::pair<std::string, std::size_t>> ready;
480 for(std::size_t index = 0; index < indegree.size(); ++index) {
481 if(indegree[index] == 0U)
482 ready.emplace(project.type_units[index].lower, index);
483 }
484 while(!ready.empty()) {
485 const std::size_t node = ready.begin()->second;
486 ready.erase(ready.begin());
487 order.push_back(node);
488 for(const std::size_t dependent : dependents[node]) {
489 --indegree[dependent];
490 if(indegree[dependent] == 0U) {
491 ready.emplace(project.type_units[dependent].lower,
492 dependent);
493 }
494 }
495 }
496 }
497
498 if(order.size() != project.type_units.size()) {
499 order.resize(project.type_units.size());
500 for(std::size_t index = 0; index < order.size(); ++index)
501 order[index] = index;
502 std::sort(order.begin(), order.end(),
503 [&project](std::size_t left, std::size_t right) {
504 if(project.type_units[left].lower !=
505 project.type_units[right].lower) {
506 return project.type_units[left].lower <
507 project.type_units[right].lower;
508 }
509 return left < right;
510 });
511 }
512
513 project.types.clear();
514 for(const std::size_t index : order) {
515 project.types += project.type_units[index].source;
516 project.types.push_back('\n');
517 }
518}

582 const Pou *callee = find_pou(project, name, PouKind::function_);
583 if(callee != nullptr) {

586 }

594{

648inline bool lvalue(std::string_view text)
649{
650 std::size_t at = skip_space(text, 0);
651 if(at >= text.size() ||
652 !((text[at] >= 'A' && text[at] <= 'Z') ||
653 (text[at] >= 'a' && text[at] <= 'z') || text[at] == '_')) {
654 return false;
655 }
656 int square = 0;
657 for(; at < text.size(); ++at) {
658 const char c = text[at];
659 if(c == '[') ++square;
660 else if(c == ']') --square;
661 else if(square == 0 && !(ident_char(c) || c == '.' || c == ' ' ||
662 c == '\t')) return false;
663 }
664 return square == 0;
665}

723 static std::string replace_symbols(
724 std::string_view text, const std::map<std::string, std::string> &symbols)
725 {
726 std::string out;
727 for(std::size_t i = 0; i < text.size();) {
728 if(text[i] == '\'' || text[i] == '"') {
729 const char quote = text[i];
730 do {
731 out.push_back(text[i++]);
732 } while(i < text.size() && text[i - 1] != quote);
733 continue;
734 }
735 if((text[i] >= 'A' && text[i] <= 'Z') ||
736 (text[i] >= 'a' && text[i] <= 'z') || text[i] == '_') {
737 const std::size_t begin = i++;
738 while(i < text.size() && ident_char(text[i])) ++i;
739 std::size_t compound_end = i;
740 if(i < text.size() && text[i] == '.') {
741 std::size_t j = i + 1;
742 if(j < text.size() &&
743 ((text[j] >= 'A' && text[j] <= 'Z') ||
744 (text[j] >= 'a' && text[j] <= 'z') || text[j] == '_')) {
745 ++j;
746 while(j < text.size() && ident_char(text[j])) ++j;
747 compound_end = j;
748 }
749 }
750 const std::string key = lower_copy(text.substr(
751 begin, compound_end - begin));
752 const auto found = symbols.find(key);
753 if(found != symbols.end()) {
754 out += found->second;
755 i = compound_end;
756 } else {
757 out.append(text.substr(begin, i - begin));
758 }
759 continue;
760 }
761 out.push_back(text[i++]);
762 }
763 return out;
764 }

766 static std::size_t matching_paren(std::string_view text, std::size_t open)
767 {
768 int depth = 0;
769 bool quoted = false;
770 char quote = 0;
771 for(std::size_t i = open; i < text.size(); ++i) {
772 const char c = text[i];
773 if(quoted) {
774 if(c == quote) quoted = false;
775 continue;
776 }
777 if(c == '\'' || c == '"') { quoted = true; quote = c; }
778 else if(c == '(') ++depth;
779 else if(c == ')' && --depth == 0) return i;
780 }
781 return std::string_view::npos;
782 }

784 struct ExprExpansion { std::string prelude; std::string value; };
785
786 ExprExpansion expand_expr(std::string text,
787 const std::map<std::string, std::string> &symbols,
788 const std::map<std::string, std::string> &types)
789 {
790 text = replace_symbols(text, symbols);
791 std::string prelude;
792 for(std::size_t i = 0; i < text.size();) {
793 if(!((text[i] >= 'A' && text[i] <= 'Z') ||
794 (text[i] >= 'a' && text[i] <= 'z') || text[i] == '_')) {
795 ++i;
796 continue;
797 }
798 const std::size_t begin = i++;
799 while(i < text.size() && ident_char(text[i])) ++i;
800 const std::string name = lower_copy(
801 std::string_view(text).substr(begin, i - begin));
802 std::size_t open = skip_space(text, i);
803 const Pou *callee = find_pou(project, name, PouKind::function_);
804 if(callee == nullptr || open >= text.size() || text[open] != '(') {
805 continue;
806 }
807 const std::size_t close = matching_paren(text, open);
808 if(close == std::string::npos) {
809 failed = true;
810 add_diag(diagnostics, DiagCode::parse_expected_token);
811 return {prelude, text};
812 }
813 std::vector<Actual> actuals = parse_actuals(
814 std::string_view(text).substr(open + 1, close - open - 1));
815 for(Actual &actual : actuals) {
816 if(!actual.output) {
817 ExprExpansion nested = expand_expr(actual.value, symbols, types);
818 actual.prelude = nested.prelude;
819 actual.value = nested.value;
820 } else {
821 actual.value = replace_symbols(actual.value, symbols);
822 }
823 }
824 const std::string result = expand_function(*callee, actuals,
825 types, prelude);
826 text.replace(begin, close - begin + 1, result);
827 i = begin + result.size();
828 }
829 return {prelude, text};
830 }

832 const Actual *actual_for(const std::vector<Actual> &actuals,
833 const std::vector<const Var *> &formals,
834 std::size_t formal_index,
835 std::string_view pseudo = {})
836 {
837 bool any_named = false;
838 for(const Actual &actual : actuals) any_named |= actual.named;
839 if(!pseudo.empty()) {
840 for(const Actual &actual : actuals) {
841 if(actual.named && actual.name == pseudo) return &actual;
842 }
843 return nullptr;
844 }
845 if(any_named) {
846 for(const Actual &actual : actuals) {
847 if(actual.named && actual.name == formals[formal_index]->lower)
848 return &actual;
849 }
850 return nullptr;
851 }
852 std::size_t position = 0;
853 for(std::size_t i = 0; i < formals.size(); ++i) {
854 if(formals[i]->kind == VarKind::output) continue;
855 if(i == formal_index) {
856 for(const Actual &actual : actuals) {
857 if(actual.output) continue;
858 if(position-- == 0) return &actual;
859 }
860 return nullptr;
861 }
862 ++position;
863 }
864 return nullptr;
865 }

867 bool validate_actuals(const Pou &callee, const std::vector<Actual> &actuals,
868 std::vector<const Var *> &formals)
869 {
870 for(const Var &var : callee.vars) {
871 if(var.kind == VarKind::input || var.kind == VarKind::output ||
872 var.kind == VarKind::inout) formals.push_back(&var);
873 }
874 bool named = false;
875 bool positional = false;
876 std::set<std::string> names;
877 for(const Actual &actual : actuals) {
878 if(actual.named) {
879 named = true;
880 if(!names.insert(actual.name).second) {
881 add_diag(diagnostics, DiagCode::sema_duplicate_argument);
882 failed = true;
883 return false;
884 }
885 } else positional = true;
886 }
887 if(named && positional) {
888 add_diag(diagnostics, DiagCode::sema_call_form_mixed);
889 failed = true;
890 return false;
891 }
892 std::vector<std::string> writable;
893 for(std::size_t i = 0; i < formals.size(); ++i) {
894 if(formals[i]->kind != VarKind::output &&
895 formals[i]->kind != VarKind::inout) continue;
896 const Actual *actual = actual_for(actuals, formals, i);
897 if(actual == nullptr) continue;
898 const std::string target = lower_copy(trim(actual->value));
899 if(std::find(writable.begin(), writable.end(), target) !=
900 writable.end()) {
901 add_diag(diagnostics, DiagCode::sema_alias_violation);
902 failed = true;
903 return false;
904 }
905 writable.push_back(target);
906 }
907 for(std::size_t i = 0; i < formals.size(); ++i) {
908 if(formals[i]->kind != VarKind::output &&
909 actual_for(actuals, formals, i) == nullptr) {
910 add_diag(diagnostics, DiagCode::sema_missing_argument);
911 failed = true;
912 return false;
913 }
914 }
915 return true;
916 }

918 void emit_alias_guards(const std::vector<const Var *> &formals,
919 const std::vector<Actual> &actuals,
920 std::string &code)
921 {
922 struct Target { std::string base; std::string index; };
923 std::vector<Target> targets;
924 for(std::size_t i = 0; i < formals.size(); ++i) {
925 if(formals[i]->kind != VarKind::output &&
926 formals[i]->kind != VarKind::inout) continue;
927 const Actual *actual = actual_for(actuals, formals, i);
928 if(actual == nullptr) continue;
929 const std::string target = trim(actual->value);
930 const std::size_t open = target.find('[');
931 const std::size_t close = target.rfind(']');
932 if(open == std::string::npos || close <= open) continue;
933 Target item;
934 item.base = lower_copy(trim(std::string_view(target).substr(0, open)));
935 item.index = trim(std::string_view(target).substr(
936 open + 1, close - open - 1));
937 for(const Target &prior : targets) {
938 if(prior.base == item.base && prior.index != item.index) {
939 const std::string guard = fresh("aliasguard");
940 declare(guard, "DINT");
941 code += guard + " := L2B_ALIAS_GUARD((" + prior.index +
942 ") - (" + item.index + "));";
943 }
944 }
945 targets.push_back(static_cast<Target &&>(item));
946 }
947 }

949 std::string bind_dynamic_lvalue(const std::string &source,
950 const std::string &type,
951 std::string &code)
952 {
953 std::string result = source;
954 std::size_t search = 0;
955 while((search = result.find('[', search)) != std::string::npos) {
956 const std::size_t close = result.find(']', search + 1);
957 if(close == std::string::npos) break;
958 std::vector<std::string> indices = split_top(
959 std::string_view(result).substr(search + 1,
960 close - search - 1), ',');
961 std::string replacement;
962 for(std::size_t i = 0; i < indices.size(); ++i) {
963 const std::string captured = fresh("locator");
964 declare(captured, "DINT");
965 code += captured + " := " + indices[i] + ";";
966 if(i != 0) replacement += ",";
967 replacement += captured;
968 }
969 result.replace(search + 1, close - search - 1, replacement);
970 search += replacement.size() + 2;
971 }
972 if(result != source) {
973 const std::string checked = fresh("boundcheck");
974 declare(checked, type);
975 code += checked + " := " + result + ";";
976 }
977 return result;
978 }

980 void capture_actuals(const std::vector<const Var *> &formals,
981 std::vector<Actual> &actuals, std::string &code)
982 {
983 std::size_t positional = 0;
984 for(Actual &actual : actuals) {
985 code += actual.prelude;
986 const Var *formal = nullptr;
987 if(actual.name == "en" || actual.name == "eno") {
988 if(actual.output) {
989 actual.value = bind_dynamic_lvalue(actual.value, "BOOL", code);
990 } else {
991 const std::string captured = fresh("actualen");
992 declare(captured, "BOOL");
993 code += captured + " := " + actual.value + ";";
994 actual.value = captured;
995 }
996 continue;
997 }
998 if(actual.named) {
999 for(const Var *candidate : formals) {
1000 if(candidate->lower == actual.name) formal = candidate;
1001 }
1002 } else {
1003 while(positional < formals.size() &&
1004 formals[positional]->kind == VarKind::output) ++positional;
1005 if(positional < formals.size()) formal = formals[positional++];
1006 }
1007 if(formal == nullptr) continue;
1008 if(formal->kind == VarKind::inout || actual.output) {
1009 actual.value = bind_dynamic_lvalue(actual.value, formal->type,
1010 code);
1011 } else {
1012 const std::string captured = fresh("actual");
1013 declare(captured, formal->type);
1014 code += captured + " := " + actual.value + ";";
1015 actual.value = captured;
1016 }
1017 }

1020 std::string expand_function(
1021 const Pou &callee, std::vector<Actual> actuals,
1022 const std::map<std::string, std::string> &caller_types,
1023 std::string &prelude)
1024 {
1025 std::vector<const Var *> formals;
1026 if(!validate_actuals(callee, actuals, formals)) return "0";
1027 capture_actuals(formals, actuals, prelude);
1028 emit_alias_guards(formals, actuals, prelude);
1029 const std::string prefix = fresh(callee.lower);
1030 const std::string ret = prefix + "_ret";
1031 const std::string done = prefix + "_done";
1032 declare(ret, callee.return_type);
1033 declare(done, "BOOL");
1034 std::map<std::string, std::string> symbols;
1035 std::map<std::string, std::string> types;
1036 symbols[callee.lower] = ret;
1037 types[ret] = lower_copy(callee.return_type);
1038 if(builtin_value_type(callee.return_type)) {
1039 prelude += ret + " := " + default_value(callee.return_type) + ";";
1040 }
1041 prelude += done + " := FALSE;";
1042 for(std::size_t i = 0; i < formals.size(); ++i) {
1043 const Var &formal = *formals[i];
1044 const Actual *actual = actual_for(actuals, formals, i);
1045 if(formal.kind == VarKind::inout) {
1046 if(actual == nullptr || !lvalue(actual->value)) {
1047 add_diag(diagnostics, DiagCode::sema_inout_requires_lvalue);
1048 failed = true;
1049 continue;
1050 }
1051 const std::string actual_lower = lower_copy(trim(actual->value));
1052 const auto type = caller_types.find(actual_lower);
1053 if(type != caller_types.end() && type->second != formal.type_lower) {
1054 add_diag(diagnostics, DiagCode::sema_type_mismatch);
1055 failed = true;
1056 }
1057 symbols[formal.lower] = actual->value;
1058 types[actual_lower] = formal.type_lower;
1059 continue;
1060 }
1061 const std::string local = prefix + "_" + formal.lower;
1062 declare(local, formal.type);
1063 symbols[formal.lower] = local;
1064 types[local] = formal.type_lower;
1065 if(formal.kind == VarKind::input && actual != nullptr) {
1066 prelude += local + " := " + actual->value + ";";
1067 } else {
1068 prelude += local + " := " + default_value(formal.type) + ";";
1069 }
1070 }
1071 for(const Var &var : callee.vars) {
1072 if(var.kind == VarKind::input || var.kind == VarKind::output ||
1073 var.kind == VarKind::inout) continue;
1074 if(var.kind == VarKind::external) {
1075 symbols[var.lower] = var.name;
1076 continue;
1077 }
1078 const std::string local = prefix + "_" + var.lower;
1079 declare(local, var.type);
1080 symbols[var.lower] = local;
1081 types[local] = var.type_lower;
1082 prelude += local + " := " +
1083 (var.init.empty() ? default_value(var.type) : var.init) + ";";
1084 }
1085 const Actual *en = nullptr;
1086 const Actual *eno = nullptr;
1087 for(const Actual &actual : actuals) {
1088 if(actual.name == "en") en = &actual;
1089 if(actual.name == "eno") eno = &actual;
1090 }
1091 if(eno != nullptr) prelude += eno->value + " := FALSE;";
1092 const std::string body = transform_body(callee, symbols, types, {}, done);
1093 if(en != nullptr) {
1094 prelude += "IF " + en->value + " THEN " + body;
1095 if(eno != nullptr) prelude += eno->value + " := TRUE;";
1096 prelude += " END_IF;";
1097 } else {
1098 prelude += body;
1099 if(eno != nullptr) prelude += eno->value + " := TRUE;";
1100 }
1101 std::string commit;
1102 for(std::size_t i = 0; i < formals.size(); ++i) {
1103 const Var &formal = *formals[i];
1104 const Actual *actual = actual_for(actuals, formals, i);
1105 if(formal.kind == VarKind::output && actual != nullptr && actual->output) {
1106 if(commit.empty()) commit = "L2B_COMMIT(";
1107 else commit += ",";
1108 commit += symbols[formal.lower] + "," + actual->value;
1109 }
1110 }
1111 if(!commit.empty()) prelude += commit + ");";
1112 return ret;
1113 }

1248 std::vector<StatementSlice> statements(std::string_view body)
1249 {
1250 std::vector<StatementSlice> result;
1251 int depth = 0;
1252 std::size_t begin = 0;
1253 for(std::size_t i = 0; i < body.size();) {
1254 if((body[i] >= 'A' && body[i] <= 'Z') ||
1255 (body[i] >= 'a' && body[i] <= 'z') || body[i] == '_') {
1256 const std::size_t word_begin = i++;
1257 while(i < body.size() && ident_char(body[i])) ++i;
1258 const std::string word = lower_copy(body.substr(word_begin, i - word_begin));
1259 if(word == "if" || word == "case" || word == "for" ||
1260 word == "while" || word == "repeat") ++depth;
1261 else if(word == "end_if" || word == "end_case" ||
1262 word == "end_for" || word == "end_while" ||
1263 word == "end_repeat") --depth;
1264 continue;
1265 }
1266 if(body[i] == ';' && depth == 0) {
1267 result.push_back(
1268 {std::string(body.substr(begin, i - begin + 1)), begin});
1269 begin = ++i;
1270 } else ++i;
1271 }
1272 const std::string tail = trim(body.substr(begin));
1273 if(!tail.empty())
1274 result.push_back({std::string(body.substr(begin)), begin});
1275 return result;
1276 }

1305 std::size_t if_separator(std::string_view text, std::size_t from,
1306 std::string &word)
1307 {
1308 int depth = 1;
1309 for(std::size_t i = from; i < text.size();) {
1310 if(!((text[i] >= 'A' && text[i] <= 'Z') ||
1311 (text[i] >= 'a' && text[i] <= 'z') || text[i] == '_')) {
1312 ++i;
1313 continue;
1314 }
1315 const std::size_t begin = i++;
1316 while(i < text.size() && ident_char(text[i])) ++i;
1317 const std::string token = lower_copy(text.substr(begin, i - begin));
1318 if(token == "if") ++depth;
1319 else if(token == "end_if") {
1320 if(--depth == 0) { word = token; return begin; }
1321 } else if(depth == 1 && (token == "else" || token == "elsif")) {
1322 word = token;
1323 return begin;
1324 }
1325 }
1326 word.clear();
1327 return std::string_view::npos;
1328 }
1329
1330 std::string transform_if(
1331 const Pou &owner, const std::string &raw,
1332 const std::map<std::string, std::string> &symbols,
1333 const std::map<std::string, std::string> &types,
1334 const std::map<std::string, FbRef> &instances,
1335 const std::string &done, std::size_t origin)
1336 {
1337 const std::size_t then_at = find_word(raw, "then", 2);
1338 if(then_at == std::string::npos) return raw;
1339 ExprExpansion condition = expand_expr(
1340 trim(std::string_view(raw).substr(2, then_at - 2)), symbols, types);
1341 std::string separator;
1342 const std::size_t split = if_separator(raw, then_at + 4, separator);
1343 if(split == std::string::npos) return raw;
1344 std::string result = condition.prelude + "IF " + condition.value +
1345 " THEN " +
1346 transform_sequence(owner,
1347 std::string_view(raw).substr(then_at + 4,
1348 split - then_at - 4),
1349 symbols, types, instances, done, origin + then_at + 4);
1350 if(separator == "else") {
1351 std::string end_word;
1352 const std::size_t end = if_separator(raw, split + 4, end_word);
1353 result += " ELSE " + transform_sequence(
1354 owner, std::string_view(raw).substr(split + 4,
1355 end - split - 4),
1356 symbols, types, instances, done, origin + split + 4);
1357 } else if(separator == "elsif") {
1358 const std::string nested = "IF " + raw.substr(split + 5);
1359 result += " ELSE " + transform_if(owner, nested, symbols, types,
1360 instances, done,
1361 origin + split + 5);
1362 }
1363 result += " END_IF;";
1364 return result;
1365 }
1366
1367 std::string transform_for(
1368 const Pou &owner, const std::string &raw,
1369 const std::map<std::string, std::string> &symbols,
1370 const std::map<std::string, std::string> &types,
1371 const std::map<std::string, FbRef> &instances,
1372 const std::string &done, std::size_t origin)
1373 {
1374 const std::string lower = lower_copy(raw);
1375 const std::size_t assign = lower.find(":=");
1376 const std::size_t to = find_word(lower, "to", assign + 2);
1377 const std::size_t by = find_word(lower, "by", to + 2);
1378 const std::size_t do_at = find_word(lower, "do", to + 2);
1379 const std::size_t end = lower.rfind("end_for");
1380 if(assign == std::string::npos || to == std::string::npos ||
1381 do_at == std::string::npos || end == std::string::npos) return raw;
1382 ExprExpansion from = expand_expr(trim(std::string_view(raw).substr(
1383 assign + 2, to - assign - 2)), symbols, types);
1384 const std::size_t to_end = by != std::string::npos && by < do_at
1385 ? by : do_at;
1386 ExprExpansion bound = expand_expr(trim(std::string_view(raw).substr(
1387 to + 2, to_end - to - 2)), symbols, types);
1388 ExprExpansion step;
1389 if(by != std::string::npos && by < do_at) {
1390 step = expand_expr(trim(std::string_view(raw).substr(
1391 by + 2, do_at - by - 2)), symbols, types);
1392 }
1393 std::string result = from.prelude + bound.prelude + step.prelude +
1394 raw.substr(0, assign + 2) + " " + from.value + " TO " +
1395 bound.value;
1396 if(by != std::string::npos && by < do_at) result += " BY " + step.value;
1397 result += " DO " + transform_sequence(
1398 owner, std::string_view(raw).substr(do_at + 2, end - do_at - 2),
1399 symbols, types, instances, done, origin + do_at + 2);
1400 if(!done.empty()) result += "IF " + done + " THEN EXIT; END_IF;";
1401 result += " END_FOR;";
1402 return result;
1403 }

1405 std::string transform_while(
1406 const Pou &owner, const std::string &raw,
1407 const std::map<std::string, std::string> &symbols,
1408 const std::map<std::string, std::string> &types,
1409 const std::map<std::string, FbRef> &instances,
1410 const std::string &done, std::size_t origin)
1411 {
1412 const std::string lower = lower_copy(raw);
1413 const std::size_t do_at = find_word(lower, "do", 5);
1414 const std::size_t end = lower.rfind("end_while");
1415 if(do_at == std::string::npos || end == std::string::npos) return raw;
1416 ExprExpansion condition = expand_expr(trim(std::string_view(raw).substr(
1417 5, do_at - 5)), symbols, types);
1418 std::string result = "WHILE TRUE DO " + condition.prelude +
1419 "IF NOT (" + condition.value + ") THEN EXIT; END_IF;" +
1420 transform_sequence(owner,
1421 std::string_view(raw).substr(do_at + 2, end - do_at - 2),
1422 symbols, types, instances, done, origin + do_at + 2);
1423 if(!done.empty()) result += "IF " + done + " THEN EXIT; END_IF;";
1424 result += " END_WHILE;";
1425 return result;
1426 }
1427
1428 std::string transform_sequence(
1429 const Pou &owner, std::string_view source,
1430 const std::map<std::string, std::string> &symbols,
1431 const std::map<std::string, std::string> &types,
1432 const std::map<std::string, FbRef> &instances,
1433 const std::string &done, std::size_t origin = 0)
1434 {
1435 std::string result;
1436 for(StatementSlice slice : statements(source)) {
1437 std::string statement = static_cast<std::string &&>(slice.text);
1438 std::size_t leading = 0;
1439 for(char value : statement) {
1440 if(value == ' ' || value == '\t' || value == '\r' ||
1441 value == '\n') {
1442 result.push_back(value);
1443 ++leading;
1444 } else {
1445 break;
1446 }
1447 }
1448 const std::size_t statement_origin =
1449 origin + slice.begin + leading;
1450 statement = replace_symbols(statement, symbols);
1451 const std::string raw = trim(statement);
1452 const std::string lower = lower_copy(raw);
1453 std::string emitted;
1454 if(lower == "return;") {
1455 emitted = done + " := TRUE;";
1456 } else if(word_at(lower, 0, "if")) {
1457 emitted = transform_if(owner, raw, symbols, types, instances,
1458 done, statement_origin);
1459 } else if(word_at(lower, 0, "for")) {
1460 emitted = transform_for(owner, raw, symbols, types, instances,
1461 done, statement_origin);
1462 } else if(word_at(lower, 0, "while")) {
1463 emitted = transform_while(owner, raw, symbols, types,
1464 instances, done, statement_origin);
1465 } else {
1466 std::size_t p = 0;
1467 const std::string first = lower_copy(read_ident(raw, p));
1468 p = skip_space(raw, p);
1469 const auto fb = instances.find(first);
1470 if(fb != instances.end() && p < raw.size() && raw[p] == '(') {
1471 const std::size_t close = matching_paren(raw, p);
1472 std::vector<Actual> actuals = parse_actuals(
1473 std::string_view(raw).substr(p + 1, close - p - 1));
1474 for(Actual &actual : actuals) {
1475 if(!actual.output) {
1476 ExprExpansion expanded = expand_expr(
1477 actual.value, symbols, types);
1478 actual.prelude = expanded.prelude;
1479 actual.value = expanded.value;
1480 } else actual.value = replace_symbols(actual.value,
1481 symbols);
1482 }
1483 emitted = expand_fb_call(fb->second, actuals);
1484 } else {
1485 ExprExpansion expanded = expand_expr(raw, symbols, types);
1486 emitted = expanded.prelude + expanded.value;
1487 }
1488 }
1489 if(!done.empty() && lower != "return;" &&
1490 contains_word(owner.body, "return")) {
1491 result += "IF NOT ";
1492 result += done;
1493 result += " THEN ";
1494 result += debug_directive(owner, statement_origin);
1495 result += emitted;
1496 result += " END_IF;";
1497 } else {
1498 result += debug_directive(owner, statement_origin) + emitted;
1499 }
1500 }
1501 return result;
1502 }
1503
1504 std::string transform_body(
1505 const Pou &owner, const std::map<std::string, std::string> &symbols,
1506 const std::map<std::string, std::string> &types,
1507 const std::map<std::string, FbRef> &instances,
1508 const std::string &done)
1509 {
1510 debug_stack_.push_back(&owner);
1511 std::string result = transform_sequence(owner, owner.body, symbols,
1512 types, instances, done, 0);
1513 debug_stack_.pop_back();
1514 return result;
1515 }
1516 std::vector<const Pou *> debug_stack_;
1517};

1633inline bool append_layout(std::uint64_t &offset, TypeLayout layout)
1634{
1635 const std::uint64_t alignment =
1636 std::min<std::uint32_t>(8U, std::max<std::uint32_t>(1U,
1637 layout.alignment));
1638 const std::uint64_t mask = alignment - 1U;
1639 if(offset > std::numeric_limits<std::uint64_t>::max() - mask)
1640 return false;
1641 offset = (offset + mask) & ~mask;
1642 if(layout.size > std::numeric_limits<std::uint64_t>::max() - offset)
1643 return false;
1644 offset += layout.size;
1645 return true;
1646}
1647
1648inline bool finish_layout(std::uint64_t &bytes)
1649{
1650 constexpr std::uint64_t mask = 7U;
1651 if(bytes > std::numeric_limits<std::uint64_t>::max() - mask)
1652 return false;
1653 bytes = (bytes + mask) & ~mask;
1654 return bytes <= std::numeric_limits<std::uint32_t>::max();
1655}

1759inline bool compile_pou_report_image(const Project &project, const Pou &pou,
1760 const CompileOptions &options,
1761 std::vector<Diagnostic> &diagnostics,
1762 Program &report)
1763{
1764 Expansion expansion{project, options, diagnostics};
1765 std::map<std::string, std::string> symbols;
1766 std::map<std::string, std::string> types;
1767 std::map<std::string, Expansion::FbRef> instances;
1768 std::set<std::string> declared;
1769
1770 for(const Var &var : pou.vars) {
1771 if(var.kind != VarKind::external) declared.insert(var.lower);
1772 }
1773 for(const Var &global : project.globals) {
1774 if(declared.insert(global.lower).second) {
1775 expansion.declarations.push_back(declaration_text(global));
1776 types[global.lower] = global.type_lower;
1777 }
1778 }
1779 for(const Var &var : pou.vars) {
1780 if(var.kind == VarKind::external) {
1781 symbols[var.lower] = var.name;
1782 continue;
1783 }
1784 const Pou *fb = find_pou(project, var.type_lower,
1785 PouKind::function_block);
1786 if(fb != nullptr) {
1787 const std::string path = expansion.fresh(var.lower);
1788 std::map<std::string, Expansion::FbRef> nested_instances;
1789 expansion.declare_instance(*fb, path, symbols, types,
1790 nested_instances);
1791 instances[var.lower] = {fb, path};
1792 for(const Var &pin : fb->vars) {
1793 if(pin.kind == VarKind::input ||
1794 pin.kind == VarKind::output) {
1795 symbols[var.lower + "." + pin.lower] =
1796 expansion.storage_name(path, pin.lower);
1797 }
1798 }
1799 } else {
1800 const std::string local = expansion.fresh(var.lower);
1801 expansion.declare(local, var.type, var.init);
1802 symbols[var.lower] = local;
1803 types[local] = var.type_lower;
1804 }
1805 }
1806
1807 if(pou.kind == PouKind::function_) {
1808 const std::string result_name = expansion.fresh("report_result");
1809 expansion.declare(result_name, pou.return_type);
1810 symbols[pou.lower] = result_name;
1811 types[result_name] = lower_copy(pou.return_type);
1812 }
1813 const std::string done = expansion.fresh("report_done");
1814 expansion.declare(done, "BOOL");
1815 const std::string body = done + " := FALSE;" + expansion.transform_body(
1816 pou, symbols, types, instances, done);
1817 if(expansion.failed) return false;
1818
1819 std::string transformed = project.types + "PROGRAM L2BReport\nVAR\n";
1820 for(const std::string &declaration : expansion.declarations) {
1821 transformed += declaration;
1822 transformed.push_back('\n');
1823 }
1824 transformed += "END_VAR\n" + body + "\nEND_PROGRAM\n";
1825 CompileOptions report_options = options;
1826 report_options.max_vars_bytes = std::numeric_limits<std::uint32_t>::max();
1827 CompileResult image = compile_single_program(transformed, report_options);
1828 if(!image.ok) {
1829 diagnostics.insert(diagnostics.end(), image.diagnostics.begin(),
1830 image.diagnostics.end());
1831 return false;
1832 }
1833 report = static_cast<Program &&>(image.program);
1834 report.program_name = pou.lower;
1835 return true;
1836}

1838inline bool project_semantics(const Project &project,

1840 const TypeLayouts &layouts,
1841 std::vector<Diagnostic> &diagnostics,
1842 std::uint16_t &max_calls,
1843 std::uint16_t &max_instances)

1846 add_diag(diagnostics, DiagCode::capacity_exceeded);
1847 return false;
1848 }
1849 std::set<std::string> global_names;
1850 for(const Var &global : project.globals) {
1851 if(!global_names.insert(global.lower).second) {
1852 add_diag(diagnostics, DiagCode::sema_external_not_unique);
1853 return false;
1854 }
1855 }
1856 std::set<std::string> names;
1857 for(const Pou &pou : project.pous) {
1858 if(!names.insert(pou.lower).second) {
1859 add_diag(diagnostics, DiagCode::sema_duplicate_pou);
1860 return false;
1861 }
1862 std::size_t parameters = 0;
1863 for(const Var &var : pou.vars) {
1864 if(var.kind == VarKind::input || var.kind == VarKind::output ||
1865 var.kind == VarKind::inout) ++parameters;
1866 }
1867 if(parameters > options.max_parameters_per_pou) {
1868 add_diag(diagnostics, DiagCode::capacity_exceeded);
1869 return false;
1870 }
1871 if(pou.kind == PouKind::function_ &&
1872 is_reserved_standard_function(pou.lower)) {
1873 add_diag(diagnostics, DiagCode::sema_standard_name_conflict);
1874 return false;
1875 }
1876 for(const Var &global : project.globals) {
1877 if(contains_word(pou.body, global.lower) &&
1878 find_var(pou.vars, global.lower) == nullptr &&
1879 pou.kind != PouKind::program) {
1880 add_diag(diagnostics, DiagCode::sema_external_required);
1881 return false;
1882 }
1883 }
1884 for(const Pou &owner : project.pous) {
1885 if(owner.lower == pou.lower) continue;
1886 for(const Var &foreign : owner.vars) {
1887 if(foreign.kind != VarKind::local &&
1888 foreign.kind != VarKind::temp) continue;
1889 if(find_var(pou.vars, foreign.lower) == nullptr &&
