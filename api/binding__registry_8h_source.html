150constexpr std::array<BindingDispatcher, generated::kStBindingFbs.size()>
151make_dispatchers()
152{
153 std::array<BindingDispatcher, generated::kStBindingFbs.size()> result{};
154 for(std::size_t index = 0; index < result.size(); ++index) {
155 const generated::StBindingFbMetadata &metadata =
156 generated::kStBindingFbs[index];
157 const generated::StBindingNativeCapabilityMasks masks =
158 generated::st_binding_native_capabilities(metadata.type);
159 result[index] = {
160 static_cast<BindingFbId>(metadata.stable_id), metadata.type,
161 masks.store_scalar, masks.load_scalar, masks.store_object,
162 masks.load_object, masks.store_sequence, masks.load_sequence,
163 masks.store_tagged_reference, masks.load_tagged_reference,
164 kInit, kStore, kStoreObject, kLoad, kLoadObject, kStoreSequence,
165 kLoadSequence, kStoreTaggedReference, kLoadTaggedReference,
166 kCycle};
167 }
168 return result;
169}

184inline bool binding_pin_registered(BindingFbId fb_id,
185 BindingPinId pin_id) noexcept
186{
187 const BindingDispatcher *dispatcher = binding_dispatcher(fb_id);
188 const BindingFbDesc *fb_desc = binding_manifest().find(fb_id);
189 if(dispatcher == nullptr || fb_desc == nullptr ||
190 pin_id >= fb_desc->pin_count) {
191 return false;
192 }
193 const BindingPinDesc &pin = fb_desc->pins[pin_id];
194 const TypeDesc *type = binding_manifest().type_table().get(pin.type_id);
195 if(type == nullptr) return false;
196 const auto stores = [&] {
197 return dispatcher->stores(pin_id) || dispatcher->stores_object(pin_id) ||
198 dispatcher->stores_sequence(pin_id) ||
199 dispatcher->stores_tagged_reference(pin_id);
200 };
201 const auto loads = [&] {
202 return dispatcher->loads(pin_id) || dispatcher->loads_object(pin_id) ||
203 dispatcher->loads_sequence(pin_id) ||
204 dispatcher->loads_tagged_reference(pin_id);
205 };
206 switch(pin.direction) {
207 case PinDirection::input: return stores();
208 case PinDirection::output: return loads();
209 case PinDirection::in_out: return stores() && loads();
210 }
211 return false;
212}

214inline bool binding_fb_registered(BindingFbId id) noexcept
215{
216 const BindingDispatcher *dispatcher = binding_dispatcher(id);
217 const BindingFbDesc *fb_desc = binding_manifest().find(id);
218 if(dispatcher == nullptr || fb_desc == nullptr ||
219 dispatcher->storage_size() == 0U || dispatcher->init_op == nullptr ||
220 dispatcher->store_op == nullptr || dispatcher->load_op == nullptr ||
221 dispatcher->store_object_op == nullptr ||
222 dispatcher->load_object_op == nullptr ||
223 dispatcher->store_sequence_op == nullptr ||
224 dispatcher->load_sequence_op == nullptr ||
225 dispatcher->store_tagged_reference_op == nullptr ||
226 dispatcher->load_tagged_reference_op == nullptr ||
227 dispatcher->cycle_op == nullptr) {
228 return false;
229 }
230 for(std::size_t index = 0; index < fb_desc->pin_count; ++index) {
231 if(!binding_pin_registered(id, fb_desc->pins[index].id)) return false;
232 }
233 return true;
234}
