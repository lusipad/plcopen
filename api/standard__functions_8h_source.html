63inline bool resolve_standard_function(std::string_view lower,
64 StandardFunction &function)
65{
66 for(std::size_t index = 0; index < kStandardFunctionManifest.size(); ++index) {
67 const std::string_view upper = kStandardFunctionManifest[index].name;
68 if(lower.size() != upper.size()) continue;
69 bool equal = true;
70 for(std::size_t c = 0; c < lower.size(); ++c) {
71 const char expected = upper[c] >= 'A' && upper[c] <= 'Z'
72 ? static_cast<char>(upper[c] - 'A' + 'a') : upper[c];
73 if(lower[c] != expected) { equal = false; break; }
74 }
75 if(equal) {
76 function = static_cast<StandardFunction>(index);
77 return true;
78 }
79 }
80 return false;
81}

84 std::uint32_t right)

90inline constexpr std::uint32_t saturating_cost_mul(std::uint32_t left,
91 std::uint32_t right)
92{
93 return left != 0U &&
94 right > std::numeric_limits<std::uint32_t>::max() / left
95 ? std::numeric_limits<std::uint32_t>::max() : left * right;

98inline constexpr std::uint32_t standard_string_cost(
99 StandardFunction function, const std::uint32_t *capacities,
100 std::uint8_t count, std::uint32_t destination_capacity)
101{
102 if(count == 0U) return 1U;
103 if(function == StandardFunction::len) {
104 return std::max(1U, saturating_cost_mul(capacities[0], 2U));
105 }
106 if(function == StandardFunction::find && count >= 2U) {
107 const std::uint32_t search = saturating_cost_mul(capacities[0],
108 capacities[1]);
109 return std::max(1U, saturating_cost_add(
110 search, saturating_cost_add(capacities[0], capacities[1])));
111 }
112 std::uint32_t total = destination_capacity;
113 for(std::uint8_t index = 0; index < count; ++index) {
114 total = saturating_cost_add(total, capacities[index]);
115 }
116 return std::max(1U, saturating_cost_mul(total, 4U));
117}
