114inline State1D sample_segment(const Segment1D &segment, std::int64_t cycle)
115{
116 if(cycle <= 0) {
117 return segment.start;
118 }
119 if(cycle >= segment.duration_cycles) {
120 return segment.finish;
121 }
122
123 const double x = static_cast<double>(cycle);
124 const double x2 = x * x;
125 const double x3 = x2 * x;
126 const double x4 = x3 * x;
127 const double x5 = x4 * x;
128
129 State1D state{};
130 state.position =
131 segment.c0 + segment.c1 * x + segment.c2 * x2 + segment.c3 * x3 +
132 segment.c4 * x4 + segment.c5 * x5;
133 state.velocity =
134 segment.c1 + 2.0 * segment.c2 * x + 3.0 * segment.c3 * x2 +
135 4.0 * segment.c4 * x3 + 5.0 * segment.c5 * x4;
136 state.acceleration =
137 2.0 * segment.c2 + 6.0 * segment.c3 * x + 12.0 * segment.c4 * x2 +
138 20.0 * segment.c5 * x3;
139 return state;
140}

142inline State1D sample(const Profile1D &profile, rt::CycleTick tick)
143{
144 std::int64_t remaining = tick.cycles();
145 for(std::size_t i = 0; i < profile.segment_count(); ++i) {
146 const Segment1D &current = profile.segment(i);
147 if(remaining <= current.duration_cycles) {
148 return sample_segment(current, remaining);
149 }
150 remaining -= current.duration_cycles;
151 }
152
153 if(profile.segment_count() == 0) {
154 return State1D{};
155 }
156 const Segment1D &last = profile.segment(profile.segment_count() - 1);
157 return last.finish;
158}

166inline Segment1D make_quintic_segment(State1D from, Target1D to, std::int64_t cycles)
167{
168 const double t = static_cast<double>(cycles);
169 const double t2 = t * t;
170 const double t3 = t2 * t;
171 const double t4 = t3 * t;
172 const double t5 = t4 * t;
173 const double dp = to.position - from.position;
174
175 Segment1D segment{};
176 segment.duration_cycles = cycles;
177 segment.start = from;
178 segment.finish = {to.position, to.velocity, to.acceleration};
179 segment.c0 = from.position;
180 segment.c1 = from.velocity;
181 segment.c2 = 0.5 * from.acceleration;
182 segment.c3 =
183 (20.0 * dp - (8.0 * to.velocity + 12.0 * from.velocity) * t -
184 (3.0 * from.acceleration - to.acceleration) * t2) /
185 (2.0 * t3);
186 segment.c4 =
187 (-30.0 * dp + (14.0 * to.velocity + 16.0 * from.velocity) * t +
188 (3.0 * from.acceleration - 2.0 * to.acceleration) * t2) /
189 (2.0 * t4);
190 segment.c5 =
191 (12.0 * dp - (6.0 * to.velocity + 6.0 * from.velocity) * t -
192 (from.acceleration - to.acceleration) * t2) /
193 (2.0 * t5);
194 return segment;
195}

197inline bool within_limits(const Segment1D &segment, Limits1D limits)
198{
199 constexpr int Samples = 96;
200 constexpr double Epsilon = 1e-9;
201
202 for(int i = 0; i <= Samples; ++i) {
203 const std::int64_t cycle =
204 (segment.duration_cycles * static_cast<std::int64_t>(i)) / Samples;
205 const State1D state = sample_segment(segment, cycle);
206 if(std::fabs(state.velocity) > limits.max_velocity + Epsilon) {
207 return false;
208 }
209 if(state.acceleration > limits.max_acceleration + Epsilon) {
210 return false;
211 }
212 if(state.acceleration < -limits.max_deceleration - Epsilon) {
213 return false;
214 }
215 if(std::fabs(jerk_at(segment, cycle)) > limits.max_jerk + Epsilon) {
216 return false;
217 }
218 }
219
220 return true;
221}

230inline rt::Result<Profile1D> plan(State1D from, Target1D to, Limits1D limits)
231{
232 if(!is_finite(from) || !is_finite(to) || !is_finite(limits) || limits.max_velocity <= 0.0 ||
233 limits.max_acceleration <= 0.0 || limits.max_deceleration <= 0.0 ||
234 limits.max_jerk <= 0.0) {
235 return rt::Result<Profile1D>::failure(rt::ErrorCode::invalid_argument);
236 }
237
238 const State1D finish{to.position, to.velocity, to.acceleration};
239 if(!state_within_limits(from, limits) || !state_within_limits(finish, limits)) {
240 return rt::Result<Profile1D>::failure(rt::ErrorCode::infeasible);
241 }
242
243 const double distance = std::fabs(to.position - from.position);
244 const double velocity_floor = limits.max_velocity * 0.25;
245 double guess = 1.0;
246 if(distance > 0.0) {
247 guess = distance / velocity_floor;
248 const double by_accel =
249 std::sqrt((2.0 * distance) / limits.max_acceleration);
250 const double by_decel =
251 std::sqrt((2.0 * distance) / limits.max_deceleration);
252 const double by_jerk = std::cbrt((6.0 * distance) / limits.max_jerk);
253 if(by_accel > guess) {
254 guess = by_accel;
255 }
256 if(by_decel > guess) {
257 guess = by_decel;
258 }
259 if(by_jerk > guess) {
260 guess = by_jerk;
261 }
262 }
263
264 const double start_stop =
265 (std::fabs(from.velocity) + std::fabs(to.velocity)) / limits.max_acceleration + 1.0;
266 if(start_stop > guess) {
267 guess = start_stop;
268 }
269
270 std::int64_t cycles = static_cast<std::int64_t>(std::ceil(guess));
271 if(cycles < 1) {
272 cycles = 1;
273 }
274
275 for(int attempt = 0; attempt < 80; ++attempt) {
276 const Segment1D segment = make_quintic_segment(from, to, cycles);
277 if(within_limits(segment, limits)) {
278 Profile1D profile{};
279 const rt::ErrorCode added = profile.add_segment(segment);
280 if(added != rt::ErrorCode::ok) {
281 return rt::Result<Profile1D>::failure(added);
282 }
283 return rt::Result<Profile1D>::success(profile);
284 }
285 cycles = cycles + cycles / 4 + 1;
286 }
287
288 return rt::Result<Profile1D>::failure(rt::ErrorCode::infeasible);
289}
