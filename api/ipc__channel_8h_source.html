35template <typename T> struct AtomicPayloadTraits
36{
37 static_assert(std::is_trivially_copyable_v<T>, "ipc payloads must be trivially copyable");
38 static constexpr std::size_t WordSize = sizeof(std::uint64_t);
39 static_assert(sizeof(T) % WordSize == 0,
40 "ipc payload size must be a multiple of uint64_t");
41
42 static constexpr std::size_t WordCount = sizeof(T) / WordSize;
43 using WordArray = std::array<std::uint64_t, WordCount>;
44 using AtomicWordArray = std::array<std::atomic<std::uint64_t>, WordCount>;
45};
46
47template <typename T>
48inline void store_payload(typename AtomicPayloadTraits<T>::AtomicWordArray &storage, const T &value)
49{
50 typename AtomicPayloadTraits<T>::WordArray words{};
51 std::memcpy(words.data(), &value, sizeof(T));
52 for(std::size_t index = 0; index < words.size(); ++index) {
53 storage[index].store(words[index], std::memory_order_relaxed);
54 }
55}
56
57template <typename T>
58inline void load_payload(const typename AtomicPayloadTraits<T>::AtomicWordArray &storage, T &value)
59{
60 typename AtomicPayloadTraits<T>::WordArray words{};
61 for(std::size_t index = 0; index < words.size(); ++index) {
62 words[index] = storage[index].load(std::memory_order_relaxed);
63 }
64 std::memcpy(&value, words.data(), sizeof(T));
65}

67inline bool claim_owner(std::atomic<std::uint64_t> &owner, std::uint64_t id)
68{
69 if(id == 0) {
70 return false;
71 }
72 std::uint64_t expected = 0;
73 // Claim is a one-shot latch for the mapping lifetime. Even the same token
74 // cannot claim twice; a restarted cooperative peer must rebuild the region.
75 return owner.compare_exchange_strong(expected, id, std::memory_order_acq_rel,
76 std::memory_order_acquire);
77}
78
79inline bool owner_matches(const std::atomic<std::uint64_t> &owner, std::uint64_t id)
80{
81 return id != 0 && owner.load(std::memory_order_acquire) == id;
82}

88public:

91 void initialize()
92 {
93 writer_owner_.store(0, std::memory_order_relaxed);
94 reader_owner_.store(0, std::memory_order_relaxed);
95 head_.store(0, std::memory_order_relaxed);
96 tail_.store(0, std::memory_order_relaxed);
97 for(auto &slot : slots_) {
98 for(auto &word : slot.words) {
99 word.store(0, std::memory_order_relaxed);
100 }
101 }
102 }

104 bool claim_writer(std::uint64_t owner)
105 {

107 }
108
109 bool claim_reader(std::uint64_t owner)
110 {
111 return detail::claim_owner(reader_owner_, owner);
112 }
113
114 IpcStatus push(std::uint64_t owner, const T &value)
115 {
116 if(!detail::owner_matches(writer_owner_, owner)) {
117 return IpcStatus::invalid_owner;
118 }
119
120 const std::uint64_t head = head_.load(std::memory_order_relaxed);
121 const std::uint64_t next = increment(head);
122 if(next == tail_.load(std::memory_order_acquire)) {
123 return IpcStatus::full;
124 }
125
126 detail::store_payload<T>(slots_[static_cast<std::size_t>(head)].words, value);
127 head_.store(next, std::memory_order_release);
128 return IpcStatus::ok;
129 }
130
131 IpcStatus pop(std::uint64_t owner, T &value)
132 {
133 if(!detail::owner_matches(reader_owner_, owner)) {
134 return IpcStatus::invalid_owner;
135 }
136
137 const std::uint64_t tail = tail_.load(std::memory_order_relaxed);
138 if(tail == head_.load(std::memory_order_acquire)) {
139 return IpcStatus::empty;
140 }
141
142 detail::load_payload<T>(slots_[static_cast<std::size_t>(tail)].words, value);
143 tail_.store(increment(tail), std::memory_order_release);
144 return IpcStatus::ok;
145 }
146
147private:
148 static constexpr std::size_t StorageSize = Capacity + 1;
149
150 struct Slot
151 {
152 typename detail::AtomicPayloadTraits<T>::AtomicWordArray words{};
153 };
154
155 static constexpr std::uint64_t increment(std::uint64_t index)
156 {
157 return (index + 1u) % static_cast<std::uint64_t>(StorageSize);
158 }
159
160 std::array<Slot, StorageSize> slots_{};
161 std::atomic<std::uint64_t> writer_owner_{0};
162 std::atomic<std::uint64_t> reader_owner_{0};
163 std::atomic<std::uint64_t> head_{0};
164 std::atomic<std::uint64_t> tail_{0};
165};

167template <typename T> class IpcDoubleBuffer
168{
169public:
170 void initialize()
171 {
172 writer_owner_.store(0, std::memory_order_relaxed);
173 pending_state_.store(0, std::memory_order_relaxed);
174 published_state_.store(0, std::memory_order_relaxed);
175 for(auto &slot : slots_) {
176 slot.stamp.store(0, std::memory_order_relaxed);
177 for(auto &word : slot.words) {
178 word.store(0, std::memory_order_relaxed);
179 }
180 }
181 }
182
183 bool claim_writer(std::uint64_t owner)
184 {
185 return detail::claim_owner(writer_owner_, owner);
186 }
187
188 IpcStatus publish(std::uint64_t owner, const T &value)
189 {
190 IpcWriteReservation reservation{};
191 IpcStatus status = begin_publish(owner, reservation);
192 if(status != IpcStatus::ok) {
193 return status;
194 }
195 status = write_pending(owner, reservation, value);
196 if(status != IpcStatus::ok) {
197 pending_state_.store(0, std::memory_order_release);
198 reservation.active = false;
199 return status;
200 }
201 return commit_publish(owner, reservation);
202 }
203
204 IpcStatus begin_publish(std::uint64_t owner, IpcWriteReservation &reservation)
205 {
206 if(!detail::owner_matches(writer_owner_, owner)) {
207 return IpcStatus::invalid_owner;
208 }
209
210 const std::uint64_t state = published_state_.load(std::memory_order_acquire);
211 const std::uint64_t sequence = unpack_sequence(state);
212 const std::size_t active_slot = unpack_slot(state);
213 const std::size_t pending_slot = (sequence == 0) ? 0 : (active_slot ^ 1u);
214 const std::uint64_t pending = pack_state(sequence + 1u, pending_slot);
215
216 std::uint64_t expected = 0;
217 if(!pending_state_.compare_exchange_strong(expected, pending, std::memory_order_acq_rel,
218 std::memory_order_acquire)) {
219 return IpcStatus::busy;
220 }
221
222 reservation.sequence = sequence + 1;
223 reservation.slot = pending_slot;
224 reservation.active = true;
225 slots_[pending_slot].stamp.store((reservation.sequence << 1u) | 1u,
226 std::memory_order_release);
227 return IpcStatus::ok;
228 }
229
230 IpcStatus write_pending(std::uint64_t owner, const IpcWriteReservation &reservation,
231 const T &value)
232 {
233 if(!detail::owner_matches(writer_owner_, owner)) {
234 return IpcStatus::invalid_owner;
235 }
236 if(!reservation.active || reservation.slot >= slots_.size()) {
237 return IpcStatus::busy;
238 }
239
240 const std::uint64_t pending = pending_state_.load(std::memory_order_acquire);
241 if(pending == 0 || pending != pack_state(reservation.sequence, reservation.slot)) {
242 return IpcStatus::busy;
243 }
244
245 detail::store_payload<T>(slots_[reservation.slot].words, value);
246 return IpcStatus::ok;
247 }
248
249 IpcStatus commit_publish(std::uint64_t owner, IpcWriteReservation &reservation)
250 {
251 if(!detail::owner_matches(writer_owner_, owner)) {
252 return IpcStatus::invalid_owner;
253 }
254 if(!reservation.active || reservation.slot >= slots_.size()) {
255 return IpcStatus::busy;
256 }
257
258 const std::uint64_t pending = pack_state(reservation.sequence, reservation.slot);
259 std::uint64_t expected = pending;
260 if(!pending_state_.compare_exchange_strong(
261 expected, CommittingState, std::memory_order_acq_rel, std::memory_order_acquire)) {
262 return IpcStatus::busy;
263 }
264
265 slots_[reservation.slot].stamp.store(reservation.sequence << 1u, std::memory_order_release);
266 published_state_.store(pack_state(reservation.sequence, reservation.slot),
267 std::memory_order_release);
268 pending_state_.store(0, std::memory_order_release);
269 reservation.active = false;
270 return IpcStatus::ok;
271 }

273 IpcStatus read(T &value, std::uint64_t &sequence, std::size_t max_retries) const
274 {
275 const std::uint64_t first_state = published_state_.load(std::memory_order_acquire);
276 if(unpack_sequence(first_state) == 0) {
277 return IpcStatus::empty;
278 }
279 if(max_retries == 0) {
280 return IpcStatus::busy;
281 }
282
283 for(std::size_t attempt = 0; attempt < max_retries; ++attempt) {
284 const std::uint64_t state = published_state_.load(std::memory_order_acquire);
285 const std::uint64_t published_sequence = unpack_sequence(state);
286 if(published_sequence == 0) {
287 return IpcStatus::empty;
288 }
289
290 const std::size_t slot_index = unpack_slot(state);
291 const std::uint64_t stable_stamp = published_sequence << 1u;
292 const Slot &slot = slots_[slot_index];
293 const std::uint64_t begin = slot.stamp.load(std::memory_order_acquire);
294 if(begin != stable_stamp) {
295 continue;
296 }
297
298 detail::load_payload<T>(slot.words, value);
299
300 const std::uint64_t end = slot.stamp.load(std::memory_order_acquire);
301 if(begin == end && published_state_.load(std::memory_order_acquire) == state) {
302 sequence = published_sequence;
303 return IpcStatus::ok;
304 }
305 }
306 return IpcStatus::busy;
307 }

318 static constexpr std::uint64_t pack_state(std::uint64_t sequence, std::size_t slot)
319 {
320 return (sequence << 1u) | static_cast<std::uint64_t>(slot & 1u);
321 }
322

324 {

326 }
327

329 {
330 return static_cast<std::size_t>(state & 1u);
331 }
332

337};
