28 rt::ErrorCode load(const Program &program, const char *configuration_name,
29 unsigned char *buffer, std::size_t buffer_bytes,
30 std::int64_t base_tick_ns)
31 {
32 unload();
33 if(configuration_name == nullptr || buffer == nullptr ||
34 base_tick_ns <= 0 ||
35 (reinterpret_cast<std::uintptr_t>(buffer) & 7U) != 0) {
36 return rt::ErrorCode::invalid_argument;
37 }
38 const ConfigurationInfo *configuration = nullptr;
39 for(const ConfigurationInfo &candidate : program.configurations) {
40 if(tasking_detail::same_name(candidate.lower, configuration_name)) {
41 configuration = &candidate;
42 break;
43 }
44 }
45 if(configuration == nullptr ||
46 configuration->base_tick_ns !=
47 static_cast<std::uint64_t>(base_tick_ns)) {
48 return rt::ErrorCode::invalid_argument;
49 }
50 TaskingReport report;
51 const rt::ErrorCode reported =
52 program.tasking_report(configuration_name, report);
53 if(reported != rt::ErrorCode::ok) return reported;
54 if(report.required_runtime_bytes > buffer_bytes ||
55 report.required_runtime_bytes >
56 std::numeric_limits<std::size_t>::max()) {
57 return rt::ErrorCode::capacity_exceeded;
58 }
59
60 program_ = &program;
61 configuration_ = configuration;
62 base_tick_ns_ = base_tick_ns;
63 resource_count_ = configuration->resources.size();
64 resources_.reset(resource_count_ == 0
65 ? nullptr
66 : new ResourceRuntime[resource_count_]);
67 unsigned char *cursor = buffer;
68 unsigned char *const end = buffer + buffer_bytes;
69
70 for(std::size_t resource_index = 0;
71 resource_index < resource_count_; ++resource_index) {
72 ResourceRuntime &runtime = resources_[resource_index];
73 runtime.info = &configuration->resources[resource_index];
74 runtime.storage = reinterpret_cast<ResourceRuntimeStorage *>(
75 take(cursor, end, sizeof(ResourceRuntimeStorage)));
76 if(runtime.storage == nullptr) return load_failed();
77 new(runtime.storage) ResourceRuntimeStorage{};
78
79 runtime.task_count = runtime.info->tasks.size();
80 runtime.tasks.reset(runtime.task_count == 0
81 ? nullptr
82 : new TaskRuntime[runtime.task_count]);
83 runtime.schedule.reset(runtime.task_count == 0
84 ? nullptr
85 : new std::uint16_t[runtime.task_count]);
86 for(std::size_t task_index = 0; task_index < runtime.task_count;
87 ++task_index) {
88 TaskRuntime &task = runtime.tasks[task_index];
89 task.info = &runtime.info->tasks[task_index];
90 task.storage = reinterpret_cast<TaskRuntimeStorage *>(
91 take(cursor, end, sizeof(TaskRuntimeStorage)));
92 if(task.storage == nullptr) return load_failed();
93 new(task.storage) TaskRuntimeStorage{};
94 runtime.schedule[task_index] =
95 static_cast<std::uint16_t>(task_index);
96 }
97 if(runtime.task_count > 1) {
98 std::sort(runtime.schedule.get(),
99 runtime.schedule.get() + runtime.task_count,
100 [&](std::uint16_t left, std::uint16_t right) {
101 const TaskInfo &a = runtime.info->tasks[left];
102 const TaskInfo &b = runtime.info->tasks[right];
103 return a.priority != b.priority
104 ? a.priority < b.priority
105 : a.declaration_order <
106 b.declaration_order;
107 });
108 }
109
110 runtime.mapping_count = runtime.info->mappings.size();
111 runtime.mappings.reset(runtime.mapping_count == 0
112 ? nullptr
113 : new MappingRuntime[
114 runtime.mapping_count]);
115 for(std::size_t mapping_index = 0;
116 mapping_index < runtime.mapping_count; ++mapping_index) {
117 MappingRuntime &mapping = runtime.mappings[mapping_index];
118 mapping.info = &runtime.info->mappings[mapping_index];
119 mapping.program = find_program(mapping.info->program);
120 if(mapping.program == nullptr) return load_failed();
121 mapping.bytes = mapping.program->required_bytes();
122 mapping.buffer = take(cursor, end, mapping.bytes);
123 if(mapping.buffer == nullptr) return load_failed();
124 }
125
126 if(!build_image_program(runtime)) return load_failed();
127 const std::size_t image_bytes =
128 static_cast<std::size_t>(
129 runtime.image_program.process_image.input_bytes) +
130 runtime.image_program.process_image.output_bytes +
131 runtime.image_program.process_image.memory_bytes;
132 unsigned char *image_buffer = take(cursor, end, image_bytes);
133 if(image_buffer == nullptr ||
134 runtime.image.load(runtime.image_program, image_buffer,
135 image_bytes) != rt::ErrorCode::ok) {
136 return load_failed();
137 }
138 for(std::size_t task_index = 0; task_index < runtime.task_count;
139 ++task_index) {
140 TaskRuntime &task = runtime.tasks[task_index];
141 task.parked_output_bytes =
142 runtime.image_program.process_image.output_bytes;
143 task.parked_memory_bytes =
144 runtime.image_program.process_image.memory_bytes;
145 task.parked_output = take(cursor, end,
146 task.parked_output_bytes);
147 task.parked_output_dirty = take(cursor, end,
148 task.parked_output_bytes);
149 task.parked_memory = take(cursor, end,
150 task.parked_memory_bytes);
151 task.parked_memory_dirty = take(cursor, end,
152 task.parked_memory_bytes);
153 if((task.parked_output_bytes != 0 &&
154 (task.parked_output == nullptr ||
155 task.parked_output_dirty == nullptr)) ||
156 (task.parked_memory_bytes != 0 &&
157 (task.parked_memory == nullptr ||
158 task.parked_memory_dirty == nullptr)))
159 return load_failed();
160 }
161 runtime.write_stamps.assign(
162 (static_cast<std::size_t>(
163 runtime.image_program.process_image.output_bytes) +
164 runtime.image_program.process_image.memory_bytes) * 8U,
165 0);
166 runtime.write_owners.assign(runtime.write_stamps.size(), 0);
167
168 for(std::size_t mapping_index = 0;
169 mapping_index < runtime.mapping_count; ++mapping_index) {
170 MappingRuntime &mapping = runtime.mappings[mapping_index];
171 const std::int64_t period = task_period(runtime, mapping);
172 if(period <= 0 ||
173 mapping.instance.load(*mapping.program, mapping.buffer,
174 mapping.bytes, period,
175 &runtime.image) !=
176 rt::ErrorCode::ok) {
177 return load_failed();
178 }
179 mapping.instance.defer_process_image_publish(true);
180 }
181 }
182 loaded_ = true;
183 return rt::ErrorCode::ok;
184 }

200 rt::ErrorCode boundary(std::uint64_t tick,
201 const EventSample *events = nullptr,
202 std::size_t event_count = 0)
203 {
204 if(!loaded_ || (event_count != 0 && events == nullptr) ||
205 (have_boundary_ && tick <= last_boundary_)) {
206 return rt::ErrorCode::invalid_argument;
207 }
208 have_boundary_ = true;
209 last_boundary_ = tick;
210 for(std::size_t resource_index = 0;
211 resource_index < resource_count_; ++resource_index) {
212 ResourceRuntime &resource = resources_[resource_index];
213 resource.storage->boundary_tick = tick;
214 if(resource.storage->pending_fault != ResourceFault::none) {
215 resource.storage->status.fault =
216 resource.storage->pending_fault;
217 resource.storage->pending_fault = ResourceFault::none;
218 ++resource.storage->status.fault_count;
219 abort_resource(resource);
220 }
221 for(std::size_t task_index = 0;
222 task_index < resource.task_count; ++task_index) {
223 apply_pending(resource, task_index, tick);
224 }
225 if(resource.storage->status.fault != ResourceFault::none) continue;
226 for(std::size_t task_index = 0;
227 task_index < resource.task_count; ++task_index) {
228 release(resource, task_index, tick, events, event_count);
229 }
230 }
231 if(debug_hooks_ != nullptr) debug_hooks_->on_scheduler_boundary();
232 return rt::ErrorCode::ok;
233 }

235 rt::ErrorCode run(std::int64_t slice)
236 {
237 if(!loaded_ || slice < 0) return rt::ErrorCode::invalid_argument;
238 while(slice > 0) {
239 if(debug_hooks_ != nullptr)
240 debug_hooks_->on_scheduler_boundary();
241 ResourceRuntime *resource = nullptr;
242 std::size_t task_index = 0;
243 if(!next_runnable(resource, task_index)) break;
244 TaskRuntime &task = resource->tasks[task_index];
245 if(task.storage->mapping_cursor >= task.info->mappings.size()) {
246 complete_task(*resource, task_index);
247 continue;
248 }
249 if(!task.storage->transaction_active) {
250 if(resource->transaction_owner != invalid_task_owner &&
251 resource->transaction_owner != task_index) {
252 return rt::ErrorCode::invalid_argument;
253 }
254 const std::uint64_t force_release =
255 task.storage->status.release_count;
256 resource->image.begin_transaction(
257 static_cast<std::uint32_t>(task_index), force_release);
258 resource->transaction_owner = task_index;
259 task.storage->transaction_active = true;
260 if(task.parked) {
261 if(!resource->image.restore_transaction(
262 task.parked_output,
263 task.parked_output_dirty,
264 task.parked_output_bytes,
265 task.parked_memory,
266 task.parked_memory_dirty,
267 task.parked_memory_bytes)) {
268 fault_task(*resource, task_index,
269 TaskFault::task_runtime_fault, nullptr);
270 continue;
271 }
272 task.parked = false;
273 }
274 }
275 const std::uint16_t mapping_index =
276 task.info->mappings[task.storage->mapping_cursor];
277 if(mapping_index >= resource->mapping_count) {
278 fault_task(*resource, task_index, TaskFault::task_runtime_fault,
279 nullptr);
280 continue;
281 }
282 MappingRuntime &mapping = resource->mappings[mapping_index];
283 if(!mapping.instance.running()) {
284 const ScanError started =
285 mapping.instance.begin(task.storage->status.remaining_budget);
286 if(started != ScanError::ok) {
287 fault_task(*resource, task_index,
288 started == ScanError::budget_exceeded
289 ? TaskFault::task_budget_exceeded
290 : TaskFault::task_runtime_fault,
291 &mapping);
292 continue;
293 }
294 }
295 std::int64_t executed = 0;
296 const ScanError result = mapping.instance.resume(slice, executed);
297 slice -= executed;
298 task.storage->status.remaining_budget =
299 mapping.instance.remaining_budget();
300 if(result != ScanError::ok) {
301 if(result == ScanError::paused) {
302 task.parked = resource->image.park_transaction(
303 task.parked_output, task.parked_output_dirty,
304 task.parked_output_bytes, task.parked_memory,
305 task.parked_memory_dirty, task.parked_memory_bytes);
306 task.storage->status.state = TaskState::paused;
307 discard_transaction(*resource, task_index);
308 if(debug_hooks_ != nullptr)
309 debug_hooks_->on_debug_event(
310 resource, task_index, DebugEventKind::task,
311 mapping.instance.instruction_offset());
312 continue;
313 }
314 fault_task(*resource, task_index,
315 result == ScanError::budget_exceeded
316 ? TaskFault::task_budget_exceeded
317 : TaskFault::task_runtime_fault,
318 &mapping);
319 continue;
320 }
321 if(mapping.instance.complete()) {
322 ++task.storage->mapping_cursor;
323 if(task.storage->mapping_cursor >= task.info->mappings.size())
324 complete_task(*resource, task_index);
325 }
326 if(executed == 0) break;
327 }
328 return rt::ErrorCode::ok;
329 }

539 static bool same_location(const LocatedVarInfo &left,
540 const LocatedVarInfo &right)
541 {
542 return left.area == right.area &&
543 left.byte_offset == right.byte_offset &&
544 left.bit == right.bit &&
545 left.bit_address == right.bit_address &&
546 left.type_id == right.type_id &&
547 left.byte_width == right.byte_width;
548 }

550 static std::uint64_t physical_begin(const LocatedVarInfo &var)
551 {
552 return static_cast<std::uint64_t>(var.byte_offset) * 8U +
553 (var.bit_address ? var.bit : 0U);
554 }
555
556 static std::uint64_t physical_end(const LocatedVarInfo &var)
557 {
558 return physical_begin(var) +
559 (var.bit_address ? 1U
560 : static_cast<std::uint64_t>(var.byte_width) *
561 8U);
562 }
563
564 static bool overlaps(const LocatedVarInfo &left,
565 const LocatedVarInfo &right)
566 {
567 return left.area == right.area &&
568 physical_begin(left) < physical_end(right) &&
569 physical_begin(right) < physical_end(left);
570 }
571
572 static bool compatible_input_alias(const LocatedVarInfo &left,
573 const LocatedVarInfo &right)
574 {
575 if(left.area != ProcessArea::input ||
576 right.area != ProcessArea::input || !overlaps(left, right)) {
577 return false;
578 }
579 const std::uint64_t left_begin = physical_begin(left);
580 const std::uint64_t left_end = physical_end(left);
581 const std::uint64_t right_begin = physical_begin(right);
582 const std::uint64_t right_end = physical_end(right);
583 if(left_begin == right_begin && left_end == right_end) return true;
584 if(left.bit_address && !right.bit_address)
585 return left_begin >= right_begin && left_begin < right_end;
586 if(!left.bit_address && right.bit_address)
587 return right_begin >= left_begin && right_begin < left_end;
588 return false;
589 }

591 static bool same_initial_value(const Program &program,
592 const LocatedVarInfo &var,
593 const Program &merged_program,
594 const LocatedVarInfo &merged)
595 {
596 const std::size_t source_end =
597 static_cast<std::size_t>(var.var_offset) + var.byte_width;
598 const std::size_t merged_end =
599 static_cast<std::size_t>(merged.var_offset) + merged.byte_width;
600 return source_end <= program.initial_data.size() &&
601 merged_end <= merged_program.initial_data.size() &&
602 std::memcmp(program.initial_data.data() + var.var_offset,
603 merged_program.initial_data.data() +
604 merged.var_offset,
605 var.byte_width) == 0;
606 }

608 bool build_image_program(ResourceRuntime &runtime)
609 {
610 ProcessImageInfo &image = runtime.image_program.process_image;
611 image.max_force_entries = 0;
612 for(std::size_t index = 0; index < runtime.mapping_count; ++index) {
613 const Program &program = *runtime.mappings[index].program;
614 image.input_bytes = std::max(image.input_bytes,
615 program.process_image.input_bytes);
616 image.output_bytes = std::max(image.output_bytes,
617 program.process_image.output_bytes);
618 image.memory_bytes = std::max(image.memory_bytes,
619 program.process_image.memory_bytes);
620 image.max_force_entries = std::max(
621 image.max_force_entries,
622 program.process_image.max_force_entries);
623 image.fingerprint ^= program.process_image.fingerprint;
624 const std::size_t prior_count = image.variables.size();
625 for(const LocatedVarInfo &var : program.process_image.variables) {
626 bool found = false;
627 for(std::size_t prior = 0; prior < prior_count; ++prior) {
628 const LocatedVarInfo &existing = image.variables[prior];
629 if(!overlaps(existing, var)) continue;
630 if(var.area == ProcessArea::input) {
631 if(!compatible_input_alias(existing, var)) return false;
632 if(same_location(existing, var)) {
633 found = true;
634 break;
635 }
636 continue;
637 }
638 const bool same_shared_logical =
639 existing.shared && var.shared &&
640 existing.stable_id == var.stable_id;
641 if(!same_shared_logical ||
642 !same_location(existing, var) ||
643 !same_initial_value(program, var,
644 runtime.image_program, existing)) {
645 return false;
646 }
647 found = true;
648 break;
649 }
650 if(found) continue;
651 if(runtime.image_program.initial_data.size() >
652 std::numeric_limits<std::uint32_t>::max())
653 return false;
654 LocatedVarInfo merged = var;
655 merged.var_offset = static_cast<std::uint32_t>(
656 runtime.image_program.initial_data.size());
657 image.variables.push_back(merged);
658 const std::size_t required =
659 static_cast<std::size_t>(var.var_offset) + var.byte_width;
660 if(required > program.initial_data.size()) return false;
661 runtime.image_program.initial_data.insert(
662 runtime.image_program.initial_data.end(),
663 program.initial_data.begin() + var.var_offset,
664 program.initial_data.begin() +
665 static_cast<std::ptrdiff_t>(required));
666 }
667 }
668 return true;
669 }

671 std::int64_t task_period(const ResourceRuntime &resource,
672 const MappingRuntime &mapping) const
673 {
674 for(const TaskInfo &task : resource.info->tasks) {
675 if(task.lower != mapping.info->task) continue;
676 if(task.kind == TaskKind::event) return base_tick_ns_;
677 if(task.interval_ticks >
678 static_cast<std::uint64_t>(
679 std::numeric_limits<std::int64_t>::max() /
680 base_tick_ns_)) {
681 return 0;
682 }
683 return static_cast<std::int64_t>(task.interval_ticks) *
684 base_tick_ns_;
685 }

689 void apply_pending(ResourceRuntime &resource, std::size_t task_index,
690 std::uint64_t tick)
691 {
692 TaskRuntime &task = resource.tasks[task_index];
693 TaskRuntimeStorage &storage = *task.storage;
694 if(storage.pending_wallclock_fault) {
695 storage.pending_wallclock_fault = false;
696 if(storage.status.state == TaskState::running)
697 fault_task(resource, task_index,
698 TaskFault::task_wallclock_exceeded, nullptr);
699 }
700 if(!storage.pending_reset && !storage.pending_restart) return;
701 const bool restart = storage.pending_restart;
702 storage.pending_reset = false;
703 storage.pending_restart = false;
704 const std::uint64_t releases = storage.status.release_count;
705 const std::uint64_t completed = storage.status.completed_count;
706 const std::uint64_t faults = storage.status.fault_count;
707 discard_transaction(resource, task_index);
708 for(const std::uint16_t mapping_index : task.info->mappings) {
709 if(mapping_index >= resource.mapping_count) continue;
710 MappingRuntime &mapping = resource.mappings[mapping_index];
711 if(restart) {
712 const std::int64_t period = task_period(resource, mapping);
713 (void)mapping.instance.load(*mapping.program, mapping.buffer,
714 mapping.bytes, period,
715 &resource.image);
716 mapping.instance.defer_process_image_publish(true);
717 mapping.instance.set_debug_hooks(debug_hooks_, mapping_index);
718 } else {
719 mapping.instance.reset();
720 }
721 }
722 storage = TaskRuntimeStorage{};

724 storage.status.release_count = releases;
725 storage.status.completed_count = completed;
726 storage.status.fault_count = faults;
727 storage.eligible_tick = tick == std::numeric_limits<std::uint64_t>::max()
728 ? tick
729 : tick + 1;
730 }

732 void release(ResourceRuntime &resource, std::size_t task_index,
733 std::uint64_t tick, const EventSample *events,
734 std::size_t event_count)
735 {
736 TaskRuntime &task = resource.tasks[task_index];
737 TaskRuntimeStorage &storage = *task.storage;
738 bool due = false;
739 if(task.info->kind == TaskKind::periodic) {
740 due = tick >= task.info->phase_ticks &&
741 (tick - task.info->phase_ticks) %
742 task.info->interval_ticks ==
743 0;
744 } else {
745 bool level = false;
746 for(std::size_t index = 0; index < event_count; ++index) {
747 if(tasking_detail::same_name(resource.info->lower,
748 events[index].resource) &&
749 tasking_detail::same_name(task.info->event,
750 events[index].event)) {
751 level = events[index].level;
752 }
753 }
754 due = level && !storage.last_event_level;
755 storage.last_event_level = level;
756 }
757 if(!due || storage.status.state == TaskState::faulted ||
758 storage.status.state == TaskState::paused ||
759 tick < storage.eligible_tick) {
760 return;
761 }
762 if(storage.status.state == TaskState::running) {
763 ++storage.status.missed_release_count;
764 return;
765 }
766 storage.status.state = TaskState::running;
767 storage.status.remaining_budget = task.info->instruction_budget;
768 ++storage.status.release_count;
769 storage.release_tick = tick;
770 storage.mapping_cursor = 0;
771 if(task.info->mappings.empty()) complete_task(resource, task_index);
772 }

774 bool next_runnable(ResourceRuntime *&resource, std::size_t &task_index)
775 {
776 for(std::size_t resource_at = 0; resource_at < resource_count_;
777 ++resource_at) {
778 ResourceRuntime &candidate = resources_[resource_at];
779 if(candidate.storage->status.fault != ResourceFault::none) continue;
780 if(candidate.transaction_owner != invalid_task_owner) {
781 const std::size_t owner = candidate.transaction_owner;
782 if(owner < candidate.task_count &&
783 candidate.tasks[owner].storage->status.state ==
784 TaskState::running) {
785 resource = &candidate;
786 task_index = owner;
787 return true;
788 }
789 candidate.image.discard_scan();
790 candidate.transaction_owner = invalid_task_owner;
791 }
792 for(std::size_t order = 0; order < candidate.task_count; ++order) {
793 const std::size_t index = candidate.schedule[order];
794 if(candidate.tasks[index].storage->status.state ==
795 TaskState::running) {
796 resource = &candidate;
797 task_index = index;
798 return true;
799 }
800 }
801 }
802 return false;

921 void record_writes(ResourceRuntime &resource, std::size_t task_index,
922 std::uint64_t tick)
923 {
924 const std::uint64_t stamp = tick ==
925 std::numeric_limits<
926 std::uint64_t>::max()
927 ? tick
928 : tick + 1;
929 const std::uint16_t owner = static_cast<std::uint16_t>(task_index + 1);
930 bool conflict = false;
931 const std::size_t output_bytes =
932 resource.image_program.process_image.output_bytes;
933 const std::size_t memory_bytes =
934 resource.image_program.process_image.memory_bytes;
935 for(const ProcessArea area : {ProcessArea::output,
936 ProcessArea::memory}) {
937 const std::size_t bytes = area == ProcessArea::output
938 ? output_bytes
939 : memory_bytes;
940 const std::size_t base =
941 area == ProcessArea::output ? 0 : output_bytes;
942 for(std::size_t byte = 0; byte < bytes; ++byte) {
943 const std::uint8_t mask =
944 resource.image.transaction_dirty(area, byte);
945 for(unsigned bit = 0; bit < 8; ++bit) {
946 if((mask & (1U << bit)) == 0) continue;
947 const std::size_t slot = (base + byte) * 8U + bit;
948 if(resource.write_stamps[slot] == stamp &&
949 resource.write_owners[slot] != owner)
950 conflict = true;
951 resource.write_stamps[slot] = stamp;
952 resource.write_owners[slot] = owner;
953 }
954 }
955 }
956 if(conflict) ++resource.storage->status.write_conflict_count;
957 }
