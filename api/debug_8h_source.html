185 DebugError add_watch(SymbolId id)
186 {
187 if(program_ == nullptr) return DebugError::invalid_argument;
188 std::uint8_t expected = watch_plan_open;
189 if(!watch_plan_state_.compare_exchange_strong(
190 expected, watch_plan_editing, std::memory_order_acq_rel,
191 std::memory_order_relaxed))
192 return expected == watch_plan_frozen ||
193 expected == watch_plan_freeze_pending
194 ? DebugError::watch_plan_frozen
195 : DebugError::snapshot_busy;
196 const auto finish = [this](DebugError error) {
197 std::uint8_t editing = watch_plan_editing;
198 if(!watch_plan_state_.compare_exchange_strong(
199 editing, watch_plan_open, std::memory_order_release,
200 std::memory_order_relaxed))
201 watch_plan_state_.store(watch_plan_frozen,
202 std::memory_order_release);
203 return error;
204 };
205 for(const Watch &watch : watches_)
206 if(watch.symbol->id == id) return finish(DebugError::ok);
207 if(watches_.size() >= options_.max_watch_symbols ||
208 watches_.size() >= publish_entry_capacity_)
209 return finish(DebugError::capacity_exceeded);
210 const SymbolInfo *symbol = find_symbol(id);
211 ConfigurationRuntime::MappingRuntime *watch_mapping =
212 symbol == nullptr ? nullptr : find_mapping(*symbol);
213 if(watch_mapping == nullptr)
214 return finish(DebugError::invalid_symbol);
215 const std::size_t word = value_word_count_ + header_words +
216 sfc_state_word_count_;
217 const std::size_t words = (symbol->size + 7U) / 8U;
218 if(word + words > publish_word_capacity_ - breakpoint_word_count_)
219 return finish(DebugError::capacity_exceeded);
220 Watch watch;
221 watch.symbol = symbol;
222 watch.mapping = watch_mapping;
223 watch.word = word;
224 watches_.push_back(watch);
225 DebugSnapshotEntry &entry = publish_entries_[watches_.size() - 1U];
226 entry.symbol_id = id;
227 entry.type_id = symbol->type_id;
228 entry.offset = static_cast<std::uint32_t>(value_bytes_);
229 entry.size = symbol->size;
230 value_word_count_ += words;
231 value_bytes_ += symbol->size;
232 snapshot_ready_.store(false, std::memory_order_release);
233 return finish(DebugError::ok);
234 }

236 DebugError add_breakpoint(const char *source, std::uint32_t line,
237 std::uint32_t column, BreakpointId &id)
238 {
239 if(last_error_ == DebugError::debugging_disabled)
240 return DebugError::debugging_disabled;
241 if(resource_ == nullptr || task_ == nullptr)
242 return DebugError::invalid_argument;
243 bool found = false;
244 id = invalid_instruction;
245 std::vector<std::uint32_t> keys;
246 keys.reserve(task_->info->mappings.size());
247 for(const std::uint16_t mapping : task_->info->mappings) {
248 const SourceMapEntry *entry =
249 find_source(mapping, source, line, column);
250 if(entry == nullptr) continue;
251 const std::uint32_t key = breakpoint_key(mapping, *entry);
252 bool duplicate = false;
253 for(const std::uint32_t candidate : keys)
254 duplicate = duplicate || candidate == key;
255 if(duplicate) continue;
256 keys.push_back(key);
257 if(!found) id = key;
258 found = true;
259 }
260 if(!found) return DebugError::invalid_source_location;
261 std::size_t additions = 0;
262 for(std::size_t index = 0; index < keys.size(); ++index) {
263 bool present = false;
264 for(const std::uint32_t existing : breakpoints_)
265 present = present || existing == keys[index];
266 if(!present) ++additions;
267 }
268 if(additions > options_.max_breakpoints - breakpoints_.size())
269 return DebugError::capacity_exceeded;
270 for(std::size_t index = 0; index < keys.size(); ++index) {
271 bool present = false;
272 for(const std::uint32_t existing : breakpoints_)
273 present = present || existing == keys[index];
274 if(present) continue;
275 breakpoints_.push_back(keys[index]);
276 set_breakpoint(keys[index]);
277 }
278 return DebugError::ok;
279 }

413 DebugError read_snapshot(DebugSnapshot &snapshot,
414 DebugSnapshotEntry *entries,
415 std::size_t entry_capacity,
416 unsigned char *values,
417 std::size_t value_capacity,
418 std::size_t retries) const noexcept
419 {
420 snapshot = {};
421 if(runtime_ == nullptr || entries == nullptr || values == nullptr)
422 return DebugError::invalid_argument;
423 if(entry_capacity < watches_.size() || value_capacity < value_bytes_)
424 return DebugError::capacity_exceeded;
425 if(!snapshot_ready_.load(std::memory_order_acquire))
426 return DebugError::snapshot_busy;
427 if(retries == 0) return DebugError::snapshot_busy;
428 for(std::size_t attempt = 0; attempt < retries; ++attempt) {
429 const std::uint64_t before =
430 publish_words_[0].load(std::memory_order_acquire);
431 if((before & 1U) != 0) continue;
432 const std::uint64_t state_bits =
433 publish_words_[1].load(std::memory_order_acquire);
434 const std::uint64_t fault =
435 publish_words_[2].load(std::memory_order_acquire);
436 const SourceMapEntry *active =
437 published_entry_.load(std::memory_order_relaxed);
438 const std::uint32_t active_mapping = static_cast<std::uint32_t>(
439 publish_words_[4].load(std::memory_order_acquire));
440 const std::uint32_t active_instruction =
441 static_cast<std::uint32_t>(
442 publish_words_[5].load(std::memory_order_acquire));
443 const std::uint8_t state = static_cast<std::uint8_t>(state_bits);
444 std::array<std::uint64_t,
445 (kMaxDebugSfcSteps + 63U) / 64U> sfc_bits{};
446 for(std::size_t word = 0; word < sfc_state_word_count_; ++word)
447 sfc_bits[word] = publish_words_[header_words + word].load(
448 std::memory_order_relaxed);
449 std::size_t destination = 0;
450 for(std::size_t index = 0; index < watches_.size(); ++index) {
451 entries[index] = publish_entries_[index];
452 const Watch &watch = watches_[index];
453 std::size_t remaining = watch.symbol->size;
454 std::size_t word = watch.word;
455 while(remaining != 0) {
456 const std::uint64_t bits =
457 publish_words_[word++].load(std::memory_order_relaxed);
458 const std::size_t bytes = std::min<std::size_t>(8, remaining);
459 std::memcpy(values + destination, &bits, bytes);
460 destination += bytes;
461 remaining -= bytes;
462 }
463 }
464 const std::uint64_t after =
465 publish_words_[0].load(std::memory_order_acquire);
466 if(before != after || (after & 1U) != 0) continue;
467 snapshot.version = after / 2U;
468 snapshot.value_count = watches_.size();
469 snapshot.value_bytes = value_bytes_;
470 snapshot.task_state = static_cast<TaskState>(state);
471 snapshot.fault = static_cast<TaskFault>(fault);
472 if(active != nullptr) {
473 std::string_view path = active->call_path;
474 std::size_t begin = 0;
475 while(begin < path.size() &&
476 snapshot.active_pou_count < kMaxDebugStackFrames) {
477 const std::size_t slash = path.find('/', begin);
478 const std::size_t end = slash == std::string_view::npos
479 ? path.size()
480 : slash;
481 snapshot.call_stack[snapshot.active_pou_count++].pou =
482 path.substr(begin, end - begin);
483 if(slash == std::string_view::npos) break;
484 begin = slash + 1U;
485 }
486 if(snapshot.active_pou_count == 0) {
487 snapshot.call_stack[0].pou = active->pou;
488 snapshot.active_pou_count = 1;
489 }
490 DebugStackFrame &leaf =
491 snapshot.call_stack[snapshot.active_pou_count - 1U];
492 leaf.pou_id = SymbolId{active->pou_id};
493 leaf.source_name = active->source_name;
494 leaf.sfc = active->sfc;
495 leaf.line = active->line;
496 leaf.column = active->column;
497 leaf.instruction_id = active->instruction_id;
498 leaf.instruction_id.mapping = active_mapping;
499 leaf.instruction_id.offset = active_instruction;
500 snapshot.call_depth = static_cast<std::uint16_t>(
501 snapshot.active_pou_count);
502 }
503 for(std::size_t index = 0;
504 index < snapshot_sfc_steps_.size(); ++index) {
505 const SnapshotSfcStep &published = snapshot_sfc_steps_[index];
506 DebugSfcStepState &step =
507 snapshot.sfc_steps[snapshot.sfc_step_count++];
508 step.network_id = published.network_id;
509 step.step_id = published.step_id;
510 step.network = *published.network;
511 step.step = *published.step;
512 step.mapping = published.mapping;
513 step.active =
514 (sfc_bits[index / 64U] &
515 (std::uint64_t{1} << (index % 64U))) != 0;
516 }
517 return DebugError::ok;
518 }
519 return DebugError::snapshot_busy;
520 }

992 void publish_snapshot(std::uint8_t snapshot_kind) noexcept
993 {
994 if(publish_words_ == nullptr || publish_word_capacity_ < header_words)
995 return;
996 const std::uint64_t prior =
997 publish_words_[0].load(std::memory_order_relaxed);
998 publish_words_[0].store(prior | 1U, std::memory_order_release);
999 publish_words_[1].store(
1000 static_cast<std::uint64_t>(task_->storage->status.state) |
1001 (static_cast<std::uint64_t>(snapshot_kind) << 8U),
1002 std::memory_order_relaxed);
1003 publish_words_[2].store(
1004 static_cast<std::uint64_t>(task_->storage->status.fault),
1005 std::memory_order_relaxed);

1007 current_entry_.load(std::memory_order_acquire);
1008 published_entry_.store(active, std::memory_order_relaxed);
1009 publish_words_[4].store(
1010 current_mapping_.load(std::memory_order_relaxed),
1011 std::memory_order_relaxed);
1012 publish_words_[5].store(
1013 current_instruction_.load(std::memory_order_relaxed),
1014 std::memory_order_relaxed);
1015 for(std::size_t word = 0; word < sfc_state_word_count_; ++word) {
1016 std::uint64_t bits = 0;
1017 const std::size_t begin = word * 64U;
1018 const std::size_t end = std::min(
1019 begin + 64U, snapshot_sfc_steps_.size());
1020 for(std::size_t index = begin; index < end; ++index) {
1021 const SnapshotSfcStep &step = snapshot_sfc_steps_[index];
1022 if(step.mapping_runtime->instance
1023 .debug_sfc_committed_step_active(
1024 step.network_index, step.step_index))
1025 bits |= std::uint64_t{1} << (index - begin);
1026 }
1027 publish_words_[header_words + word].store(
1028 bits, std::memory_order_relaxed);
1029 }
1030 for(const Watch &watch : watches_) {
1031 std::size_t remaining = watch.symbol->size;
1032 std::size_t source = 0;
1033 std::size_t word = watch.word;

1035 std::uint64_t bits = 0;
1036 const std::size_t count = std::min<std::size_t>(8, remaining);
1037 if(!watch.mapping->instance.read_variable(
1038 watch.symbol->offset +
1039 static_cast<std::uint32_t>(source),
1040 static_cast<std::uint32_t>(count),
1041 reinterpret_cast<unsigned char *>(&bits)))
1042 break;

1046 }

1049 std::memory_order_release);

1051 }
