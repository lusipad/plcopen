158// Lossless implicit-widening whitelist (approved st-l1a-semantics 3.1):
159// signed chain, unsigned chain, narrow-unsigned to wider-signed,
160// REAL->LREAL, bit-string widening. Everything else is explicit.
161constexpr bool widens_to(Type from, Type to)
162{
163 if(from == to) {
164 return false;
165 }
166 if(is_signed_int(from) && is_signed_int(to)) {
167 return width_bits(from) < width_bits(to);
168 }
169 if(is_unsigned_int(from) && is_unsigned_int(to)) {
170 return width_bits(from) < width_bits(to);
171 }
172 if(is_unsigned_int(from) && is_signed_int(to)) {
173 return width_bits(from) < width_bits(to);
174 }
175 if(from == Type::real && to == Type::lreal) {
176 return true;
177 }
178 if(is_bitstring(from) && is_bitstring(to)) {
179 return width_bits(from) < width_bits(to);
180 }
181 return false;
182}

243// Deterministic double -> modular-u64 reduction for float->integer
244// conversions whose rounded value exceeds 64 bits (matrix 4.2/4.3 wrap
245// policy). fmod on a finite double is exact, so both platforms agree.
246inline std::uint64_t wrap_double_to_u64(double value)
247{
248 constexpr double kTwo64 = 18446744073709551616.0;
249 constexpr double kTwo63 = 9223372036854775808.0;
250 // fmod on a finite double is exact; the negative branch negates in the
251 // integer domain because m + 2^64 is NOT exact in double (spacing near
252 // 2^64 is 4096 -- small negatives would collapse to 2^64 and wrap to 0).
253 double m = std::fmod(value, kTwo64);
254 const bool negative = m < 0.0;
255 if(negative) {
256 m = -m;
257 }
258 std::uint64_t magnitude = 0;
259 if(m >= kTwo63) {
260 magnitude = static_cast<std::uint64_t>(m - kTwo63) +
261 0x8000000000000000ULL;
262 } else {
263 magnitude = static_cast<std::uint64_t>(m);
264 }
265 return negative ? (0ULL - magnitude) : magnitude;
266}

268// Canonical slot form (approved st-l1a-semantics 2.4): signed values are
269// sign-extended int64, unsigned and bit-string values zero-extended uint64.
270// Re-canonicalizing after every arithmetic step is what makes each width's
271// wrap semantics (matrix 1.8/4.2) hold, and what makes every whitelist
272// widening a runtime no-op.
273inline std::uint64_t canon(Type type, std::uint64_t raw)
274{
275 switch(type) {
276 case Type::bool_:
277 return raw ? 1 : 0;
278 case Type::sint:
279 return static_cast<std::uint64_t>(wrap8(static_cast<std::int64_t>(raw)));
280 case Type::int_:
281 return static_cast<std::uint64_t>(wrap16(static_cast<std::int64_t>(raw)));
282 case Type::dint:
283 return static_cast<std::uint64_t>(wrap32(static_cast<std::int64_t>(raw)));
284 case Type::usint:
285 case Type::byte_:
286 return raw & 0xFFULL;
287 case Type::uint_:
288 case Type::word:
289 return raw & 0xFFFFULL;
290 case Type::udint:
291 case Type::dword:
292 return raw & 0xFFFFFFFFULL;
293 default:
294 // lint / lint-width families / time: 64-bit native.
295 return raw;
296 }
297}
