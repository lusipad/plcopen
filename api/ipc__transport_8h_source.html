164inline ServoSetpointWire encode_setpoint(const ServoSetpoints &source)
165{
166 ServoSetpointWire wire{};
167 wire.position = source.position;
168 wire.velocity = source.velocity;
169 wire.acceleration = source.acceleration;
170 wire.torque = source.torque;
171 wire.torque_limit = source.torque_limit;
172 wire.torque_velocity_limit = source.torque_velocity_limit;
173 wire.torque_acceleration_limit = source.torque_acceleration_limit;
174 wire.torque_deceleration_limit = source.torque_deceleration_limit;
175 wire.torque_jerk_limit = source.torque_jerk_limit;
176 switch(source.torque_direction) {
177 case axis::Direction::current:
178 wire.torque_direction = 0;
179 break;
180 case axis::Direction::positive:
181 wire.torque_direction = 1;
182 break;
183 case axis::Direction::negative:
184 wire.torque_direction = 2;
185 break;
186 case axis::Direction::shortest_way:
187 wire.torque_direction = 3;
188 break;
189 }
190 wire.torque_mode = source.torque_mode ? 1U : 0U;
191 return wire;
192}

194inline void decode_setpoint(const ServoSetpointWire &wire, ServoSetpoints &target)
195{
196 axis::Direction direction = axis::Direction::current;
197 switch(wire.torque_direction) {
198 case 0:
199 direction = axis::Direction::current;
200 break;
201 case 1:
202 direction = axis::Direction::positive;
203 break;
204 case 2:
205 direction = axis::Direction::negative;
206 break;
207 case 3:
208 direction = axis::Direction::shortest_way;
209 break;
210 default:
211 break;
212 }
213 target.position = wire.position;
214 target.velocity = wire.velocity;
215 target.acceleration = wire.acceleration;
216 target.torque = wire.torque;
217 target.torque_limit = wire.torque_limit;
218 target.torque_velocity_limit = wire.torque_velocity_limit;
219 target.torque_acceleration_limit = wire.torque_acceleration_limit;
220 target.torque_deceleration_limit = wire.torque_deceleration_limit;
221 target.torque_jerk_limit = wire.torque_jerk_limit;
222 target.torque_direction = direction;
223 target.torque_mode = wire.torque_mode != 0U;
224}

226inline ServoFeedbackWire encode_feedback_value(const ServoFeedback &source)
227{
228 ServoFeedbackWire wire{};
229 wire.position = source.position;
230 wire.velocity = source.velocity;
231 wire.acceleration = source.acceleration;
232 wire.torque = source.torque;
233 for(std::size_t index = 0; index < ServoFeedback::DigitalInputCount; ++index) {
234 if(source.digital_inputs[index]) {
235 wire.digital_inputs |= 1U << index;
236 }
237 }
238 if(source.info.communication_ready)
239 wire.axis_info |= CommunicationReady;
240 if(source.info.ready_for_power_on)
241 wire.axis_info |= ReadyForPowerOn;
242 if(source.info.home_abs_switch)
243 wire.axis_info |= HomeAbsSwitch;
244 if(source.info.limit_switch_pos)
245 wire.axis_info |= LimitSwitchPos;
246 if(source.info.limit_switch_neg)
247 wire.axis_info |= LimitSwitchNeg;
248 if(source.info.warning)
249 wire.axis_info |= Warning;
250 return wire;
251}

253inline void decode_feedback_value(const ServoFeedbackWire &wire, ServoFeedback &target)
254{
255 target.position = wire.position;
256 target.velocity = wire.velocity;
257 target.acceleration = wire.acceleration;
258 target.torque = wire.torque;
259 for(std::size_t index = 0; index < ServoFeedback::DigitalInputCount; ++index) {
260 target.digital_inputs[index] = (wire.digital_inputs & (1U << index)) != 0U;
261 }
262 target.info.communication_ready = (wire.axis_info & CommunicationReady) != 0U;
263 target.info.ready_for_power_on = (wire.axis_info & ReadyForPowerOn) != 0U;
264 target.info.home_abs_switch = (wire.axis_info & HomeAbsSwitch) != 0U;
265 target.info.limit_switch_pos = (wire.axis_info & LimitSwitchPos) != 0U;
266 target.info.limit_switch_neg = (wire.axis_info & LimitSwitchNeg) != 0U;
267 target.info.warning = (wire.axis_info & Warning) != 0U;
268}

272inline bool encode_setpoints(std::uint64_t tick, const ServoSetpoints *values, std::size_t count,
273 ServoSetpointFrame &frame)
274{
275 if(values == nullptr || count == 0 || count > MaxAxes)
276 return false;
277 frame = ServoSetpointFrame{};
278 frame.tick = tick;
279 frame.axis_count = static_cast<std::uint32_t>(count);
280 for(std::size_t index = 0; index < count; ++index) {
281 if(!detail::valid_direction(values[index].torque_direction) ||
282 !detail::valid_setpoint_source(values[index])) {
283 return false;
284 }
285 const ServoSetpointWire wire = detail::encode_setpoint(values[index]);
286 if(!detail::valid_setpoint_wire(wire))
287 return false;
288 frame.axes[index] = wire;
289 }
290 return true;
291}
292
293inline bool decode_setpoints(const ServoSetpointFrame &frame, ServoSetpoints *values,
294 std::size_t count, std::uint64_t &tick)
295{
296 if(values == nullptr || frame.axis_count == 0 || frame.axis_count > MaxAxes ||
297 count != frame.axis_count || frame.reserved != 0U) {
298 return false;
299 }
300 for(std::size_t index = 0; index < count; ++index)
301 if(!detail::valid_setpoint_wire(frame.axes[index]))
302 return false;
303 for(std::size_t index = 0; index < count; ++index)
304 detail::decode_setpoint(frame.axes[index], values[index]);
305 tick = frame.tick;
306 return true;
307}

309inline bool encode_feedback(std::uint64_t tick, const ServoFeedback *values, std::size_t count,
310 ServoFeedbackFrame &frame)
311{
312 if(values == nullptr || count == 0 || count > MaxAxes)
313 return false;
314 frame = ServoFeedbackFrame{};
315 frame.tick = tick;
316 frame.axis_count = static_cast<std::uint32_t>(count);
317 for(std::size_t index = 0; index < count; ++index) {
318 if(!detail::valid_feedback_source(values[index]))
319 return false;
320 const ServoFeedbackWire wire = detail::encode_feedback_value(values[index]);
321 if(!detail::valid_feedback_wire(wire))
322 return false;
323 frame.axes[index] = wire;
324 }
325 return true;
326}

328inline bool decode_feedback(const ServoFeedbackFrame &frame, ServoFeedback *values,
329 std::size_t count, std::uint64_t &tick)
330{
331 if(values == nullptr || frame.axis_count == 0 || frame.axis_count > MaxAxes ||
332 count != frame.axis_count || frame.reserved != 0U) {
333 return false;
334 }
335 for(std::size_t index = 0; index < count; ++index)
336 if(!detail::valid_feedback_wire(frame.axes[index]))
337 return false;
338 for(std::size_t index = 0; index < count; ++index)
339 detail::decode_feedback_value(frame.axes[index], values[index]);
340 tick = frame.tick;
341 return true;

394template <std::size_t Depth = 8> struct ServoIpcRegion
395{
396 static_assert(Depth > 0, "X5 transport depth must be positive");
397
398 TransportHeader header{};
399 rt::IpcSpscRing<ServoSetpointFrame, Depth> setpoints{};
400 rt::IpcSpscRing<ServoFeedbackFrame, Depth> feedback{};
401 rt::IpcDoubleBuffer<TransportStatusSnapshot> status{};

403 bool initialize(std::size_t axis_count, std::uint64_t period_ns, std::uint64_t generation)
404 {
405 if(axis_count == 0 || axis_count > MaxAxes || period_ns == 0 || generation == 0) {
406 return false;
407 }
408 header.state.store(static_cast<std::uint32_t>(TransportState::initializing),
409 std::memory_order_relaxed);
410 header.magic = TransportMagic;
411 header.abi_version = TransportAbiVersion;
412 header.header_bytes = static_cast<std::uint32_t>(sizeof(TransportHeader));
413 header.region_bytes = static_cast<std::uint32_t>(sizeof(*this));
414 header.max_axes = static_cast<std::uint32_t>(MaxAxes);
415 header.axis_count = static_cast<std::uint32_t>(axis_count);
416 header.transport_depth = static_cast<std::uint32_t>(Depth);
417 header.setpoint_frame_bytes = static_cast<std::uint32_t>(sizeof(ServoSetpointFrame));
418 header.feedback_frame_bytes = static_cast<std::uint32_t>(sizeof(ServoFeedbackFrame));
419 header.status_bytes = static_cast<std::uint32_t>(sizeof(TransportStatusSnapshot));
420 header.reserved0 = 0;
421 header.reserved1 = 0;
422 header.generation = generation;
423 header.period_ns = period_ns;
424 header.last_error.store(static_cast<std::uint32_t>(TransportError::none),
425 std::memory_order_relaxed);
426 header.executor_heartbeat.store(0, std::memory_order_relaxed);
427 header.bus_heartbeat.store(0, std::memory_order_relaxed);
428 header.phase_tick.store(0, std::memory_order_relaxed);
429 setpoints.initialize();
430 feedback.initialize();
431 status.initialize();
432 header.state.store(static_cast<std::uint32_t>(TransportState::priming),
433 std::memory_order_release);
434 return true;
435 }

437 AttachStatus attach_status() const
438 {
439 const std::uint32_t raw_state = header.state.load(std::memory_order_acquire);
440 if(raw_state == static_cast<std::uint32_t>(TransportState::uninitialized) ||
441 raw_state == static_cast<std::uint32_t>(TransportState::initializing)) {
442 return AttachStatus::not_initialized;
443 }
444 if(raw_state > static_cast<std::uint32_t>(TransportState::faulted)) {
445 return AttachStatus::layout_mismatch;
446 }
447 if(header.magic != TransportMagic)
448 return AttachStatus::invalid_magic;
449 if(header.abi_version != TransportAbiVersion) {
450 return AttachStatus::version_mismatch;
451 }
452 if(header.header_bytes != sizeof(TransportHeader) || header.region_bytes != sizeof(*this) ||
453 header.max_axes != MaxAxes || header.transport_depth != Depth ||
454 header.setpoint_frame_bytes != sizeof(ServoSetpointFrame) ||
455 header.feedback_frame_bytes != sizeof(ServoFeedbackFrame) ||
456 header.status_bytes != sizeof(TransportStatusSnapshot) || header.period_ns == 0 ||
457 header.generation == 0) {
458 return AttachStatus::layout_mismatch;
459 }
460 if(header.axis_count == 0 || header.axis_count > MaxAxes) {
461 return AttachStatus::invalid_axis_count;
462 }
463 return AttachStatus::ok;
464 }

466 TransportState state() const
467 {
468 return static_cast<TransportState>(header.state.load(std::memory_order_acquire));
469 }
470
471 bool transition(TransportState expected, TransportState desired)
472 {
473 if((expected == TransportState::priming && desired == TransportState::running) ||
474 (expected == TransportState::running && desired == TransportState::draining) ||
475 (expected == TransportState::draining && desired == TransportState::stopped)) {
476 std::uint32_t raw = static_cast<std::uint32_t>(expected);
477 return header.state.compare_exchange_strong(raw, static_cast<std::uint32_t>(desired),
478 std::memory_order_acq_rel,
479 std::memory_order_acquire);
480 }
481 return false;
482 }
