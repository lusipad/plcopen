137// A4 corner blend: symmetric quintic Bezier with collinear control triples
138// (P0,P1,P2 on the incoming line, P3,P4,P5 on the outgoing line), which makes
139// the curve C2 against straight lines: tangent along the lines and zero
140// curvature at both junctions. Sampling runs through an embedded arc-length
141// table so the spatial speed stays continuous across the junctions.
142struct QuinticBlendSegment

148 Vec3 p3{};
149 Vec3 p4{};
150 Vec3 p5{};
151 double length = 0.0;
152 double max_deviation = 0.0;
153 double max_curvature = 0.0;
154 double cumulative[TableSize] = {};
155};

162inline double normalize_sweep(double sweep, double orientation)
163{
164 constexpr double Tau = 6.28318530717958647692;
165 if(orientation >= 0.0) {
166 while(sweep < 0.0) {
167 sweep += Tau;
168 }
169 } else {
170 while(sweep > 0.0) {
171 sweep -= Tau;
172 }
173 }
174 return sweep;
175}

177inline rt::Result<ArcSegment> make_arc(Vec3 start, Vec3 via, Vec3 finish)
178{
179 // Solve the circumcenter in centroid-relative coordinates: the |p|^2
180 // terms then cancel on the triangle's own scale instead of against the
181 // absolute origin, so the fit stays conditioned at machine offsets
182 // (PCS translation, tool offsets). Mathematically identical to the
183 // absolute-coordinate determinant form.
184 const double gx = (start.x + via.x + finish.x) / 3.0;
185 const double gy = (start.y + via.y + finish.y) / 3.0;
186 const double ax = start.x - gx;
187 const double ay = start.y - gy;
188 const double bx = via.x - gx;
189 const double by = via.y - gy;
190 const double cx = finish.x - gx;
191 const double cy = finish.y - gy;
192 const double d = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by));
193
194 if(!std::isfinite(d) || std::fabs(d) <= 1e-12) {
195 return rt::Result<ArcSegment>::failure(rt::ErrorCode::invalid_argument);
196 }
197
198 const double a2 = ax * ax + ay * ay;
199 const double b2 = bx * bx + by * by;
200 const double c2 = cx * cx + cy * cy;
201 const Vec3 center{
202 (a2 * (by - cy) + b2 * (cy - ay) + c2 * (ay - by)) / d + gx,
203 (a2 * (cx - bx) + b2 * (ax - cx) + c2 * (bx - ax)) / d + gy,
204 0.0,
205 };
206 const double radius = norm({start.x - center.x, start.y - center.y, 0.0});
207 if(!std::isfinite(radius) || radius <= 1e-12) {
208 return rt::Result<ArcSegment>::failure(rt::ErrorCode::invalid_argument);
209 }
210
211 const double orientation =
212 (via.x - start.x) * (finish.y - via.y) - (via.y - start.y) * (finish.x - via.x);
213 const double start_angle = angle_of(start, center);
214 const double raw_sweep = angle_of(finish, center) - start_angle;
215 const double sweep = normalize_sweep(raw_sweep, orientation);
216 if(std::fabs(sweep) <= 1e-12) {
217 return rt::Result<ArcSegment>::failure(rt::ErrorCode::invalid_argument);
218 }
219
220 ArcSegment arc{};
221 arc.start = start;
222 arc.via = via;
223 arc.finish = finish;
224 arc.center = center;
225 arc.radius = radius;
226 arc.start_angle = start_angle;
227 arc.sweep = sweep;
228 arc.length = std::fabs(sweep) * radius;
229 return rt::Result<ArcSegment>::success(arc);
230}

297template <typename SampleFn> double approximate_length(SampleFn sample_fn)
298{
299 constexpr int Samples = 32;
300 double length = 0.0;
301 Vec3 previous = sample_fn(0.0);
302 for(int i = 1; i <= Samples; ++i) {
303 const double u = static_cast<double>(i) / Samples;
304 const Vec3 current = sample_fn(u);
305 length += norm(current - previous);
306 previous = current;
307 }
308 return length;
309}

365 Vec3 control,

367 double tolerance)
368{

370 return rt::Result<QuadraticBlendSegment>::failure(rt::ErrorCode::invalid_argument);
371 }
372
373 const double length =
374 approximate_length([&](double u) { return quadratic_point(start, control, finish, u); });
375 const Vec3 chord = finish - start;

378 return rt::Result<QuadraticBlendSegment>::failure(rt::ErrorCode::invalid_argument);

382 norm(control - (start + chord * (dot(control - start, chord) / dot(chord, chord)))) * 0.5;
383 if(deviation > tolerance + 1e-12) {
384 return rt::Result<QuadraticBlendSegment>::failure(rt::ErrorCode::out_of_range);
385 }
386
387 return rt::Result<QuadraticBlendSegment>::success({start, control, finish, length, deviation});

425inline Vec3 quintic_point(const QuinticBlendSegment &blend, double u)
426{
427 const double v = 1.0 - u;
428 const double v2 = v * v;
429 const double u2 = u * u;
430 return blend.p0 * (v2 * v2 * v) + blend.p1 * (5.0 * v2 * v2 * u) +
431 blend.p2 * (10.0 * v2 * v * u2) + blend.p3 * (10.0 * v2 * u2 * u) +
432 blend.p4 * (5.0 * v * u2 * u2) + blend.p5 * (u2 * u2 * u);
433}

435inline Vec3 quintic_derivative(const QuinticBlendSegment &blend, double u)
436{
437 const double v = 1.0 - u;
438 const double v2 = v * v;
439 const double u2 = u * u;
440 return (blend.p1 - blend.p0) * (5.0 * v2 * v2) +
441 (blend.p2 - blend.p1) * (20.0 * v2 * v * u) +
442 (blend.p3 - blend.p2) * (30.0 * v2 * u2) +
443 (blend.p4 - blend.p3) * (20.0 * v * u2 * u) +
444 (blend.p5 - blend.p4) * (5.0 * u2 * u2);

447inline Vec3 quintic_second_derivative(const QuinticBlendSegment &blend, double u)
448{
449 const double v = 1.0 - u;
450 const Vec3 d0 = blend.p2 - blend.p1 * 2.0 + blend.p0;

474// start/finish are the truncated junction points on the adjacent segments;
475// corner is the original path corner. The maximum deviation of the symmetric
476// uniform-thirds construction is closed-form: |B(1/2) - corner| =
477// (23/96)|(finish-corner) - (corner-start)|.
478inline rt::Result<QuinticBlendSegment> make_quintic_blend(Vec3 start,
479 Vec3 corner,
480 Vec3 finish,
481 double tolerance)
482{
483 if(tolerance <= 0.0 || !std::isfinite(tolerance)) {
484 return rt::Result<QuinticBlendSegment>::failure(rt::ErrorCode::invalid_argument);
485 }
486 const Vec3 in = corner - start;
487 const Vec3 out = finish - corner;
488 if(norm(in) <= 1e-12 || norm(out) <= 1e-12) {
489 return rt::Result<QuinticBlendSegment>::failure(rt::ErrorCode::invalid_argument);
490 }
491
492 QuinticBlendSegment blend{};
493 blend.p0 = start;
494 blend.p1 = corner - in * (2.0 / 3.0);
495 blend.p2 = corner - in * (1.0 / 3.0);
496 blend.p3 = corner + out * (1.0 / 3.0);
497 blend.p4 = corner + out * (2.0 / 3.0);
498 blend.p5 = finish;
499
500 blend.max_deviation = norm(quintic_point(blend, 0.5) - corner);
501 if(blend.max_deviation > tolerance + 1e-12) {
502 return rt::Result<QuinticBlendSegment>::failure(rt::ErrorCode::out_of_range);
503 }
504
505 // Arc-length table (planning-phase work) for junction-continuous sampling.
506 double accumulated = 0.0;
507 Vec3 previous = quintic_point(blend, 0.0);
508 blend.cumulative[0] = 0.0;
509 for(std::size_t i = 1; i < QuinticBlendSegment::TableSize; ++i) {
510 const double u =
511 static_cast<double>(i) / static_cast<double>(QuinticBlendSegment::TableSize - 1);
512 const Vec3 current = quintic_point(blend, u);
513 accumulated += norm(current - previous);
514 blend.cumulative[i] = accumulated;
515 previous = current;
516 }
517 blend.length = accumulated;
518 if(!std::isfinite(blend.length) || blend.length <= 1e-12) {
519 return rt::Result<QuinticBlendSegment>::failure(rt::ErrorCode::invalid_argument);
520 }
521
522 double max_curvature = 0.0;
523 for(std::size_t i = 0; i <= 64; ++i) {
524 const double u = static_cast<double>(i) / 64.0;
525 const Vec3 d1 = quintic_derivative(blend, u);
526 const Vec3 d2 = quintic_second_derivative(blend, u);
527 const double speed = norm(d1);
528 if(speed <= 1e-12) {
529 continue;
530 }
531 const double curvature = norm(cross(d1, d2)) / (speed * speed * speed);
532 if(curvature > max_curvature) {
533 max_curvature = curvature;
534 }
535 }
536 blend.max_curvature = max_curvature;
537 return rt::Result<QuinticBlendSegment>::success(blend);
538}

540inline double quintic_parameter_at_length(const QuinticBlendSegment &blend, double arclength)
541{
542 const double target = clamp_arclength(arclength, blend.length);
543 constexpr std::size_t Last = QuinticBlendSegment::TableSize - 1;
544 std::size_t low = 0;
545 for(std::size_t i = 1; i <= Last; ++i) {
546 if(blend.cumulative[i] >= target) {
547 low = i - 1;
548 break;
549 }
550 low = i - 1;
551 }
552 const double segment = blend.cumulative[low + 1] - blend.cumulative[low];
553 const double fraction = segment > 0.0 ? (target - blend.cumulative[low]) / segment : 0.0;
554 return (static_cast<double>(low) + fraction) / static_cast<double>(Last);
555}

578inline Vec3 path_second_derivative(const QuinticBlendSegment &blend, double arclength)
579{
580 const double u = quintic_parameter_at_length(blend, arclength);
581 const Vec3 d1 = quintic_derivative(blend, u);
582 const Vec3 d2 = quintic_second_derivative(blend, u);
583 const double sigma2 = dot(d1, d1);
584 if(sigma2 <= 1e-30) {
585 return {0.0, 0.0, 0.0};
586 }
587 return (d2 - d1 * (dot(d2, d1) / sigma2)) * (1.0 / sigma2);
588}

590inline Vec3 path_third_derivative(const QuinticBlendSegment &blend, double arclength)
591{
592 const double u = quintic_parameter_at_length(blend, arclength);
593 const Vec3 d1 = quintic_derivative(blend, u);
594 const Vec3 d2 = quintic_second_derivative(blend, u);
595 const Vec3 d3 = quintic_third_derivative(blend, u);
596 const double sigma2 = dot(d1, d1);
597 if(sigma2 <= 1e-30) {
598 return {0.0, 0.0, 0.0};
599 }
600 const double alpha = dot(d1, d2);
601 const double beta = dot(d2, d2) + dot(d1, d3);
602 const double sigma5 = sigma2 * sigma2 * std::sqrt(sigma2);
603 return (d3 * sigma2 - d2 * (3.0 * alpha) + d1 * (4.0 * alpha * alpha / sigma2 - beta)) *
604 (1.0 / sigma5);

767template <std::size_t Capacity> class ArcLengthTable
768{
769public:
770 rt::ErrorCode build(const PathSegment &segment)
771 {
772 lengths_.clear();
773 for(std::size_t i = 0; i < Capacity; ++i) {
774 const double ratio = static_cast<double>(i) / static_cast<double>(Capacity - 1);
775 const rt::ErrorCode pushed = lengths_.push_back(segment.length() * ratio);
776 if(pushed != rt::ErrorCode::ok) {
777 return pushed;
778 }
779 }
780 return rt::ErrorCode::ok;
781 }
782
783 std::size_t size() const
784 {
785 return lengths_.size();
786 }
787
788 double total_length() const
789 {
790 return lengths_.empty() ? 0.0 : lengths_[lengths_.size() - 1];
791 }
792
793 double parameter_at_length(double arclength) const
794 {
795 if(lengths_.empty() || total_length() <= 0.0) {
796 return 0.0;
797 }
798 return clamp_arclength(arclength, total_length()) / total_length();
799 }
800
801private:
802 static_assert(Capacity >= 2, "ArcLengthTable capacity must be at least two");
803 rt::StaticVector<double, Capacity> lengths_{};
804};
