164inline void append_unsigned(std::string &out, std::uint64_t value)
165{
166 char digits[20]{};
167 std::size_t count = 0;
168 do {
169 digits[count++] = static_cast<char>('0' + value % 10U);
170 value /= 10U;
171 } while(value != 0U);
172 while(count != 0U) out.push_back(digits[--count]);
173}

205 std::string canonical_dump() const
206 {
207 std::string out;
208 for(const BindingFbDesc &fb_desc : fbs_) {
209 out += "fb ";
210 binding_manifest_detail::append_unsigned(
211 out, static_cast<std::uint32_t>(fb_desc.id));
212 out += " set=";
213 binding_manifest_detail::append_unsigned(
214 out, static_cast<std::uint8_t>(fb_desc.set));
215 out += " name=";
216 out.append(fb_desc.lower_name.data(), fb_desc.lower_name.size());
217 out += " source=";
218 out.append(fb_desc.source.data(), fb_desc.source.size());
219 out += " declared=";
220 out += fb_desc.declared ? '1' : '0';
221 out += " generated=";
222 out += fb_desc.generated ? '1' : '0';
223 out += " excluded=";
224 out += fb_desc.excluded ? '1' : '0';
225 out += " scope=";
226 out.append(fb_desc.scope_reason.data(), fb_desc.scope_reason.size());
227 out += " rejection=";
228 out.append(fb_desc.rejection_test.data(),
229 fb_desc.rejection_test.size());
230 out += '\n';
231 for(std::size_t index = 0; index < fb_desc.pin_count; ++index) {
232 const BindingPinDesc &pin = fb_desc.pins[index];
233 out += "pin ";
234 binding_manifest_detail::append_unsigned(out, pin.id);
235 out += " name=";
236 out.append(pin.lower_name.data(), pin.lower_name.size());
237 out += " direction=";
238 binding_manifest_detail::append_unsigned(
239 out, static_cast<std::uint8_t>(pin.direction));
240 out += " type=";
241 binding_manifest_detail::append_unsigned(out, pin.type_id);
242 out += " declared=";
243 out += pin.declared ? '1' : '0';
244 out += " generated=";
245 out += pin.generated ? '1' : '0';
246 out += " excluded=";
247 out += pin.excluded ? '1' : '0';
248 out += " scope=";
249 out.append(pin.scope_reason.data(), pin.scope_reason.size());
250 out += " rejection=";
251 out.append(pin.rejection_test.data(),
252 pin.rejection_test.size());
253 out += '\n';
254 }
255 }
256 for(const BindingTargetDesc &target_desc : targets_) {
257 out += "target ";
258 binding_manifest_detail::append_unsigned(
259 out, static_cast<std::uint8_t>(target_desc.kind));
260 out += " type=";
261 binding_manifest_detail::append_unsigned(out, target_desc.type_id);
262 out += " bindable=";
263 out += target_desc.bindable ? '1' : '0';
264 out += '\n';
265 }
266 return out;
267 }

274 BindingManifest()
275 {
276 (void)install_binding_types(types_);
277 for(std::size_t fb_index = 0;
278 fb_index < generated::kStBindingFbs.size(); ++fb_index) {
279 const generated::StBindingFbMetadata &source =
280 generated::kStBindingFbs[fb_index];
281 for(std::size_t pin_index = 0; pin_index < source.pin_count;
282 ++pin_index) {
283 const generated::StBindingPinMetadata &pin =
284 generated::kStBindingPins[source.first_pin + pin_index];
285 pins_[fb_index][pin_index] = {
286 pin.stable_id, pin.name,
287 binding_manifest_detail::pin_direction(pin.direction),
288 binding_manifest_detail::pin_type_id(pin.st_type),
289 true, true, false, {}, {}};
290 }
291 fbs_[fb_index] = {
292 static_cast<BindingFbId>(source.stable_id),
293 binding_manifest_detail::binding_set(source.set),
294 source.parser_key, source.source, true, true, false, {}, {},
295 pins_[fb_index].data(), source.pin_count};
296 }
297 }

303 std::array<BindingTargetDesc, 2> targets_{{
304 {BindingTargetKind::axis, binding_type::axis_ref, true},
305 {BindingTargetKind::group, binding_type::group_ref, true},
306 }};
307};
