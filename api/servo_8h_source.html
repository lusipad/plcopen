3// ADR-0004 (Accepted 2026-07-05): the stable narrow hardware interface for
4// Phase B5 (plcopen-fieldbus) and B7 (RT reference executor). The core L5
5// never holds a Servo pointer — an outer executor calls the interface at the
6// cycle boundary and bridges feedback into the existing AxisModel hooks and
7// snapshots into write_setpoints. Adapters are outer-ring composition, not a
8// kernel dependency: the core keeps value semantics, deterministic replay,
9// and zero OS contact.
10//
11// Interface contract: no allocation, no exceptions, bounded time. The
12// executor owns threads and clocks. CSP/CSV/CST drive-side mode semantics
13// extend this interface with B5 (real-drive context); the mode manager
14// skeleton lives in mode_manager.h.

63inline ServoSetpoints make_setpoints(const axis::AxisSnapshot &snapshot)
64{

80inline void bridge_feedback(axis::AxisModel &axis, const ServoFeedback &feedback)
81{
82 axis.set_actual_feedback(feedback.position, feedback.velocity, feedback.acceleration,
83 feedback.torque);
84 for (std::size_t i = 0; i < ServoFeedback::DigitalInputCount; ++i)
85 {
86 axis.set_digital_input(i, feedback.digital_inputs[i]);
87 }
88 axis.set_axis_info_inputs(feedback.info);
