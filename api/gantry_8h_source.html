40 rt::ErrorCode forward(const double *joints,
41 std::size_t joint_count,
42 geom::Vec3 &cartesian) const override
43 {
44 if(joint_count != count_) {
45 return rt::ErrorCode::invalid_argument;
46 }
47 const double x = scale_[0] * joints[0] + offset_[0];
48 const double y = count_ > 1 ? scale_[1] * joints[1] + offset_[1] : 0.0;
49 const double z = count_ > 2 ? scale_[2] * joints[2] + offset_[2] : 0.0;
50 cartesian = geom::Vec3{x, y, z};
51 return rt::ErrorCode::ok;
52 }

54 rt::ErrorCode inverse(geom::Vec3 cartesian,
55 const double *,
56 std::size_t joint_count,
57 double *joints_out) const override
58 {
59 if(joint_count != count_) {
60 return rt::ErrorCode::invalid_argument;
61 }
62 const double coordinates[3] = {cartesian.x, cartesian.y, cartesian.z};
63 for(std::size_t i = 0; i < count_; ++i) {
64 if(scale_[i] == 0.0 || !std::isfinite(coordinates[i])) {
65 return rt::ErrorCode::invalid_argument;
66 }
67 joints_out[i] = (coordinates[i] - offset_[i]) / scale_[i];
68 }
69 return rt::ErrorCode::ok;
70 }
71
72 double singularity_margin(const double *, std::size_t) const override
73 {
74 return 1e30; // linear mechanism: no singular configurations
75 }
76
77private:
78 std::size_t count_ = 0;
79 double scale_[3] = {1.0, 1.0, 1.0};
80 double offset_[3] = {0.0, 0.0, 0.0};
81};
