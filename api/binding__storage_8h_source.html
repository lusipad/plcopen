63constexpr BindingStorageKind binding_storage_kind(Type type, TypeId type_id)
64{
65 if(type == Type::axis_ref || type_id == binding_type::axis_ref ||
66 type_id == binding_type::mc_input_ref ||
67 type_id == binding_type::mc_output_ref) {
68 return BindingStorageKind::axis_targets;
69 }
70 if(type == Type::group_ref || type_id == binding_type::group_ref) {
71 return BindingStorageKind::group_targets;
72 }
73 switch(type_id) {
74 case binding_type::mc_path_table:
75 return BindingStorageKind::path_tables;
76 case binding_type::mc_path_description:
77 return BindingStorageKind::path_descriptions;
78 case binding_type::mc_cam_switch_table_view:
79 return BindingStorageKind::cam_switch_tables;
80 case binding_type::mc_cam_switch_outputs_view:
81 return BindingStorageKind::cam_switch_outputs;
82 case binding_type::mc_cam_track_options_view:
83 return BindingStorageKind::cam_track_options;
84 case binding_type::mc_cam_table_view:
85 return BindingStorageKind::cam_tables;
86 case binding_type::mc_time_position:
87 return BindingStorageKind::position_profiles;
88 case binding_type::mc_time_velocity:
89 return BindingStorageKind::velocity_profiles;
90 case binding_type::mc_time_acceleration:
91 return BindingStorageKind::acceleration_profiles;
92 case binding_type::mc_kin_transform_ref:
93 return BindingStorageKind::kin_transforms;
94 default: return BindingStorageKind::none;
95 }
96}
