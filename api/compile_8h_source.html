75inline void finalize_debug_symbols(Program &artifact)
76{
77 artifact.symbols.clear();
78 for(const ConfigurationInfo &configuration : artifact.configurations) {
79 for(const ResourceInfo &resource : configuration.resources) {
80 for(const ProgramMappingInfo &mapping : resource.mappings) {
81 const Program *image = nullptr;
82 if(artifact.programs.empty()) {
83 if(artifact.program_name == mapping.program)
84 image = &artifact;
85 } else {
86 for(const Program &candidate : artifact.programs)
87 if(candidate.program_name == mapping.program) {
88 image = &candidate;
89 break;
90 }
91 }
92 if(image == nullptr) continue;
93 for(const VarInfo &var : image->vars) {
94 SymbolInfo symbol;
95 symbol.configuration = configuration.lower;
96 symbol.resource = resource.lower;
97 symbol.instance = mapping.lower;
98 symbol.program = mapping.program;
99 symbol.variable = var.lower;
100 symbol.qualified_name = configuration.lower + "." +
101 resource.lower + "." + mapping.lower + "." +
102 mapping.program + "." + var.lower;
103 symbol.id = stable_symbol_id(symbol.qualified_name);
104 symbol.type_id = var.type_id;
105 symbol.offset = var.offset;
106 const TypeDesc *desc = image->types.get(var.type_id);
107 symbol.size = desc == nullptr
108 ? 0
109 : static_cast<std::uint32_t>(desc->size);
110 for(const LocatedVarInfo &located :
111 image->process_image.variables) {
112 if(located.lower == var.lower &&
113 located.var_offset == var.offset) {
114 symbol.located = true;
115 symbol.size = located.byte_width;
116 symbol.physical_area =
117 static_cast<std::uint8_t>(located.area);
118 symbol.physical_byte_offset = located.byte_offset;
119 symbol.physical_bit = located.bit;
120 symbol.physical_width = located.byte_width;
121 symbol.physical_bit_address =
122 located.bit_address;
123 break;
124 }
125 }
126 artifact.symbols.push_back(
127 static_cast<SymbolInfo &&>(symbol));
128 }
129 }
130 }
131 }

154inline CompileResult compile_single_program_impl(
155 std::string_view source,
156 const CompileOptions &options,
157 bool trusted_debug_provenance)
158{
159 CompileResult result;
160 Parser parser(source, options.max_diagnostics, options.max_nesting,
161 trusted_debug_provenance);
162 ParseResult parsed = parser.parse();
163 result.diagnostics =
164 static_cast<std::vector<Diagnostic> &&>(parsed.diagnostics);
165
166 SemaLimits sema_limits;
167 sema_limits.max_vars_bytes = options.max_vars_bytes;
168 sema_limits.max_fb_instances =
169 options.max_fb_instances < kMaxFbInstances
170 ? options.max_fb_instances
171 : kMaxFbInstances;
172 sema_limits.max_axis_refs =
173 options.max_axis_refs < kMaxAxisBindings
174 ? options.max_axis_refs
175 : kMaxAxisBindings;
176 sema_limits.max_group_refs =
177 options.max_group_refs < kMaxGroupBindings
178 ? options.max_group_refs
179 : kMaxGroupBindings;
180 sema_limits.max_diagnostics = options.max_diagnostics;
181 sema_limits.max_user_types = options.max_user_types;
182 sema_limits.max_enum_members = options.max_enum_members;
183 sema_limits.max_type_name_bytes = options.max_type_name_bytes;
184 sema_limits.max_array_elements = options.max_array_elements;
185 sema_limits.max_struct_fields = options.max_struct_fields;
186 sema_limits.max_aggregate_depth = options.max_aggregate_depth;
187 sema_limits.max_input_image_bytes = options.max_input_image_bytes;
188 sema_limits.max_output_image_bytes = options.max_output_image_bytes;
189 sema_limits.max_memory_image_bytes = options.max_memory_image_bytes;
190 sema_limits.max_retain_entries = options.max_retain_entries;
191 sema_limits.max_force_entries = options.max_force_entries;
192 Sema sema(parsed.ast, result.diagnostics, sema_limits);
193 SemaResult analyzed = sema.run();
194
195 if(!parsed.ok || !analyzed.ok) {
196 return result;
197 }
198
199 CodegenLimits codegen_limits;
200 codegen_limits.max_code_bytes = options.max_code_bytes;
201 codegen_limits.max_vars_bytes = options.max_vars_bytes;
202 codegen_limits.max_stack_slots = options.max_stack_slots;
203 codegen_limits.max_diagnostics = options.max_diagnostics;
204 codegen_limits.source_name = options.source_name;
205 codegen_limits.debug_mode = options.debug_mode;
206 codegen_limits.debug_line_offset = options.debug_line_offset;
207 codegen_limits.debug_pou = options.debug_pou;
208 codegen_limits.debug_call_depth = options.debug_call_depth;
209 codegen_limits.debug_require_provenance =
210 options.debug_require_provenance;
211 Codegen codegen(parsed.ast, analyzed, result.diagnostics, codegen_limits);
212 if(!codegen.run(result.program)) {
213 return result;
214 }
215 // The snapshot schema is tied to the complete current-source program
216 // manifest. Fingerprint is deliberately not included in its own input.
217 const std::string manifest = result.program.canonical_manifest();
218 std::uint64_t fingerprint = 1469598103934665603ULL;
219 for(unsigned char byte : manifest) {
220 fingerprint ^= byte;
221 fingerprint *= 1099511628211ULL;
222 }
223 result.program.process_image.fingerprint = fingerprint;
224 result.ok = true;
225 return result;
226}

242inline CompileResult compile_without_sfc(
243 std::string_view source,
244 const CompileOptions &options = CompileOptions{})
245{
246 const std::string lower = l2b_detail::lower_copy(source);
247 const bool has_program = tasking_detail::has_program_declaration(source);
248 const bool has_configuration =
249 tasking_detail::source_contains_token(source, "configuration");
250 const bool has_library_pou =
251 l2b_detail::contains_word(lower, "function") ||
252 l2b_detail::contains_word(lower, "function_block");
253 const bool has_object_extension =
254 l2b_detail::contains_word(lower, "method") ||
255 l2b_detail::contains_word(lower, "interface") ||
256 l2b_detail::contains_word(lower, "extends") ||
257 l2b_detail::contains_word(lower, "generic") ||
258 l2b_detail::contains_word(lower, "ref_to");
259 if(has_library_pou && !has_program && !has_object_extension &&
260 !l2b_detail::contains_word(lower, "var_global") &&
261 !l2b_detail::contains_word(lower, "var_external")) {
262 return compile_single_program(source, options);
263 }
264 const bool project = l2b_detail::contains_word(lower, "function") ||
265 l2b_detail::contains_word(lower, "function_block") ||
266 (has_program && has_configuration) ||
267 l2b_detail::contains_word(lower, "var_global") ||
268 l2b_detail::contains_word(lower, "var_external") ||
269 l2b_detail::contains_word(lower, "method") ||
270 l2b_detail::contains_word(lower, "interface") ||
271 l2b_detail::contains_word(lower, "extends") ||
272 l2b_detail::contains_word(lower, "generic") ||
273 l2b_detail::contains_word(lower, "ref_to") ||
274 lower.find("end_program") !=
275 lower.rfind("end_program");
276 std::string masked_source;
277 if(project && has_configuration)
278 masked_source = tasking_detail::mask_configuration_blocks(source);
279 const std::string_view compile_source = masked_source.empty()
280 ? source
281 : std::string_view(masked_source);
282 CompileResult result = project
283 ? l2b_detail::compile_project(compile_source,
284 options)
285 : compile_single_program(source, options);
286 if(!has_configuration) return result;
287
288 const char *task_controls[] = {"reset_task", "create_task", "delete_task"};
289 for(const char *control : task_controls) {
290 std::size_t offset = 0;
291 if(tasking_detail::source_contains_call(source, control, offset)) {
292 tasking_detail::add_diagnostic(
293 result.diagnostics, options.max_diagnostics,
294 DiagCode::unsupported_l5_task_control, source, offset);
295 result.ok = false;
296 }
297 }
298 if(!result.ok) return result;
299
300 tasking_detail::ParseLimits limits;
301 limits.base_tick_ns = options.base_tick_ns;
302 limits.max_configurations = options.max_configurations;
303 limits.max_resources = options.max_resources;
304 limits.max_tasks = options.max_tasks_per_resource;
305 limits.max_program_mappings = options.max_program_mappings_per_task;
306 limits.max_diagnostics = options.max_diagnostics;
307 if(!tasking_detail::parse_configurations(
308 source, limits, result.program.configurations,
309 result.diagnostics)) {
310 result.ok = false;
311 return result;
312 }
313
314 for(const ConfigurationInfo &configuration :
315 result.program.configurations) {
316 std::vector<bool> mapped(result.program.programs.size(), false);
317 for(const ResourceInfo &resource : configuration.resources) {
318 for(const ProgramMappingInfo &mapping : resource.mappings) {
319 bool found = false;
320 for(std::size_t index = 0;
321 index < result.program.programs.size(); ++index) {
322 if(result.program.programs[index].program_name ==
323 mapping.program) {
324 mapped[index] = true;
325 found = true;
326 break;
327 }
328 }
329 if(!found) {
330 tasking_detail::add_diagnostic(
331 result.diagnostics, options.max_diagnostics,
332 DiagCode::sema_unknown_identifier, source,
333 mapping.source_offset);
334 result.ok = false;
335 }
336 }
337 }
338 for(std::size_t index = 0; index < mapped.size(); ++index) {
339 if(mapped[index]) continue;
340 tasking_detail::add_diagnostic(
341 result.diagnostics, options.max_diagnostics,
342 DiagCode::warning_program_unmapped, source,
343 configuration.source_offset);
344 }
345 }
346 return result;
347}

359inline std::uint32_t align8(std::uint32_t value)
360{
361 return (value + 7U) & ~std::uint32_t{7U};

364inline bool add_bytes(std::uint32_t &cursor, std::uint64_t count)
365{
366 if(count > std::numeric_limits<std::uint32_t>::max() - cursor)
367 return false;
368 cursor += static_cast<std::uint32_t>(count);
369 return true;

372inline bool lay_out_runtime(Program &program)
373{
374 std::uint32_t cursor = align8(program.vars_bytes);
375 if(!add_bytes(cursor, align8(program.vars_bytes))) return false;
376 for(SfcNetworkInfo &network : program.sfc_networks) {
377 network.runtime_offset = cursor;
378 network.committed_active_offset = cursor;
379 if(!add_bytes(cursor, network.steps.size())) return false;
380 network.staged_active_offset = cursor;
381 if(!add_bytes(cursor, network.steps.size())) return false;
382 network.shadow_active_offset = cursor;
383 if(!add_bytes(cursor, network.steps.size())) return false;
384 network.firing_offset = cursor;
385 if(!add_bytes(cursor, network.transitions.size())) return false;
386 cursor = align8(cursor);
387 network.committed_block_offset = cursor;
388 if(!add_bytes(cursor, network.action_blocks.size() * 16ULL))
389 return false;
390 network.staged_block_offset = cursor;
391 if(!add_bytes(cursor, network.action_blocks.size() * 16ULL))
392 return false;
393 network.shadow_block_offset = cursor;
394 if(!add_bytes(cursor, network.action_blocks.size() * 16ULL))
395 return false;
396 network.committed_action_offset = cursor;
397 if(!add_bytes(cursor, network.actions.size())) return false;
398 network.staged_action_offset = cursor;
399 if(!add_bytes(cursor, network.actions.size())) return false;
400 network.shadow_action_offset = cursor;
401 if(!add_bytes(cursor, network.actions.size())) return false;
402 network.direct_action_offset = cursor;
403 if(!add_bytes(cursor, network.actions.size())) return false;
404 cursor = align8(cursor);
405 network.static_bytes = cursor - network.runtime_offset;
406 }
407 program.sfc_runtime_bytes = cursor;
408 return true;
409}

411inline bool merge_region_layout(Program &base, const Program &region)
412{
413 if(base.fb_bytes != region.fb_bytes ||
414 base.fbs.size() != region.fbs.size()) return false;
415 for(const VarInfo &var : base.vars) {
416 const VarInfo *matching = nullptr;
417 for(const VarInfo &candidate : region.vars) {
418 if(candidate.lower == var.lower) {
419 matching = &candidate;
420 break;
421 }
422 }
423 if(matching == nullptr && var.lower.rfind("l2b", 0) == 0) continue;
424 if(matching == nullptr || matching->offset != var.offset ||
425 matching->type_id != var.type_id) return false;
426 }
427 if(region.vars_bytes > base.vars_bytes) {
428 base.vars_bytes = region.vars_bytes;
429 base.initial_data.resize(base.vars_bytes, 0);
430 }
431 return true;
432}

477inline bool build_network(const sfc_detail::NetworkDraft &draft,
478 SfcNetworkInfo &info)
479{
480 info.name = draft.decl.name;
481 info.lower = draft.decl.lower;
482 info.step_count = static_cast<std::uint32_t>(draft.decl.steps.size());
483 info.reachable_step_count = info.step_count;
484 info.max_parallel_active_steps = draft.max_parallel_active_steps;
485 info.worst_case_transition_evaluations =
486 static_cast<std::uint32_t>(draft.decl.transitions.size());
487 std::map<std::string, std::uint16_t> step_indices;
488 std::map<std::string, std::uint16_t> action_indices;
489 for(std::size_t index = 0; index < draft.decl.steps.size(); ++index) {
490 const SfcStepDecl &decl = draft.decl.steps[index];
491 SfcStepInfo step;
492 step.name = decl.name;
493 step.lower = decl.lower;
494 step.initial = decl.initial;
495 step.terminal = decl.terminal;
496 step.line = decl.line;
497 step.column = decl.column;
498 step.end_line = decl.end_line;
499 step.end_column = decl.end_column;
500 step_indices[step.lower] = static_cast<std::uint16_t>(index);
501 info.steps.push_back(static_cast<SfcStepInfo &&>(step));
502 }
503 for(std::size_t index = 0; index < draft.decl.actions.size(); ++index) {
504 const SfcActionDecl &decl = draft.decl.actions[index];
505 SfcActionInfo action;
506 action.name = decl.name;
507 action.lower = decl.lower;
508 action.line = decl.line;
509 action.column = decl.column;
510 action.end_line = decl.end_line;
511 action.end_column = decl.end_column;
512 action_indices[action.lower] = static_cast<std::uint16_t>(index);
513 info.actions.push_back(static_cast<SfcActionInfo &&>(action));
514 }
515 for(const SfcTransitionDecl &decl : draft.decl.transitions) {
516 SfcTransitionInfo transition;
517 for(const std::string &name : decl.sources)
518 transition.sources.push_back(step_indices[name]);
519 for(const std::string &name : decl.targets)
520 transition.targets.push_back(step_indices[name]);
521 transition.simultaneous = decl.simultaneous;
522 transition.line = decl.line;
523 transition.column = decl.column;
524 transition.end_line = decl.end_line;
525 transition.end_column = decl.end_column;
526 info.transitions.push_back(static_cast<SfcTransitionInfo &&>(transition));
527 }
528 for(std::size_t step_index = 0;
529 step_index < draft.decl.steps.size(); ++step_index) {
530 for(const SfcActionBlockDecl &decl :
531 draft.decl.steps[step_index].actions) {
532 SfcActionBlockInfo block;
533 block.step = static_cast<std::uint16_t>(step_index);
534 block.action = action_indices[decl.lower];
535 block.qualifier = static_cast<std::uint8_t>(decl.qualifier);
536 block.duration_ns = decl.duration_ns;
537 block.line = decl.line;
538 block.column = decl.column;
539 block.end_line = decl.end_line;
540 block.end_column = decl.end_column;
541 info.action_blocks.push_back(block);
542 }
543 }
544 for(const sfc_detail::NetworkDraft::ParallelRegion &draft_region :
545 draft.parallel_regions) {
546 SfcParallelRegionInfo region;
547 region.divergence_transition =
548 static_cast<std::uint16_t>(draft_region.divergence);
549 if(draft_region.convergence !=
550 std::numeric_limits<std::size_t>::max())
551 region.convergence_transition =
552 static_cast<std::uint16_t>(draft_region.convergence);
553 for(const std::string &entry :
554 draft.decl.transitions[draft_region.divergence].targets)
555 region.branch_entries.push_back(step_indices[entry]);
556 for(std::size_t exit : draft_region.exits)
557 region.branch_exits.push_back(static_cast<std::uint16_t>(exit));
558 info.parallel_regions.push_back(
559 static_cast<SfcParallelRegionInfo &&>(region));
560 }
561 info.worst_case_action_executions =
562 static_cast<std::uint32_t>(info.actions.size());
563 return true;
564}

566inline void add_sfc_cost(Program &program)
567{
568 std::uint64_t cost = program.worst_case_instructions;
569 bool bounded = program.worst_case_bounded;
570 const std::uint64_t fixed_cost = program.sfc_runner_fixed_cost();
571 if(fixed_cost == std::numeric_limits<std::uint64_t>::max() ||
572 cost > std::numeric_limits<std::uint64_t>::max() - fixed_cost)
573 bounded = false;
574 else
575 cost += fixed_cost;
576 for(const SfcNetworkInfo &network : program.sfc_networks) {
577 for(const SfcTransitionInfo &transition : network.transitions) {
578 bounded = bounded && transition.condition.worst_case_bounded;
579 if(cost > std::numeric_limits<std::uint64_t>::max() -
580 transition.condition.worst_case_instructions) {
581 bounded = false;
582 break;
583 }
584 cost += transition.condition.worst_case_instructions;
585 }
586 for(const SfcActionInfo &action : network.actions) {
587 bounded = bounded && action.region.worst_case_bounded;
588 if(cost > std::numeric_limits<std::uint64_t>::max() -
589 action.region.worst_case_instructions) {
590 bounded = false;
591 break;
592 }
593 cost += action.region.worst_case_instructions;
594 }
595 }
596 program.worst_case_bounded = bounded;
597 program.worst_case_instructions = bounded ? cost : 0;
598}

602inline CompileResult compile(std::string_view source,
603 const CompileOptions &options = CompileOptions{})
604{
605 sfc_detail::Prepared prepared = sfc_detail::prepare(source, options);
606 if(!prepared.present) {
607 CompileResult plain = compile_without_sfc(source, options);
608 if(plain.ok) {
609 finalize_debug_symbols(plain.program);
610 finalize_debug_instruction_ids(plain.program);
611 }
612 return plain;
613 }
614 CompileResult result;
615 result.diagnostics = prepared.diagnostics;
616 if(!prepared.ok) return result;
617
618 const std::string base_source = sfc_detail::render(
619 prepared, std::numeric_limits<std::size_t>::max(),
620 sfc_detail::RegionKind::none, 0);
621 result = compile_without_sfc(base_source, options);
622 if(!result.ok) return result;
623
624 for(std::size_t network_index = 0;
625 network_index < prepared.networks.size(); ++network_index) {
626 const sfc_detail::NetworkDraft &draft = prepared.networks[network_index];
627 Program *target = sfc_detail::select_program(
628 result.program, draft.decl.program_lower);
629 if(target == nullptr) {
630 sfc_detail::add_diag(result.diagnostics,
631 DiagCode::sema_unknown_identifier,
632 source, draft.begin);
633 result.ok = false;
634 return result;
635 }
636 SfcNetworkInfo info;
637 sfc_compile_detail::build_network(draft, info);
638 const VarInfo *result_var = sfc_detail::find_var(
639 *target, "sfc6_region_result");
640 if(result_var == nullptr) {
641 sfc_detail::add_diag(result.diagnostics,
642 DiagCode::sema_unknown_identifier,
643 source, draft.begin);
644 result.ok = false;
645 return result;
646 }
647 for(std::size_t index = 0; index < info.transitions.size(); ++index) {
648 CompileResult image = compile_without_sfc(
649 sfc_detail::render(prepared, network_index,
650 sfc_detail::RegionKind::transition, index),
651 options);
652 Program *region = image.ok ? sfc_detail::select_program(
653 image.program, draft.decl.program_lower) : nullptr;
654 if(region == nullptr ||
655 !sfc_compile_detail::merge_region_layout(*target, *region)) {
656 result.diagnostics.insert(result.diagnostics.end(),
657 image.diagnostics.begin(),
658 image.diagnostics.end());
659 if(image.diagnostics.empty())
660 sfc_detail::add_diag(result.diagnostics,
661 DiagCode::sema_unsafe_sfc_network,
662 source, draft.begin);
663 result.ok = false;
664 return result;
665 }
666 sfc_compile_detail::copy_region(
667 *region, info.transitions[index].condition);
668 sfc_compile_detail::bind_region_source_map(
669 *target, info.transitions[index].condition,
670 draft.decl.lower.c_str(),
671 static_cast<std::uint32_t>(
672 std::max(0, draft.decl.transitions[index].condition_line)),
673 static_cast<std::uint32_t>(
674 std::max(0, draft.decl.transitions[index].condition_column)),
675 static_cast<std::uint32_t>(1U + index),
676 DebugRegionKind::sfc_transition);
677 target->stack_slots = std::max(target->stack_slots,
678 region->stack_slots);
679 info.transitions[index].condition.result_offset =
680 result_var->offset;
681 }
682 for(std::size_t index = 0; index < info.actions.size(); ++index) {
683 CompileResult image = compile_without_sfc(
684 sfc_detail::render(prepared, network_index,
685 sfc_detail::RegionKind::action, index),
686 options);
687 Program *region = image.ok ? sfc_detail::select_program(
688 image.program, draft.decl.program_lower) : nullptr;
689 if(region == nullptr ||
690 !sfc_compile_detail::merge_region_layout(*target, *region)) {
691 result.diagnostics.insert(result.diagnostics.end(),
692 image.diagnostics.begin(),
693 image.diagnostics.end());
694 if(image.diagnostics.empty())
695 sfc_detail::add_diag(result.diagnostics,
696 DiagCode::sema_unsafe_sfc_network,
697 source, draft.begin);
698 result.ok = false;
699 return result;
700 }
701 sfc_compile_detail::copy_region(*region,
702 info.actions[index].region);
703 sfc_compile_detail::bind_region_source_map(
704 *target, info.actions[index].region,
705 draft.decl.lower.c_str(),
706 static_cast<std::uint32_t>(
707 std::max(0, draft.decl.actions[index].body_line)),
708 static_cast<std::uint32_t>(
709 std::max(0, draft.decl.actions[index].body_column)),
710 static_cast<std::uint32_t>(
711 1U + draft.decl.transitions.size() + index),
712 DebugRegionKind::sfc_action);
713 target->stack_slots = std::max(target->stack_slots,
714 region->stack_slots);
715 }
716 target->sfc_networks.push_back(static_cast<SfcNetworkInfo &&>(info));
717 }
718
719 const auto finish = [&](Program &program) -> bool {
720 if(!sfc_compile_detail::lay_out_runtime(program)) return false;
721 sfc_compile_detail::add_sfc_cost(program);
722 program.process_image.fingerprint = 0;
723 const std::string manifest = program.canonical_manifest();
724 std::uint64_t fingerprint = 1469598103934665603ULL;
725 for(unsigned char byte : manifest) {
726 fingerprint ^= byte;
727 fingerprint *= 1099511628211ULL;
728 }
729 program.process_image.fingerprint = fingerprint;
730 return true;
731 };
732 if(result.program.programs.empty()) {
733 if(!finish(result.program)) {
734 sfc_detail::add_diag(result.diagnostics, DiagCode::capacity_exceeded,
735 source, 0);
736 result.ok = false;
737 }
738 } else {
739 for(Program &program : result.program.programs) {
740 if(!finish(program)) {
741 sfc_detail::add_diag(result.diagnostics,
742 DiagCode::capacity_exceeded, source, 0);
743 result.ok = false;
744 return result;
745 }
746 }
747 Program aggregate = static_cast<Program &&>(result.program);
748 Program root = aggregate.programs.front();
749 root.programs = static_cast<std::vector<Program> &&>(
750 aggregate.programs);
751 root.configurations = static_cast<std::vector<ConfigurationInfo> &&>(
752 aggregate.configurations);
753 root.pous = static_cast<std::vector<PouInfo> &&>(aggregate.pous);
754 root.instances = static_cast<std::vector<InstanceInfo> &&>(
755 aggregate.instances);
756 root.max_call_depth = aggregate.max_call_depth;
757 root.max_instance_depth = aggregate.max_instance_depth;
758 root.layout_bytes = aggregate.layout_bytes;
759 result.program = static_cast<Program &&>(root);
760 }
761 if(result.ok) {
762 finalize_debug_symbols(result.program);
763 finalize_debug_instruction_ids(result.program);
764 }
765 return result;
766}

768enum class IncrementalStatus : std::uint8_t
769{
770 ok = 0,
771};

773class IncrementalCompiler
774{
775public:
776 IncrementalStatus update_pou(std::string name, std::string source)
777 {
778 lower_in_place(name);
779 fragments_[name] = static_cast<std::string &&>(source);
780 return IncrementalStatus::ok;
781 }
782
783 CompileResult compile()
784 {
785 // The public unit of update is a POU. L0 semantics 2.3 explicitly
786 // permits a clean full reparse; the contract is semantic equivalence,
787 // not an IDE performance claim. std::map order keeps assembly stable.
788 std::string project;
789 for(const auto &fragment : fragments_) {
790 project += fragment.second;
791 project.push_back('\n');
792 }
793 return st::compile(project);
794 }
795
796private:
797 static void lower_in_place(std::string &text)
798 {
799 for(char &c : text) {
800 if(c >= 'A' && c <= 'Z') {
801 c = static_cast<char>(c - 'A' + 'a');
802 }
803 }
804 }
805
806 std::map<std::string, std::string> fragments_;
807};
